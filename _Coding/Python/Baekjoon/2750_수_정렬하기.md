```python
n = int(input())  
lst = [int(input()) for _ in range(n)]  
for i in range(n-1):  
    # 가장 작은 값을 앞에 세우고 시작 해야 할 것 같은데?  
    for j in range(i+1, n):  
        while lst[i] > lst[j]:  
            lst[i], lst[j] = lst[j], lst[i]  
            # 그냥 돌리면 [2, 3, 4, 1, 5]가 나오고, 1은 5보다 작으니 그냥 그 자리에 있음  
            # 아니면 돌리다가 작으면 계속 앞이랑 비교 하는 코드를 넣자  
            # n개 중 2개 선택 하는 모든 조합과도 같음  
            # 루프, 백트래킹  
for k in lst:  
    print(k)
```
