### 코드 개선 방법
- [구현] 수식 대신 방향을 정하고 대각선 4방향으로 한 칸씩 가기. 직관적
- [중복] 먹은 디저트는 set
- [시간 복잡도] 20**4 = 16만

### 코드
```python
# S2105 디저트 카페

# 함수
def is_range(r, c):
    return 0 <= r < size and 0 <= c < size

# 1. 경로 찾기
def find_route(garo, sero, jwa):
    sr, sc = jwa
    route = set()
    fail = 0
    for i in range(4):
        if fail:
            break
        else:
            # 1. 대각선 왼아래 (1, -1)
            if i == 0:
                for g1 in range(garo):
                    nr, nc = sr + 1, sc - 1
                    if is_range(nr, nc):
                        num = arr[nr][nc]
                        if num in route:
                            fail = 1
                            break

                        else:
                            route.add(num)
                            sr, sc = nr, nc
                    else:
                        fail = 1
                        break

            # 2. 대각선 오른아래 (1, 1)
            elif i == 1:
                for s1 in range(sero):
                    nr, nc = sr + 1, sc + 1
                    if is_range(nr, nc):
                        num = arr[nr][nc]
                        if num in route:
                            fail = 1
                            break

                        else:
                            route.add(num)
                            sr, sc = nr, nc
                    else:
                        fail = 1
                        break

            # 3. 대각선 오른위 (-1, 1)
            elif i == 2:
                for g2 in range(garo):
                    nr, nc = sr - 1, sc + 1
                    if is_range(nr, nc):
                        num = arr[nr][nc]
                        if num in route:
                            fail = 1
                            break

                        else:
                            route.add(num)
                            sr, sc = nr, nc
                    else:
                        fail = 1
                        break

            # 4. 대각선 왼위 (-1, -1)
            elif i == 3:
                for s2 in range(sero):
                    nr, nc = sr - 1, sc - 1
                    if is_range(nr, nc):
                        num = arr[nr][nc]
                        if num in route:
                            fail = 1
                            break

                        else:
                            route.add(num)
                            sr, sc = nr, nc
                    else:
                        fail = 1
                        break

    if not fail:
        return len(route)
    else:
        return -1

# 2. find_route 포함 while문 돌기
def find():

    ans = -1
    for garo in range(1, size):
        for sero in range(1, size):
            for i in range(size):
                for j in range(size):
                    pos = find_route(garo, sero, (i, j))
                    ans = max(ans, pos)

    return ans

# 입력
T = int(input())
for tc in range(1, 1+T):
    size = int(input())
    arr = [list(map(int, input().split())) for _ in range(size)]

    print(f"#{tc}", find())
```
