### 코드

def su(n):
    # base case도 여기, 큰 수에서 작은 수 역으로 들어감
    if n == 1:
        return 1
    else:
        return su(n//2) + su(n-1)
    # 주루루룩 가서 밑 바닥 찍고 return 하면서 더하는?


T = int(input())
for tc in range(1, T+1):
    goal = int(input())
    print(f"#{tc}", su(goal))
