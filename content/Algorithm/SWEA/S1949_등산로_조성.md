### 실패 / 성공 이유
- [문제 읽기] '최대' 깊이 K까지가 아니라 최대 K만큼. 전에 틀렸던 부분이라 13분 만에 찾음
- [문제 읽기] 가장 높은 봉우리는 깎기 전 기준. 깎은 뒤 기준으로 생각

### 코드 개선 방법
- [완전탐색] 칸마다 1~K 깎아보고 봉우리들에서 경로 탐색
- [습관] 답이 안 나오면 내 해석이 틀렸을 가능성으로 생각을 전환

from collections import deque


# 함수
def is_range(r, c):
    return 0 <= r < size and 0 <= c < size


# 1. 최고값 좌표 반환
def highest(array):
    mx = 0
    lst = []
    for r in range(size):
        for c in range(size):
            if array[r][c] >= mx:
                if array[r][c] > mx:
                    mx = array[r][c]
                    lst = []
                lst.append((r, c))

    return lst


# 2. bfs로 경사로 찾기
def find_route(sr, sc):
    v = [[0] * size for _ in range(size)]
    v[sr][sc] = 1
    q = deque()
    q.append((sr, sc))
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if is_range(nr, nc) and v[nr][nc] < v[r][c] + 1:
                if arr[nr][nc] < arr[r][c]:
                    q.append((nr, nc))
                    v[nr][nc] = v[r][c] + 1

    mx = 0
    for rr in v:
        mx = max(mx, max(rr))
    return mx


def cut(start):
    ans = 0
    for d in range(1, 1 + depth):
        for r in range(size):
            for c in range(size):
                arr[r][c] -= d
                for pr, pc in start:
                    route = find_route(pr, pc)
                    ans = max(ans, route)
                arr[r][c] += d

    return ans


# 입력
T = int(input())
for tc in range(1, 1 + T):
    size, depth = map(int, input().split())
    arr = [list(map(int, input().split())) for _ in range(size)]

    start = highest(arr)
    ans = cut(start)
    print(f"#{tc}", ans)
