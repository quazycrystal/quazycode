### 걸린 시간: 1시간 12분 (구상 19분, 구현 38분, 디버깅 15분) (실패 횟수: -)

### 실패 / 성공 이유
- [좌표] 방향 바꾸는 함수에서 r, c를 바꿔 적음
- [초기화] 시작점 arr[i][j]를 다시 열어주지 않음

### 코드 개선 방법
- [룩업 테이블] 블록 번호 × 들어온 방향 → 나가는 방향 매핑
- [자료구조] 웜홀은 이중 리스트에 넣고 dict로 쌍을 엮기
- [종료 조건] 블랙홀 또는 시작점으로 돌아오면 끝

### 코드
```python
# 함수

def is_range(r, c):
    return 0 <= r < size and 0 <= c < size


direction = {1: (-1, 0), 2: (1, 0), 3: (0, -1), 4: (0, 1)}
# 상하좌우

# 방향 매핑
one = {1: 2, 2: 4, 3: 1, 4: 3}
two = {1: 4, 2: 1, 3: 2, 4: 3}
three = {1: 3, 2: 1, 3: 4, 4: 2}
four = {1: 2, 2: 3, 3: 4, 4: 1}
five = {1: 2, 2: 1, 3: 4, 4: 3}


def go(sr, sc, d):
    dr, dc = direction[d]
    r, c = sr, sc
    t = 0
    while True:
        r, c = r + dr, c + dc
        if not is_range(r, c):
            t = 5
            break

        elif arr[r][c] > 0:
            t = arr[r][c]
            break

        elif arr[r][c] == -1:
            t = -1
            break

    return (r, c, t, d)


# 2. 방향 바꿔준다봉
def change(sr, sc, t, d):
    stop = 0
    global score
    if 0 < t < 6:
        score += 1

    if t == 1:
        d = one[d]
    elif t == 2:
        d = two[d]
    elif t == 3:
        d = three[d]
    elif t == 4:
        d = four[d]
    elif t == 5:
        d = five[d]
    elif t == 6:
        sr, sc = six[(sr, sc)]
    elif t == 7:
        sr, sc = seven[(sr, sc)]
    elif t == 8:
        sr, sc = eight[(sr, sc)]
    elif t == 9:
        sr, sc = nine[(sr, sc)]
    elif t == 10:
        sr, sc = ten[(sr, sc)]

    elif t == -1:
        stop = 1

    return stop, sr, sc, d


# 입력
T = int(input())

for tc in range(1, 1 + T):
    size = int(input())
    arr = [list(map(int, input().split())) for _ in range(size)]
    # 포탈 정보 뽑기
    temp = []
    six = {}
    seven = {}
    eight = {}
    nine = {}
    ten = {}
    for r in range(size):
        for c in range(size):
            if 5 < arr[r][c] < 11:
                temp.append((arr[r][c], r, c))

    temp.sort()
    while temp:
        a = temp.pop()
        b = temp.pop()
        num, sr, sc = a
        _, er, ec = b

        if num == 6:
            six[(sr, sc)] = (er, ec)
            six[(er, ec)] = (sr, sc)
        elif num == 7:
            seven[(sr, sc)] = (er, ec)
            seven[(er, ec)] = (sr, sc)
        elif num == 8:
            eight[(sr, sc)] = (er, ec)
            eight[(er, ec)] = (sr, sc)
        elif num == 9:
            nine[(sr, sc)] = (er, ec)
            nine[(er, ec)] = (sr, sc)
        elif num == 10:
            ten[(sr, sc)] = (er, ec)
            ten[(er, ec)] = (sr, sc)

    # 풀이
    ans = 0
    for i in range(size):
        for j in range(size):
            if arr[i][j] == 0:
                arr[i][j] = -1  # 시작점 다시 와도 끝
                for dirr in range(1, 5):
                    # 여기서 시작점, 방향 결정
                    rr, cc, dd = i, j, dirr
                    score = 0
                    while True:
                        nr, nc, nt, nd = go(rr, cc, dd)
                        stop, nnr, nnc, nnd = change(nr, nc, nt, nd)
                        if stop:
                            break
                        else:
                            rr, cc, dd = nnr, nnc, nnd

                    ans = max(ans, score)

                arr[i][j] = 0  # 다시 열어줘야

    print(f"#{tc}", ans)
```
