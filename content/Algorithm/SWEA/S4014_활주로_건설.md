### 코드
```python
# S 활주로 건설

# 함수
def find_r(line):
    using = [0] * size

    cnt = 0
    prev = line[0]
    result = 0
    for right in range(size):
        now = line[right]
        if now == prev:
            cnt += 1
        elif now > prev:
            if now == prev + 1 and cnt >= length:
                cnt = 1
                prev = now
                for put in range(right - 1, right - 1 - length, -1):
                    if using[put] == 0:
                        using[put] = 1
                    else:
                        return result, -1
            else:
                return result, -1
        else:
            cnt = 1
            prev = now

    return result + 1, using

def find_l(line, using):
    cnt = 0
    prev = line[-1]
    result = 0
    for left in range(size-1, -1, -1):
        now = line[left]
        if now == prev:
            cnt += 1
        elif now > prev:
            if now == prev + 1 and cnt >= length:
                cnt = 1
                prev = now
                for put in range(left + 1, left + 1 + length):
                    if using[put] == 0:
                        using[put] = 1
                    else:
                        return result, -1
            else:
                return result, using
        else:
            cnt = 1
            prev = now

    return result + 1, using

# 풀이
T = int(input())
for tc in range(1, 1+T):
    # 입력
    size, length = map(int, input().split())
    arr = [list(map(int, input().split())) for _ in range(size)]
    ans = 0

    for row in arr:
        r_rr, r_use = find_r(row)
        if r_use != -1:
            r_ll, r_use = find_l(row, r_use)
            if r_rr and r_ll:
                ans += 1

    for c in range(size):
        col = []
        for r in range(size):
            col.append(arr[r][c])

        c_rr, c_use = find_r(col)

        if c_use != -1:
            c_ll, c_use = find_l(col, c_use)

            if c_rr and c_ll:
                ans += 1

    print(f"#{tc}", ans)



```
