### 코드 개선 방법
- [상대 좌표] 충전 범위 마름모는 상대 좌표로 그리기
- [분기] A, B가 같은 영역인지 따로 분기하지 않고 (BC_A, BC_B) 중복 순열로 최댓값, 같은 BC면 한 번만
- [시간] 0초부터 계산하므로 들어오자마자 계산
- [예외] 충전소 없는 칸 처리

### 코드
```python
from pprint import *


# 0. 범위 체크
def is_range(r, c):
    return 0 <= r < 10 and 0 <= c < 10


# 1. arr에 네모 그리는 함수
def draw_bc(r, c, size, idx):
    # 항상 상대 좌표로!
    for up in range(size + 1):
        ur = r + up
        ug = size - up
        for side in range(c - ug, c + ug + 1):
            if is_range(ur, side):
                arr[ur][side].append(idx)

    for down in range(1, size + 1):
        dr = r - down
        dg = size - down
        for side in range(c - dg, c + dg + 1):
            if is_range(dr, side):
                arr[dr][side].append(idx)


# 2. 가래는 대로 이동하는 함수:
direction = {0: (0, 0), 1: (-1, 0), 2: (0, 1), 3: (1, 0), 4: (0, -1)}


def ab_move(r1, c1, r2, c2):
    global ans
    ar, ac = r1, c1
    br, bc = r2, c2

    for idx in range(time + 1):  # 0초부터 계산하므로

        A = arr[ar][ac] if arr[ar][ac] else [bc_num]  # 마지막에는 power 0 저장됨
        B = arr[br][bc] if arr[br][bc] else [bc_num]

        mx_sm = 0
        sm = 0
        for a in A:  # 중복 순열 -
            for b in B:
                if a != b:
                    sm = power_table[a] + power_table[b]
                else:
                    sm = power_table[a]

                if sm > mx_sm:
                    mx_sm = sm

        ans += mx_sm

        # 다음 좌표 산출 로직
        if idx < time:  # 맨 끝 좌표의 다음 좌표는 없음
            ad = lst_A[idx]
            bd = lst_B[idx]

            dar, dac = direction[ad]
            dbr, dbc = direction[bd]

            ar, ac = ar + dar, ac + dac
            br, bc = br + dbr, bc + dbc  # 사용자 좌표 밖으로 가는 경우 없으므로 그냥 바로 갱신


# 입력
T = int(input())
for tc in range(1, 1 + T):
    time, bc_num = map(int, input().split())
    lst_A = list(map(int, input().split()))  # 이동 방향 담겨 있음
    lst_B = list(map(int, input().split()))
    arr = [[[] for _ in range(10)] for _ in range(10)]  # 3중 리스트!

    power_table = []
    for idx in range(bc_num):
        c, r, s, p = map(int, input().split())
        draw_bc(r - 1, c - 1, s, idx)  # 1, 1부터 시작
        power_table.append(p)

    power_table.append(0)  # 충전소 없는거 처리

    ans = 0
    ab_move(0, 0, 9, 9)

    print(f"#{tc}", ans)

```
