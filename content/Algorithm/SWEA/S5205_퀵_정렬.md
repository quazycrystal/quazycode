### 코드
```python
# 요렇게 짜면 짧은 축
def quick(lst):
    if len(lst) < 2:
        return lst
    left = []
    right = []
    pivot = lst[0]
    for i in range(1, len(lst)): # lst[0] 제외
        # 피벗과 같을 수도 있다!
        if lst[i] > pivot:
            right.append(lst[i])
        else:
            left.append(lst[i])
    return quick(left) + [pivot] + quick(right)

T = int(input())
for tc in range(1, 1+T):
    N = int(input())
    origin = list(map(int, input().split()))
    ans_lst = quick(origin)
    print(f"#{tc}", ans_lst[N//2])


```
