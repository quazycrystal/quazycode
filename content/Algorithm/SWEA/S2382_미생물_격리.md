### 코드 개선 방법
- [이동] 끝 칸에 닿으면 방향 반대로, 군집 수 //2. 0이면 제거
- [합치기] 같은 칸 군집은 수를 더하고 가장 큰 군집 방향 유지 (같은 수는 없음)
- [복사] 이미 복사본이면 원본 수정해도 됨

### 코드
```python
direction = {1: (-1, 0), 2: (1, 0), 3: (0, -1), 4: (0, 1)}
# 함수 0: 경계 확인
def is_edge(r, c):
    return r == 0 or r == (size - 1) or c == 0 or c == (size - 1)

# 함수 1 이동 함수, 리스트 갱신
def move(lst, arr):
    new_where_micro = set()
    new_arr = [[[] for _ in range(size)] for _ in range(size)] # 100*100 생기는데...;
    for r, c in lst:
        micros = arr[r][c]
        for num, dirr in micros:
            dr, dc = direction[dirr]
            nr, nc = r + dr, c + dc
            if 0 <= nr < size and 0 <= nc < size:
                if is_edge(nr, nc): # 끝에 있으면 방향 전환, 군집 수 //2
                    if dirr == 2 or dirr == 4:
                        dirr = dirr - 1
                    elif dirr == 1 or dirr == 3:
                        dirr = dirr + 1

                    if (num // 2) > 0: # 군집 0이면 사라짐 처리
                        new_arr[nr][nc].append((num//2, dirr))
                        new_where_micro.add((nr, nc))
                else:
                    new_arr[nr][nc].append((num, dirr))
                    new_where_micro.add((nr, nc))

    return new_where_micro, new_arr

# 함수 2: 합치고 나누고
def summ(lst):
    for r, c in lst:
        micros = arr[r][c]
        sm = 0
        mx_num = 0
        new_dirr = 0

        for num, dirr in micros:
            sm += num
            if num > mx_num: # 같은 미생물 수 존재 X
                mx_num = num
                new_dirr = dirr

        arr[r][c] = [(sm, new_dirr)]


# 입력
T = int(input())
for tc in range(1, 1+T):
    size, time, chunk_num = map(int, input().split())

    arr = [[[] for _ in range(size)] for _ in range(size)]
    where_micro = set()

    for _ in range(chunk_num):
        r, c, micro_num, d = map(int, input().split())
        arr[r][c].append((micro_num, d))
        where_micro.add((r, c))

    for _ in range(time):
        n_where_micro, n_arr = move(where_micro, arr)
        where_micro, arr = n_where_micro, n_arr
        summ(where_micro)

    # === 계산 끗 ===
    ans = 0
    for row in range(size):
        for col in range(size):
            for num, dirr in arr[row][col]:
                ans += num

    print(f"#{tc}", ans)
```
