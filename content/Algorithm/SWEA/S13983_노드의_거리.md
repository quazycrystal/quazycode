```python
def bfs(s):
    global ans
    v[s] = 1
    q = []
    q.append(s)

    ans += 1
    while q:
        cur = q.pop(0)
        for nxt in adj[cur]:
            # 범위 내, 미방문, 조건
            if v[nxt] == 0:
                    v[nxt] = 1
                    q.append(nxt)
                if nxt == end:
                    ans = q[-1]

T = int(input())
node, edge = map(int(input().split()))
v = [0] * (node+1)
adj = [[] for _ in range(node+1)]
for _ in range(edge):
    t = list(map(int(input().split())))
    adj[t[0]].append[t[1]]
    adj[t[1]].append[t[0]] # 양방향

srt, end = map(int(input().split()))
bfs(srt)
ans = 0
```
