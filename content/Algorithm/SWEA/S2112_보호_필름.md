### 코드 개선 방법
- [재귀] 백트래킹 안에서 for를 돌리지 말고 현재 행 idx를 넘기기. d와 idx 모두 종료 조건
- [선택] 안 넣음 / A / B 중 하나만 고른다는 느낌
- [가지치기] 체크 함수에서 한 열이라도 실패하면 컷. K개만큼 넣으면 무조건 통과하니 상한

### 코드
```python
# S2112 보호필름

def check(arr):
    ok_col = 0
    # 안 된거 부족 갯수 알면?

    for c in range(col):
        cnt0 = 0
        cnt1 = 0
        for r in range(row):
            if arr[r][c] == 0:
                if cnt1 > 0:
                    cnt1 = 0
                cnt0 += 1
            elif arr[r][c] == 1:
                if cnt0 > 0:
                    cnt0 = 0
                cnt1 += 1

            if cnt0 >= thresh or cnt1 >= thresh:
                ok_col += 1
                break

        if cnt0 < thresh and cnt1 < thresh:
            break

    return ok_col

# 2. 백트

def btk(d, idx):
    global ans

    ok_col = check(arr)  # check도 한 번만!
    if ok_col == col:
        ans = d
        return

    if d == thresh or idx == row:
        return

    if ans <= d:
        return

    else:
        choice_r = arr[idx]
        btk(d, idx + 1)

        arr[idx] = [0] * col
        btk(d+1, idx + 1)

        arr[idx] = [1] * col
        btk(d+1, idx + 1)

        arr[idx] = choice_r

# 입력
T = int(input())
for tc in range(1, 1+T):
    row, col, thresh = map(int, input().split())
    arr = [list(map(int, input().split())) for _ in range(row)]

    # thresh만큼 물약 넣으면 무조건 조건 만족!
    ans = 10**18
    btk(0, 0)
    print(f"#{tc}", ans)
```
