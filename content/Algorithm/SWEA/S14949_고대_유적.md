### 코드
```python
T = int(input())
# 여러개의 테스트 케이스가 주어지므로, 각각을 처리합니다.
for test_case in range(1, T + 1):
    n, m = map(int, input().split())
    col_mx = 0
    row_mx = 0
    arr = [list(map(int, input().split())) for _ in range(n)]
    for i in range(n):
        for j in range(m):
            if arr[i][j] == 1:
                col = 0
                row = 0
                for k in range (i, n): # 세로 탐색
                    if arr[k][j] == 1:
                        col += 1
                    else:
                        break
                for l in range (j, m): # 가로 탐색
                    if arr[i][l] == 1:
                        row += 1
                    else:
                        break
                if col > col_mx:
                    col_mx = col
                if row > row_mx:
                    row_mx = row
    print (f'#{test_case} {max(col_mx, row_mx)}')

```
