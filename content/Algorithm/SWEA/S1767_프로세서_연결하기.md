- [전처리] 가장자리 코어는 이미 연결. 나머지만 리스트로

def btk(d, chip, length, arr):
    global mx_chip, mn_line
    if d == chip_num:
        if chip >= mx_chip:
            if chip > mx_chip:
                mx_chip = chip
                mn_line = length
            else:
                if mn_line > length:
                    mn_line = length
        return

    if chip + (chip_num - d) < mx_chip:

        arr_copy, connected, line_len = check(r, c, dr, dc, arr)

# 2. 판정
def check(sr, sc, dr, dc, arr):
    temp = []
    obstacle = 0
    line_len = 0
    arr_copy = [line[:] for line in arr]
    while True:
        nr, nc = sr + dr, sc + dc
        if 0 <= nr < size and 0 <= nc < size:
            if arr_copy[nr][nc] == 0:
                temp.append((nr, nc))
                arr_copy[nr][nc] = 2
                line_len += 1
                sr, sc = nr, nc
            else:
                obstacle = 1
                break
        else:
            break

    if obstacle == 0: # 중단 안 되고 왔다 = 연결
        return arr_copy, 1, line_len
    else:
        return arr, 0, 0

# 입력
T = int(input())
for tc in range(1, 1+T):
    size = int(input())
    arr = [list(map(int, input().split())) for _ in range(size)]

    # 연결해야 하는 칩 좌표
    to_connect = []
    for i in range(size):
        for j in range(size):
            if arr[i][j] == 1:
                if 0 < i < size-1 and 0 < j < size-1:
                    to_connect.append((i, j))

    chip_num = len(to_connect)

    mx_chip = 0
    mn_line = 10000 # 어차피 12*12

    btk(0, 0, 0, arr)
    print(f"#{tc}", mn_line)
