### 코드
```python
l = int(input())
n = int(input())
cake = [0 for _ in range(l)]
visit = [list(map(int, input().split())) for _ in range(n)]
exp_mx = 0
exp = 0
get = [0 for _ in range (n+1)]

for i in range(n): # 방문자만큼
    srt = visit[i][0]
    end = visit[i][1]+1 # 끝 포함
    for j in range (srt, end):
        if cake[j] == 0:
            cake[j] = i+1
        else:
            pass # 안에 들어있으면 바꾸지 않음
    if end-srt > exp_mx:
        exp_mx = end-srt
        exp = i+1

for k in cake:
    get[k] += 1
    get[0] = 0
    real = get.index(max(get))

print(exp)
print(real)
```
