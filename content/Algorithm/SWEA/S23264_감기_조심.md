### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [인덱스] 첫 자리 1행 1열, 1씩 빼기
- [거리] 걸린 날이 1일

### 코드 개선 방법
- [방문 처리] arr를 같이 써도 되지만 실수 줄이려고 v 따로

### 코드
```python
def bfs(sr, sc):
    v[sr][sc] = 1
    q = []
    q.append((sr, sc))
    arr[sr][sc] = 0  # 감기 걸린 교육생 제거
    while q:
        cur = q.pop(0)
        r, c = cur[0], cur[1]
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr = r + dr
            nc = c + dc
            # 범위 체크
            if 0 <= nr < row and 0 <= nc < col:
                if v[nr][nc] == 0: # 방문 안 했고
                    if arr[nr][nc] == 1: # 교육생이 있다면
                        v[nr][nc] = v[r][c] + 1
                        q.append((nr, nc))
                        arr[nr][nc] = 0 # 감기 걸린 교육생 제거

T = int(input())
for tc in range(1, 1+T):
    row, col = map(int, input().split())
    arr = [list(map(int, input().split())) for _ in range(row)]
    v = [[0] * col for _ in range(row)]
    s_r, s_c = map(int, input().split())

    # 첫 자리 1행 1열이므로 1씩 빼줘야
    s_r = s_r-1
    s_c = s_c-1

    # BFS 실행
    bfs(s_r, s_c)

    # 모두 감기 걸릴 때 까지 걸린 시간
    time = max(map(max, v))
    # 살아남은 교육생
    safe = 0
    for rr in arr:
        safe += rr.count(1)

    print(f"#{tc}", time, safe)
```
