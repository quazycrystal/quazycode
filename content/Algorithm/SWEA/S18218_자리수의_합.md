def sm(n): # 5, 4, 3..
    if n == 0:
        return 0
    else:
        return sm(n-1) + int(s[n-1]) # 인덱스니까


T = int(input())
for tc in range(1, T + 1):
    s = input()
    N = len(str(s))
    print(f"#{tc}", sm(N))
