### 코드
```python
for tc in range(1, 11):
    n, s = input().split() # n은 문자열 길이, s는 문자열
    stk = []
    ans = ''
    for i in s:
        if not stk:
            stk.append(i)

        else:
            if i == stk[-1]:
                stk.pop()
            else:
                stk.append(i)
    for j in stk:
        ans += j
    print(f'#{tc} {int(ans)}')
```
