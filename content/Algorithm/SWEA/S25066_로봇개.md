### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [문제 읽기] 대각선 이동 중 모서리가 아니면 마주보는 방향으로 반사
- [조건] 사람은 1 통과 불가, 개는 가능

### 코드 개선 방법
- [BFS] 한 레벨(사이클)씩 묶어서 처리

### 코드
```python
from collections import deque

# 1. 개 이동 함수 - 초기 방향, 한 번 튕긴 방향
d = {6: [(-1, -1), (1, 1)],
     7: [(-1, 0), (1, 0)],
     8: [(-1, 1), (1, -1)],
     9: [(0, 1), (0, -1)]}

def dog(six, seven, eight, nine): # 각 개의 좌표값 받고, 이동한 좌표값 줌
     temp = [six, seven, eight, nine]
     idx_temp = [6, 7, 8, 9]
     re_temp = []
     for i in range(4):
          r, c = temp[i]
          dr, dc = d[idx_temp[i]][0]
          nr, nc = r + dr, c + dc
          if 0 <= nr < row and 0 <= nc < col:
               re_temp.append((nr, nc))
               # 원 arr 위치 이동
               arr[r][c] = 0
               if arr[nr][nc] != 1: # 어차피 1일 때 사람 못 지나가니까
                    arr[nr][nc] = idx_temp[i]
          else:
               d[idx_temp[i]].reverse() # 배열 뒤집기
               dr, dc = d[idx_temp[i]][0]
               nr, nc = r + dr, c + dc
               re_temp.append((nr, nc))
               # 원 arr 위치 이동
               arr[r][c] = 0
               if arr[nr][nc] != 1:
                    arr[nr][nc] = idx_temp[i]

     return re_temp[0], re_temp[1], re_temp[2], re_temp[3]

def human(s):
     global s6, s7, s8, s9
     global flag
     sr, sc = s
     arr[sr][sc] = 0
     q = deque()
     q.append(s)
     while q:
          r, c = q.popleft()
          # 정답
          if (r, c) == (g_r, g_c):
               flag = 1
               return

          s6, s7, s8, s9 = dog(s6, s7, s8, s9)

          for _ in range(len(q)+1):
               # 한 레벨(사이클) 별로 묶기
               for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < row and 0 <= nc < col:
                        if arr[nr][nc] == 0:
                             arr[nr][nc] = arr[r][c] - 1
                             q.append((nr, nc))


# 입력
T = int(input())
row, col = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)]

s_r, s_c = 0, 0
g_r, g_c = 0, 0
s6, s7, s8, s9 = (0, 0), (0, 0), (0, 0), (0, 0)
flag = 0

for j in range(row):
     for k in range(col):
          if arr[j][k] == 2:
               s_r, s_c = j, k
          elif arr[j][k] == 3:
               g_r, g_c = j, k
          elif arr[j][k] == 6:
               s6 = (j, k)
          elif arr[j][k] == 7:
               s7 = (j, k)
          elif arr[j][k] == 8:
               s8 = (j, k)
          elif arr[j][k] == 9:
               s9 = (j, k)

human((s_r, s_c))
print(arr)
```
