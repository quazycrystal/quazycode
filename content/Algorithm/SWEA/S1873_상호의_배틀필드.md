### 실패 / 성공 이유
- [분기] 분기를 자꾸 만듦
- [엣지 케이스] 맵 밖으로 못 나가도 방향은 바뀌어야

### 코드 개선 방법
- [자료구조] 명령별 이동과 방향 기호를 dict로
- [구현] 문제를 성실히 구현만 하면 됨. 이동한 좌표를 받아오는 것 잊지 않기

### 코드
```python
dr = [-1, 1, 0, 0]
dc = [0, 0, -1, 1] # 상하좌우
def move(s, sr, sc): # 명령, 움직일 좌표
    # 이동
    if s == "U":
        nr, nc = sr + dr[0], sc + dc[0]
        if 0 <= nr < row and 0 <= nc < col:
            if arr[nr][nc] == ".":
                arr[sr][sc] = "."
                arr[nr][nc] = "^"
                return nr, nc
            else:
                arr[sr][sc] = "^"
        else:
            arr[sr][sc] = "^"

    if s == "D":
        nr, nc = sr + dr[1], sc + dc[1]
        if 0 <= nr < row and 0 <= nc < col: # 분기 자꾸 만들지 말고ㅠㅠ
            if arr[nr][nc] == ".":
                arr[sr][sc] = "."
                arr[nr][nc] = "v"
                return nr, nc
            else:
                arr[sr][sc] = "v"
        else:
            arr[sr][sc] = "v"

    if s == "L":
        nr, nc = sr + dr[2], sc + dc[2]
        if 0 <= nr < row and 0 <= nc < col:
            if arr[nr][nc] == ".":
                arr[sr][sc] = "."
                arr[nr][nc] = "<"
                return nr, nc
            else:
                arr[sr][sc] = "<"
        else:
            arr[sr][sc] = "<"

    if s == "R":
        nr, nc = sr + dr[3], sc + dc[3]
        if 0 <= nr < row and 0 <= nc < col:
            if arr[nr][nc] == ".":
                arr[sr][sc] = "."
                arr[nr][nc] = ">"
                return nr, nc
            else:
                arr[sr][sc] = ">"
        else:
            arr[sr][sc] = ">" # 인덱스 나가도 방향은 바뀌어야 함...

    # 포탄 발사
    if s == "S":
        if arr[sr][sc] == "^":
            for a in range(1, row):
                nr, nc = sr + (dr[0] * a), sc + (dc[0] * a)
                if 0 <= nr < row and 0 <= nc < col:
                    if arr[nr][nc] == "*":
                        arr[nr][nc] = "."
                        break
                    if arr[nr][nc] == "#":
                        break

        if arr[sr][sc] == "v":
            for b in range(1, row):
                nr, nc = sr + (dr[1] * b), sc + (dc[1] * b)
                if 0 <= nr < row and 0 <= nc < col:
                    if arr[nr][nc] == "*":
                        arr[nr][nc] = "."
                        break
                    if arr[nr][nc] == "#":
                        break

        if arr[sr][sc] == "<":
            for c in range(1, col):
                nr, nc = sr + (dr[2] * c), sc + (dc[2] * c)
                if 0 <= nr < row and 0 <= nc < col:
                    if arr[nr][nc] == "*":
                        arr[nr][nc] = "."
                        break
                    if arr[nr][nc] == "#":
                        break

        if arr[sr][sc] == ">":
            for d in range(1, col):
                nr, nc = sr + (dr[3] * d), sc + (dc[3] * d)
                if 0 <= nr < row and 0 <= nc < col:
                    if arr[nr][nc] == "*":
                        arr[nr][nc] = "."
                        break
                    if arr[nr][nc] == "#":
                        break

    return sr, sc

for tc in range(1, 1+T):

T = int(input())

dr = [-1, 1, 0, 0]  # 위, 아래, 왼쪽, 오른쪽
dc = [0, 0, -1, 1]
tank_char = ['^', 'v', '<', '>']
cmd_to_dir = {'U': 0, 'D': 1, 'L': 2, 'R': 3} # 다 짝 지으셨군
# 이러면

for testcase in range(1, T + 1):
    H, W = map(int, input().split())
    game_map = [list(input()) for _ in range(H)]

    # 초기 위치 및 방향
    for r in range(H):
        for c in range(W):
            if game_map[r][c] in tank_char:
                tr, tc = r, c
                td = tank_char.index(game_map[r][c])
                break

    N = int(input())
    commands = input()

    for cmd in commands:
        # 이동
        if cmd in cmd_to_dir:
            td = cmd_to_dir[cmd]  # 방향 변경
            nr = tr + dr[td]
            nc = tc + dc[td]

            if 0 <= nr < H and 0 <= nc < W and game_map[nr][nc] == '.':
                game_map[tr][tc] = '.'
                tr, tc = nr, nc

            game_map[tr][tc] = tank_char[td]

        elif cmd == 'S':
            nr = tr + dr[td]
            nc = tc + dc[td]

            while 0 <= nr < H and 0 <= nc < W:
                if game_map[nr][nc] == '*':  # 벽돌 벽
                    game_map[nr][nc] = '.'
                    break
                elif game_map[nr][nc] == '#':  # 강철 벽
                    break
                nr += dr[td]
                nc += dc[td]

    print(f'#{testcase}', end=' ') # 이렇게 해도 이어지나본데?
    for row in game_map:
        print(''.join(row))
```
