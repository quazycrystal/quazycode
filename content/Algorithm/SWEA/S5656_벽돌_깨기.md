technique: [중력]

### 걸린 시간:  (실패 횟수: -)

### 실패 / 성공 이유
- [재귀] 재귀가 12 + 12^2 + 12^3 + 12^4번 불림. 매번 배열 복제하면 느림

### 코드 개선 방법
- [복사] for 안에서 매번 새로 복제하지 말고 위에 하나 만들어두고 갈아치우기
- [BFS] 지나간 칸은 방문 표시만, 나중에 한꺼번에 없앰. 1짜리는 퍼뜨리지 않음
- [가지치기] 벽돌이 다 깨지면 바로 종료

### 코드
```python
# 약품 투여 k개

from collections import deque


# 함수
def is_range(r, c):
    return 0 <= r < row and 0 <= c < col


def pop_bfs(sr, sc, arr):
    v = [[0] * col for _ in range(row)]
    v[sr][sc] = 1
    q = deque()
    q.append((sr, sc))
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            # 상하좌우
            if arr[r][c] == 1:
                if v[r][c] == 0:
                    v[r][c] = 1  # 1개짜리는 순회 안 함

            elif arr[r][c] > 1:
                for mul in range(1, arr[r][c]):
                    nr, nc = r + (dr * mul), c + (dc * mul)
                    if is_range(nr, nc) and v[nr][nc] == 0:
                        v[nr][nc] = 1
                        if arr[nr][nc] > 1:
                            q.append((nr, nc))

    return v


# 2. 중력
def gravity(arr):
    for c in range(col):
        temp = []
        for r in range(row - 1, -1, -1):
            if arr[r][c] > 0:
                temp.append(arr[r][c])
                arr[r][c] = 0

        reversed_r = (row-1)
        temp_idx = 0
        while temp:
            arr[reversed_r][c] = temp.pop(0)
            reversed_r -= 1
            temp_idx += 1


# 3. 백트래킹
def btk(d, arr_ori):
    global ans
    
    if d == marble_num:
        temp_ans = row*col
        num0 = 0
        for rr in arr_ori:
            num0 += rr.count(0)

        temp_ans -= num0

        ans = min(ans, temp_ans)
        return


    arr = [[0] * col for _ in range(row)]
    for sel_c in range(col):
        for i in range(row):
            for j in range(col):
                arr[i][j] = arr_ori[i][j] # 새로 안 만들고 갱신만

        sel_r = 0
        for r in range(row):
            if arr[r][sel_c] > 0:
                sel_r = r
                break

        # 방문 배열 받아와서
        visited = pop_bfs(sel_r, sel_c, arr)

        # 터뜨리기
        for i in range(row):
            for j in range(col):
                if arr[i][j] > 0 and visited[i][j] == 1:
                    arr[i][j] = 0

        # 중력
        gravity(arr)
        btk(d+1, arr)


# 입력
T = int(input())
for tc in range(1, 1 + T):
    marble_num, col, row = map(int, input().split())
    arr_ori = [list(map(int, input().split())) for _ in range(row)]

    ans = (row*col) + 1
    btk(0, arr_ori)
    print(f"#{tc}", ans)

```
