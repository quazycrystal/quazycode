### 걸린 시간: 1시간 6분 (구상 28분, 구현 37분) (실패 횟수: -)

### 실패 / 성공 이유
- [구상] 벽으로 막히지 않는 무한 격자라 좌표로 가야 함. 주기(0일 때 빼는지, -1 되는 시점에 빼는지)가 헷갈림

### 코드 개선 방법
- [자료구조] 격자 대신 좌표 dict. 범위 체크 불필요
- [동시 갱신] 새로 태어난 세포는 이번 턴 방문 체크용 집합에 바로 넣지 않기
- [정렬] 생명력 큰 세포부터 번식시켜 같은 칸 경쟁 해결
- [복사] for 안에서 dict pop 대신 새로 만들어 덮어씌우기

### 코드
```python
# S5653 줄기 세포 배양

def wake():
    now_wake = []
    still_sleep = {}
    for tpl in sleep:
        r, c = tpl
        every.add((r, c))
        if sleep[tpl][1] == 0:
            now_wake.append((sleep[tpl][0], r, c))
        else:
            still_sleep[tpl] = sleep[tpl]
            still_sleep[tpl][1] -= 1  # 지속 시간 1씩 까주기

    # 갈아주기
    return still_sleep, now_wake

# 2. 사방으로 새로운 생명 탄생
def birth(lst):
    lst.sort(key = lambda x: -x[0]) # 생명력 적은 애 부터 bfs
    while lst:
        t, r, c = lst.pop()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if (nr, nc) not in every:
                sleep[(nr, nc)] = [t, t]

        awake[(r, c)] = [t, t-1]

# 3. 죽음 처리
def die():
    still_awake = {}
    for tpl in awake:
        if awake[tpl][1] == 0:
            dead.add(tpl)
        else:
            still_awake[tpl] = awake[tpl]
            still_awake[tpl][1] -= 1

    return still_awake


# 입력
T = int(input())
for tc in range(1, 1+T):
    row, col, time = map(int, input().split())
    arr = [list(map(int, input().split())) for _ in range(row)]

    # 정보 추출, 좌표화
    sleep = {}
    awake = {}
    dead = set()
    every = set()

    for r in range(row):
        for c in range(col):
            if arr[r][c] > 0:
                every.add((r, c))
                sleep[(r, c)] = [arr[r][c], arr[r][c]]

    # 풀이
    for now_time in range(time):

        still_sleep, now_wake = wake()
        sleep = still_sleep
        if now_wake:
            birth(now_wake)

        still_awake = die()
        awake = still_awake

    ans = len(sleep) + len(awake)
    print(f"#{tc}", ans)
```
