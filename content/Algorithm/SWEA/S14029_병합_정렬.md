def merge(lst):
    # 왼, 오 정의
    global ans
    if len(lst) < 2:
        return lst

    mid = len(lst) // 2
    left = merge(lst[:mid])
    right = merge(lst[mid:]) # 바로 부른다
    merged = []
    i = 0
    j = 0
    while 0 <= i < len(left) and 0 <= j < len(right):
        if left[i] < right[j]:
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1

    if left[-1] > right[-1]:
        ans += 1
    return merged + left[i:] + right[j:]

T = int(input())
for tc in range(1, 1+T):
    N = int(input())
    origin = list(map(int, input().split()))

    ans = 0
    ans_lst = merge(origin)

    print(f"#{tc}", ans_lst[N//2], ans)
