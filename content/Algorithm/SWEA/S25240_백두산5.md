### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [문제 읽기] '최대' 2개면 0, 1, 2개 모두. 분화구가 2개 미만일 수도
- [문제 읽기] 분화구보다 낮은 곳이 아니라 현재 칸보다 낮은 곳으로 흐름
- [조건] 가장자리도 분화구가 될 수 있고, 0은 흐르지 않음

### 코드 개선 방법
- [조합] 분화구를 미리 찾고 조합으로 고른 뒤 v를 공유한 bfs

### 코드
```python
from collections import deque
from itertools import *


# bfs
def bfs(sr, sc):  # 이미
    v[sr][sc] = 1
    q = deque()
    q.append((sr, sc))
    # mx = arr[sr][sc]  # 분화구 높이
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < row and 0 <= nc < col:
                if v[nr][nc] == 0 and 0 < arr[nr][nc] < arr[r][c]:
                    v[nr][nc] = 1
                    q.append((nr, nc))


# 입력
T = int(input())
for tc in range(1, 1 + T):
    row, col = map(int, input().split())
    arr = [list(map(int, input().split())) for _ in range(row)]

    pos_boom = []  # 폭발 가능한 분화구 미리 찾기
    pos_ans = []  # 나오는 정답들 모아

    # 전처리 - 화산 분화구 찾기
    for i in range(row):
        for j in range(col):
            flag = 0  # 주변 지역보다 내가 가장 높은 곳인지?
            for di, dj in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ni, nj = i + di, j + dj
                if 0 <= ni < row and 0 <= nj < col:
                    if arr[ni][nj] >= arr[i][j]:
                        flag = 1

            if flag == 0:  # 내가 제일 높아!
                pos_boom.append((i, j))  # 화산 분화구로 등록

    final_ans = 0
    if len(pos_boom) == 0:
        pass

    elif len(pos_boom) == 1:

        v = [[0] * col for _ in range(row)]
        ans = 0
        # v 공유 하는게 중요!
        bfs(pos_boom[0][0], pos_boom[0][1])

        for rr in v:
            ans += rr.count(1)

        final_ans = ans

    else:
        for combi in combinations(pos_boom, 2):  # 조합으로 분화구 리스트에서 2개 골라온다
            v = [[0] * col for _ in range(row)]
            ans = 0
            # v 공유 하는게 중요!
            bfs(combi[0][0], combi[0][1])
            bfs(combi[1][0], combi[1][1])

            for rr in v:
                ans += rr.count(1)

            pos_ans.append(ans)
        final_ans = max(pos_ans)

    print(f"#{tc}", final_ans)
```
