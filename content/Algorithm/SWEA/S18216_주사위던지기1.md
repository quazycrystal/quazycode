def btk(n, lst):
    if n == N: # N보다 큰지? 작은지? 같은지 주의
        print(*lst)
        return
    else:
        for j in dice:
            btk(n+1, lst+[j])

T = int(input())
for tc in range(1, T+1):
    N = int(input())
    lst = []
    # v = [0] * 6
    dice = [1, 2, 3, 4, 5, 6]
    cnt = 0
    print(f'#{tc}')
    btk(0, lst)  # 초기 상태는 주사위 안 돌린 때
