### 걸린 시간: 1시간 39분 (저녁 26분, 아침 15분, 점심 이후 58분) (실패 횟수: -)

### 실패 / 성공 이유
- [상태 확장] 빨강·파랑을 따로 움직이는 객체로 보고 큐를 2개 만듦. 백트래킹식 사고

### 코드 개선 방법
- [상태 확장] (빨강 좌표, 파랑 좌표)를 하나의 상태로 묶어 bfs
- [이동] 이동 함수는 보정 없이 끝까지. 겹치면 이동 거리가 긴 쪽을 한 칸 뒤로
- [종료] 정답일 때만 break. 파랑이 빠지면 그 경우만 버림

### 코드
```python
# C 2개의 사탕 (구슬 탈출 2차)

from collections import deque

# 함수
def is_range(r, c):
    return 0 <= r < row and 0 <= c < col

# 1. bfs 안에서 돌릴, 구슬 이동하는 함수 - 보정 없음!
def move(start, delta):
    sr, sc = start
    dr, dc = delta
    r, c = sr, sc
    while True:
        nr, nc = r + dr, c + dc
        if not is_range(nr, nc):
            break

        if arr[nr][nc] == "#":
            break

        if arr[nr][nc] == "O":
            r, c = nr, nc
            break

        r, c = nr, nc

    return (r, c), abs(sr - r) + abs(sc - c)  # 이동 거리!


def bfs(red, blue):
    v = {(red, blue)}
    q = deque()
    q.append((0, red, blue))
    now_red = ()
    # now_blue = ()
    now_can = 0
    while q:

        now_can, now_red, now_blue = q.popleft()

        # 정답 탈출
        if now_can > 10:
            break

        if now_red == goal and now_blue != goal: # 정답일 때만 break!!!
            break

        # 다음 위치 물색
        for d in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nxt_red, rdist = move(now_red, d)
            nxt_blue, bdist = move(now_blue, d)

            # 거리 보정
            if nxt_red == nxt_blue and nxt_red != goal:
                if rdist > bdist:
                    nrr, nrc = nxt_red
                    nxt_red = (nrr - d[0], nrc - d[1])
                else:
                    nbr, nbc = nxt_blue
                    nxt_blue = (nbr - d[0], nbc - d[1])

            if (nxt_red, nxt_blue) not in v:
                if nxt_blue != goal:
                    v.add((nxt_red, nxt_blue))
                    q.append((now_can + 1, nxt_red, nxt_blue))

    if now_red == goal and now_can < 11:
        return now_can
    else:
        return -1

# 입력
row, col = map(int, input().split())
arr = [list(input()) for _ in range(row)]

# 공, 도착지점 위치 찾기
red = ()
blue = ()
goal = ()
for r in range(row):
    for c in range(col):
        if arr[r][c] == "R":
            red = (r, c)
        elif arr[r][c] == "B":
            blue = (r, c)
        elif arr[r][c] == "O":
            goal = (r, c)

# 풀이
dist = bfs(red, blue)
print(dist)
```
