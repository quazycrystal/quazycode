### 코드
```python
ok = [(-2, 1), (-1, 2), (1, 2), (2, 1), (-2, -1), (-1, -2), (1, -2), (2, -1)]
v = [[0] * 6 for _ in range(6)]
def name(s):
    col = s[0]
    row = int(s[1])
    col = ord(col) - 65
    row = 6 - row
    return (row, col) # 원래 익숙한 행 / 렬 기반

def go(pre, now):
    row = now[0] - pre[0]
    col = now[1] - pre[1]

    if (row, col) in ok and v[pre[0]][pre[1]] == 0: # 중복 체크!!!!!!!!!!!!!!!!
        v[pre[0]][pre[1]] = 1
        return True
    else:
        return False

calls = [input() for _ in range(36)]
calls = calls + [calls[0]]

ans = "Invalid"

for i in range(36):
    pre_name = name(calls[i])
    now_name = name(calls[i+1])
    check = go(pre_name, now_name)
    if not check:
        break

if check:
    for rr in v:
        cnt = rr.count(0)
        if cnt == 0:
            ans = "Valid"

print(ans)

```
