### 걸린 시간: 1트 구상 18분, 구현 1시간 24분 이후 디버깅 / 2트 2시간 정도 (실패 횟수: -)

### 실패 / 성공 이유
- [복사] 원본 → 물고기 이동 → 상어 이동 3단계인데 물고기 이동 후 arr를 복사 안 함
- [스왑] 앞이 상어(0)면 스왑하지 말라는 조건이 빠져 상어와 물고기가 자리를 바꿈
- [자료구조] 위치 리스트, 2중 arr, 상어 좌표를 다 갱신하다 머릿속에서 섞임
- [갱신] 물고기가 방향을 바꾸면 그 방향을 저장해야 함

### 코드 개선 방법
- [복사] 백트래킹에서 물고기 이동은 모든 경우에 공통. 그 결과를 복사한 뒤 상어 이동만 분기
- [자료구조] N = 4라 관리 어려운 위치 리스트 대신 arr 완탐
- [디버깅] arr 사이에 구분 글자를 넣어 출력

### 코드
```python
# 디버깅: 1시간 25분 (총 2시간 45분)

# 0. 범위 체크
def is_range(r, c):
    return 0 <= r < size and 0 <= c < size

# 1. 방향 정하기
def choose_nxt(r, c, d, arr):
    for nxt_d in range(8):
        now_d = (nxt_d + d) % 8 # 방향 들어온 곳부터 시작
        dr, dc = direction[now_d]
        nr, nc = r + dr, c + dc
        if is_range(nr, nc):
            if arr[nr][nc][0] != -1:
                idx, dirr = arr[r][c]
                arr[r][c] = idx, now_d # 물고기 방향 갱신
                arr[r][c], arr[nr][nc] = arr[nr][nc], arr[r][c]
                # 물고기거나 빈 칸이면 스왑
                return

# 2. 좌표 찾기
def find_fish(target, arr):
    for r in range(size):
        for c in range(size):
            if arr[r][c][0] == target:
                return r, c
    return -1

# 3. 물고기 움직
def fish_move(arr):
    # 물고기 작은 순으로, arr 바로 바꾼다
    where_shark = find_fish(-1, arr)
    for target in range(1, 17):
        jwa = find_fish(target, arr)
        if jwa != -1:
            r, c = jwa
            d = arr[r][c][1]
            choose_nxt(r, c, d, arr)

    return where_shark

# 4. 상어 움직 - 백트래킹, arr 넘긴다
def shark_move(d, ori_arr, sm):
    global ans
    ans = max(sm, ans)

    if d == 15:
        return

    arr = [line[:] for line in ori_arr]
    shark = fish_move(arr)

    r, c = shark
    dirr = (arr[r][c][1] % 8)
    dr, dc = direction[dirr]

    for s in range(1, 4):
        arr_copy = [line[:] for line in arr]
        nr, nc = r + (dr * s), c + (dc * s)

        if is_range(nr, nc):
            # 먹힐 물고기 정보
            if arr_copy[nr][nc] != (0, 0):
                # 비었으면 먹을 수 없음
                eaten_idx, eaten_dirr = arr_copy[nr][nc]
                arr_copy[r][c] = (0, 0)
                arr_copy[nr][nc] = (-1, eaten_dirr)

                shark_move(d+1, arr_copy, sm + eaten_idx)

# 입력
size = 4
ori_arr = [[0] * size for _ in range(size)]

# 방향
direction = {1: (-1, 0),
             2: (-1, -1),
             3: (0, -1),
             4: (1, -1),
             5: (1, 0),
             6: (1, 1),
             7: (0, 1),
             0: (-1, 1)} # 나눗셈 용이하게 하기 위해

for idx in range(4):
    i1, d1, i2, d2, i3, d3, i4, d4 = list(map(int, input().split()))
    ori_arr[idx][0] = (i1, d1)
    ori_arr[idx][1] = (i2, d2)
    ori_arr[idx][2] = (i3, d3)
    ori_arr[idx][3] = (i4, d4)

ans = 0
ans += ori_arr[0][0][0]
first_dirr = ori_arr[0][0][1] # 방향
ori_arr[0][0] = (-1, first_dirr) # -1은 상어!

# 풀이
shark_move(0, ori_arr, ans) # 첫 물고기도 합계로,,,
print(ans)
```
