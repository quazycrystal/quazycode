### 코드
```python
# B1051 숫자 정사각형
row, col = map(int, input().split())
arr = [list(map(int, input())) for _ in range(row)]
size = min(row, col)
ans = 0

for i in range(size, 0, -1): # 0인건 없음
    for r in range(row-i+1):
        for c in range(col-i+1):
            if arr[r][c] == arr[r+i-1][c]:
                if arr[r+i-1][c] == arr[r+i-1][c+i-1]:
                    if arr[r+i-1][c+i-1] == arr[r][c+i-1]:
                        ans = i**2
                        break
        if ans != 0:
            break
    if ans != 0:
        break
print(ans)

```
