### 걸린 시간: 1차: 2시간 20분 실패 (구상 17분, 구현 1시간 4분, 디버깅 59분) / 3트 리팩토링 약 45분 / 1트(재풀이) 38분 (실패 횟수: 1회 (2시간 20분 실패))

### 실패 / 성공 이유
- [거리] '지나야 하는 칸 수의 최솟값'을 택시거리로 착각. 실제로 갈 수 있는지 봐야 함 → 최소 거리면 BFS
- [정렬] (크기, 행, 열)로 정렬해 크기가 우선 기준이 됨. 뒤의 행·열 순서는 소용없음
- [속도] 물고기마다 bfs를 돌리고 sort를 반복해 1052ms

### 코드 개선 방법
- [BFS] 단계마다 bfs 한 번만. 돌리면서 먹을 수 있는 물고기를 모으고 (거리, 행, 열)로 정렬 (180ms)
- [정렬] key가 여러 개면 튜플로
- [이동 조건] 지나가기는 같거나 작을 때, 먹기는 작을 때만
- [습관] 막히는 작은 케이스를 직접 넣어보고 제출

from collections import deque

# 입력
N = int(input())
arr = [list(map(int, input().split())) for _ in range(N)]

# 변수
now_size = 2
fish_cnt = 0

# 물고기 찾아요
r, c = 0, 0
fish = []
for i in range(N):
    for j in range(N):
        if arr[i][j] == 9:
            r, c = i, j
            arr[i][j] = 0 # 원본에서 치운다

        elif 0 < arr[i][j] < 7:
            f_size = arr[i][j]
            fish.append((f_size, i, j))

fish.sort()

# 함수 설계 - 이동(거리 찾기), 물고기 찾기
def is_range(sr, sc):
    return 0 <= sr < N and 0 <= sc < N

def move(sr, sc, gr, gc):
    v = [[-1] * N for _ in range(N)]
    v[sr][sc] = 0
    q = deque()
    q.append((sr, sc))
    while q:
        r, c = q.popleft()
        if (r, c) == (gr, gc):
            break
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if is_range(nr, nc) and arr[nr][nc] <= now_size:
                if v[nr][nc] == -1:
                    v[nr][nc] = v[r][c] + 1
                    q.append((nr, nc))

    if v[gr][gc] == -1:
        return 0, sr, sc
    else:
        return v[gr][gc], gr, gc

def search(sr, sc):
    global fish_cnt, now_size
    if fish:
        temp = []
        i = 0
        while i < len(fish):
            if fish[i][0] < now_size: # 현 사이즈보다 작아야만 먹을 수 있음
                temp.append(fish[i])
                i += 1
            else:
                break

        mn_dist = 500
        picked = ()

        temp.sort(key=lambda x: (x[1], x[2])) # temp 행 렬 기준 재 소팅
        for s, fr, fc in temp:
            dist, _, _ = move(sr, sc, fr, fc)

            if dist != 0: # 이동 했으면
                if dist <= mn_dist:
                    if dist < mn_dist: # 만약 dist 같으면 제일 먼저 들어온 것이 들어감
                        mn_dist = dist
                        picked = (s, fr, fc) # 맨 앞 dist 아님에 주의, 사이즈가 와야

        if picked != ():
            fish.remove(picked)
            arr[picked[1]][picked[2]] = 0
            fish_cnt += 1

            if fish_cnt == now_size:
                now_size += 1
                fish_cnt = 0

            # 뽑혔다면
            return mn_dist, picked[1], picked[2]
        else:
            return 0, 0, 0

    else:
        return 0, 0, 0

time = 0
if fish:
    while True:
