### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [거리] 시작을 1로 방문 표시하니 정답에서 1 빼기
- [좌표] 8방향이라 좌표 실수 조심

### 코드 개선 방법
- [템플릿] 방문 표시와 큐 저장이 한 세트. 위치 갱신은 큐가 해 줌
- [변수] 함수 인자는 sx, sy, 입력 받을 땐 s_x, s_y로 통일

### 코드
```python
def bfs(sx, sy, ex, ey,c_size):
    global ans
    v[sx][sy] = 1
    q = []
    q.append((sx, sy))
    dy = [2, 1, -1, -2, -2, -1, 1, 2] # 헷갈리니 튜플 말고
    dx = [1, 2, 2, 1, -1, -2, -2, -1] # 시계방향
    while q:
        cur = q.pop(0) # 빼먹지 않게 주의!!!!!!!!!!!!!!!!!
        x, y = cur[0], cur[1] # 현재 위치 받기

        # ans는 항상 여기에!!!
        if (x, y) == (ex, ey):  # 정답 위치에 도착
            ans = v[x][y] - 1  # 초기에 1에서 시작했으므로 1 빼자

        for i in range(8):
            nx = x+dx[i]
            ny = y+dy[i]

            # 방문 처리, 범위, 조건
            if 0 <= nx < c_size and 0 <= ny < c_size: # size * size
                if v[nx][ny] == 0: # 갈 수 있는 다음 위치에 방문 안 한 곳 있으면
                    v[nx][ny] += v[x][y] + 1  # 누적해서 거리 -> nx, ny 다르니까 숫자 안 겹침
                    q.append((nx, ny)) # 큐에 저장


T = int(input())
for tc in range(1, T+1):
    size = int(input())

    # 시작과 끝 좌표
    s_x, s_y = map(int, input().split())
    e_x, e_y = map(int, input().split())

    # 필요한 것 만들기 : 배열(또는 인접 리스트), 방문록
    v = [[0]*size for _ in range(size)]
    ans = 0

    bfs(s_x, s_y, e_x, e_y, size)
    print(ans)
```
