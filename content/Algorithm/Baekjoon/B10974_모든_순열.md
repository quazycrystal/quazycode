def btk(d):
    if d == N:
        print(*lst)
        return
    for i in range(1, 1+N):
        if v[i] == 0:
            v[i] = 1
            lst[d] = i
            btk(d+1)
            v[i] = 0

N = int(input())
v = [0] * (N+1) # 0 안 씀
lst = [0] * N
