---
algorithm: [BFS, 시뮬레이션]
data_structure: [3차원 배열, 2차원 배열, set, 덱]
technique: [룩업 테이블]
---

### 걸린 시간: 1차: 구상 40분, 구현 2시간 20분 / 2차: 약 5시간 (19시 - 24시) / 재풀이: 구상 10분, 구현 58분, 디버깅 4시간 (실패 횟수: 1차 실패, 재풀이 디버깅 스스로 못 찾음)

### 실패 / 성공 이유
- [제어 흐름] 직진 칸에 벽이 있으면 continue로 양옆(대각) 확인까지 건너뜀. continue는 아래 코드를 전부 무시
- [테케] 커스텀 테케가 마침 1자로 갈 수 있어서 엣지가 안 잡힘. 직진은 막히고 옆으로만 갈 수 있는 테케가 필요했음
- [좌표] 바람은 E 모양으로 퍼지는데 ㅓ 모양으로 생각
- [벽] 1차: 오는 방향 관계없이 set에 넣어서 a↔b 관계가 안 묶임

### 코드 개선 방법
- [벽] 3차원 배열로 그 칸에서 못 가는 방향을 저장. 또는 a: b, b: a 양쪽 저장
- [룩업 테이블] 직진/왼/오 side 배열을 따로 빼기. 온풍기별 퍼지는 온도도 미리 계산하면 bfs 감소
- [동시 갱신] 온도 조절은 큰 쪽이 작은 쪽에 준다고만 보면 중복 걱정 없음
- [조건] 온도는 0 아래로 안 내려감
- [디버깅] continue / break는 디버깅 루틴에 넣어 의심

from collections import deque

# 함수

def is_range(r, c):
    return 0 <= r < size and 0 <= c < size

direction = {2: (0, -1), 3: (-1, 0), 4: (0, 1), 5: (1, 0)}
side = {2: [(-1, 0), (1, 0)],
        3: [(0, -1), (0, 1)],
        4: [(-1, 0), (1, 0)],
        5: [(0, -1), (0, 1)]}


# 1. 에어컨 쏴주는 함수
def air(sr, sc, d):
    fdr, fdc = direction[d]
    sr, sc = sr + fdr, sc + fdc
    if is_range(sr, sc):
        v = [[0] * size for _ in range(size)]
        v[sr][sc] = 5
        temp[sr][sc] += 5
        q = deque()
        q.append((sr, sc))
        while q:
            r, c = q.popleft()
            # 디버깅 포인트 !!!!!!!!!! 온도 0 이하로는 안 내려감

            # 1. 앞으로 가기
            fr, fc = r + fdr, c + fdc
            if is_range(fr, fc) and v[fr][fc] == 0:
                if (r, c) in walls and (fr, fc) in walls[(r, c)]:
                    pass

                else:
                    if v[r][c] > 1:
                        q.append((fr, fc))
                        v[fr][fc] = v[r][c] - 1
                        temp[fr][fc] += v[fr][fc]

            # 2. 양 옆으로 가기
            for sdr, sdc in side[d]:
                sir, sic = r + sdr, c + sdc
                if is_range(sir, sic): # 옆으로 갈 땐 v 체크 안 해도 돼
                    if (r, c) in walls and (sir, sic) in walls[(r, c)]:
                        continue

                    sider, sidec = sir + fdr, sic + fdc
                    if is_range(sider, sidec) and v[sider][sidec] == 0:
                        if (sir, sic) in walls and (sider, sidec) in walls[(sir, sic)]:
                            continue

                        if v[r][c] > 1:
                            q.append((sider, sidec))
                            v[sider][sidec] = v[r][c] - 1
                            temp[sider][sidec] += v[sider][sidec]


# 2. 공기 순환
def spread():
    new_temp = [[0] * size for _ in range(size)]
    for r in range(size):
        for c in range(size):
            for dr, dc in ((1, 0), (0, 1)):
                nr, nc = r + dr, c + dc

                if is_range(nr, nc):
                    if (r, c) in walls and (nr, nc) in walls[(r, c)]:
                        continue

                    diff = abs(temp[r][c] - temp[nr][nc]) // 4

                    if temp[r][c] > temp[nr][nc]:
                        new_temp[nr][nc] += diff
                        new_temp[r][c] -= diff
                    else:
                        new_temp[nr][nc] -= diff
                        new_temp[r][c] += diff

    for i in range(size):
        for j in range(size):
            temp[i][j] += new_temp[i][j]

# 3. 열 전도
def minus1():
    for r in range(size):
        for c in range(size):
            if r in {0, size-1} or c in {0, size-1}:
                if temp[r][c] > 0:
                    temp[r][c] -= 1

# 4. 온도 체크
def check():
    cnt = 0
    for r, c in goal_jwa:
        if temp[r][c] >= goal:
            cnt += 1

    if cnt == len(goal_jwa):
        return 1
    else:
        return 0

        if (r - 1, c) not in walls:

direction = {1: (0, 1), 2: (0, -1), 3: (-1, 0), 4: (1, 0)}
side = {1: (3, 4),

def on(sr, sc, d):
    v = [[0] * col for _ in range(row)]
    dr, dc = direction[d]
    sr, sc = sr + dr, sc + dc

    if is_range(sr, sc):
        v[sr][sc] = 5
        temp[sr][sc] += v[sr][sc]
        q = deque()
        q.append((sr, sc))
        while q:
            r, c = q.popleft()

            # 음수로 떨어지면 안 됨!
            if v[r][c] == 0:
                break

            dr, dc = direction[d]

            # 1. 내 바로 앞
            nr1, nc1 = r + dr, c + dc
            if is_range(nr1, nc1) and not find_wall(nr1, nc1, d):
                if v[nr1][nc1] == 0:
                    v[nr1][nc1] = v[r][c] - 1
                    temp[nr1][nc1] += v[nr1][nc1]
                    q.append((nr1, nc1))

            # 2. E 형태로 내 양쪽 옆
            for num in side[d]:
                side_dr, side_dc = direction[num]
                nr2, nc2 = r + side_dr, c + side_dc
                if is_range(nr2, nc2) and not find_wall(nr2, nc2, num):
                    nr3, nc3 = nr2 + dr, nc2 + dc
                    if is_range(nr3, nc3) and not find_wall(nr3, nc3, d):
                        if v[nr3][nc3] == 0:
                            v[nr3][nc3] = v[r][c]-1

def spread():
    # 이 과정은 모든 칸에 대해서 동시에 발생한다.
    v = [[0] * col for _ in range(row)]
    arr = [[0] * col for _ in range(row)]

    for r in range(row):
        for c in range(col): # 400 * 4
            plus_diff = 0
            minus_diff = 0
            v[r][c] = 1
            now_temp = temp[r][c]

            for idx in (1, 2, 3, 4):
                dr, dc = direction[idx]
                nr, nc = r + dr, c + dc

                # 전체 온도에서 퍼진 온도만큼 빠진다
                if is_range(nr, nc) and not find_wall(nr, nc, idx):
                    if v[nr][nc] == 0:
                        diff = (abs(now_temp - temp[nr][nc]) // 4)
                        if now_temp > temp[nr][nc]:
                            arr[nr][nc] += diff
                            minus_diff += diff
                        elif now_temp < temp[nr][nc]:
                            arr[nr][nc] -= diff
                            plus_diff += diff


            arr[r][c] += (plus_diff - minus_diff)

    for i in range(row):
        for j in range(col):
            arr[i][j] += temp[i][j]

    return arr
