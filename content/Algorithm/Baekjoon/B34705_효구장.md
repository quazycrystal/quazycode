# 부분 집합
def meat(n, sm):
    global ans
    if n == N:
        if sm >= mn and sm <= mx:
            ans = "YES"
        return
    meat(n+1, sm + meat_l[n]) # 포함
    meat(n+1, sm) # 미포함


T = int(input())
for tc in range(1, 1+T):
    mn, mx = map(int, input().split())
    meat_l = list(map(int, input().split()))
    N = 5
    ans = "NO"
    meat(0, 0)
    print(ans)
