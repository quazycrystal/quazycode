### 걸린 시간: 약 3시간 40분 (실패 횟수: 2회)

### 실패 / 성공 이유
- [정답 처리] 만족도 0일 때 처리를 안 해서 점수가 더해짐
- [함수 설계] 함수화 실패로 중복 변수명이 생겨 다 밀고 다시. 문제와 생각을 섞어서 구상한 탓
- [시간 복잡도] 완탐 대신 빈 칸 많은 것만 뽑아서 틀림. 4N^4 = 16만이라 완탐 충분

### 코드 개선 방법
- [시간 복잡도] N이 50 이하면 탐색에 겁먹지 말기
- [정렬] key=lambda x: (x[0], x[1]). 동점이면 알아서 다음 기준으로 정렬
- [디버깅] 랜덤 테케를 하나 더 만들어 넣으니 2분 만에 원인 찾음
- [dict] {v: k for k, v in d.items()}로 뒤집기. items, get, update, setdefault, fromkeys, popitem

### 코드
```python
# 입력
size = int(input())
want = dict()  # 입력으로 받는 원하는 애들
sunseo = []
for _ in range(size * size):
    p_name, *lst = map(int, input().split())
    want[p_name] = lst
    sunseo.append(p_name)  # 딕셔너리에는 순서가 없으므로, 순서 뽑아올 것

seat = dict()
# 이미 앉은 애들 애 번호: (좌표 튜플)
arr = [[0] * size for _ in range(size)]


def step1(name):  # 가능한 자리 리스트를 다음으로 넘겨 줘야 함
    # 1. 들어온 애가 좋아하는 애한테 접근 한다
    like_lst = want[name]
    temp = []
    pos_seat = []
    for like in like_lst:
        if like in seat:
            temp.append(seat[like])

    # 3. 좋아하는 애 인접 좌표 넘기기 - 내 자리가 될 수 있는 곳
    cnt_adj = dict()  # 여기에 밑에서 똑같은 좌표 여러번 나오면 값을 올려줄 것임
    # 똑같은 좌표 많다는 것은 내 옆에 좋아하는 애가 많이 온다는 것
    if temp:
        for tpl in temp:
            r, c = tpl
            for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                nr, nc = r + dr, c + dc
                if 0 <= nr < size and 0 <= nc < size:
                    if (nr, nc) in cnt_adj and arr[nr][nc] == 0:  # 다음 자리에 숫자가 이미 차 있으면 불가능
                        cnt_adj[(nr, nc)] += 1
                    else:
                        cnt_adj[(nr, nc)] = 1
        mx = 0
        for loc in cnt_adj:  # loc은 좌표
            if cnt_adj[loc] >= mx:
                if cnt_adj[loc] > mx:
                    pos_seat = []
                    mx = cnt_adj[loc]
                pos_seat.append(loc)

    # 3-1 좌표가 아예 없으면 빈 리스트가 넘어감 (가능 리스트)
    return pos_seat


def step2(lst):
    # 앞에서 추려진 가능 리스트랑, 현재 자리 정해질 사람 이름 받는다

    pos_seat = []

    if lst:

        mx = 0  # 밖에 있어야 전체 lst 중 최대를 뽑음

        for tpl in lst:
            r, c = tpl

            cnt = 0  # 빈 칸 세요
            if arr[r][c] == 0:
                for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < size and 0 <= nc < size:
                        if arr[nr][nc] == 0:  # 가능한 곳 주변이 얼마나 비었는지
                            cnt += 1

                if cnt >= mx:
                    if cnt > mx:
                        pos_seat = []
                        mx = cnt
                    pos_seat.append((r, c))

    return pos_seat  # 여전히 좋아하는 학생이 자리 안 잡았으면 빈 리스트 준다


def step3(lst):
    ans = ()

    if lst:
        lst.sort(key = lambda x: (x[0], x[1]))  # 인접 학생 칸도 같고, 빈 칸 갯수도 같으면 좌표값 제일 빠른 거

        ans = (min_r, min_c)

    else:  # 앞에서 좋아하는 학생이 단 한 명도 자리에 안 앉았다!! (초기 경우)
        # 인접 빈 칸이 가장 많은 칸을 줘야 함, 가장 빠른 거!
        temp = []
        mx = 0
        for row in range(size):
            for col in range(size):
                cnt = 0
                if arr[row][col] == 0:  # 일단 나도 비어있고
                    for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                        nr, nc = row + dr, col + dc
                        if 0 <= nr < size and 0 <= nc < size:
                            if arr[nr][nc] == 0:  # 가능한 곳 주변이 얼마나 비었는지
                                cnt += 1

                    if cnt >= mx:
                        if cnt > mx:
                            temp = []
                            mx = cnt
                        temp.append((row, col))

        ans = temp[0]  # 제일 처음으로 들어간 거

    return ans


for ppl in sunseo:
    pos = step1(ppl)
    pos2 = step2(pos)
    ans = step3(pos2)
    r, c = ans
    # 좌표 추가, arr 표시
    seat[ppl] = ans
    arr[r][c] = ppl

satis = 0
for i in range(size):
    for j in range(size):
        name = arr[i][j]
        like_lst = set(want[name])
        cnt = 0
        for di, dj in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            ni, nj = i + di, j + dj
            if 0 <= ni < size and 0 <= nj < size:
                if arr[ni][nj] in like_lst:
                    cnt += 1

        if cnt != 0:
            satis += 10 ** (cnt - 1)

print(satis)
```
