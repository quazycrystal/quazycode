row, col = map(int, input().split())
arr = [list(input()) for _ in range(row)]

def hcheck(s):
    cnt = s
    for rr in range(row):
        r_cnt = 0
        for cc in range(col):
            if r_cnt == cnt:
                start.append((rr, (cc - 1)))
                r_cnt = 0

            if arr[rr][cc] == "*":
                r_cnt += 1
            else:
                r_cnt = 0
    return (cnt, start)
