### 코드 개선 방법
- [정렬] 방문 좌표를 row, col 기준으로 각각 sort해 최대 최소

### 코드
```python
direction = {0: (1, 0), 1: (0, 1), 2: (-1, 0), 3: (0, -1)}
# 상, 우, 하, 좌, 시계방향(처음에 북쪽)
d_idx = 0

def turn(com):
    global d_idx
    if com == "L":
        d_idx -= 1
        d_idx = d_idx % 4

    if com == "R":
        d_idx += 1
        d_idx = d_idx % 4

    return direction[d_idx]

def move(com, jwa, direct):
    r, c = jwa[0], jwa[1]
    dr, dc = direct[0], direct[1]
    nr, nc = 0, 0

    if com == "F":
        nr, nc = r + dr, c + dc
    elif com == "B":
        nr, nc = r - dr, c - dc
    else:
        nr, nc = r, c # 다른거 들어오면 이동 안 함

    return((nr, nc))


T = int(input())
for tc in range(1, 1+T):
    start = (0, 0)
    commands = input()
    trail = [(0, 0)] # 시작점 넣어두기
    for command in commands:
        d = turn(command)  # 방향 정해짐
        start = move(command, start, d)
        trail.append(start) # 이동한 좌표 넣기

    c_trail = trail # col sort 할 거 하나 더

    trail.sort() # row 기준 sort
    row = trail[-1][0] - trail[0][0]

    c_trail.sort(key = lambda x: x[1]) # col 기준 sort
    col = c_trail[-1][1] - c_trail[0][1]

    print(row*col)
```
