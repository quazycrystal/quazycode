### 코드 개선 방법
- [BFS] 유닛 좌상단 좌표로 bfs. 이동할 때 유닛 영역 전체가 장애물과 겹치지 않는지 확인
- [시간 복잡도] 500 * 500 * 10 * 10 = 2500만

### 코드
```python
# B2194 유닛 이동시키기

# 시간 총 20분 정도 소요

from collections import deque

# 함수

def is_range(r, c):
    return 0 <= r < b_row and 0 <= c < b_col
# 1. bfs 이동
def move(sr, sc):
    v = [[-1] * b_col for _ in range(b_row)]
    v[sr][sc] = 0
    q = deque()
    q.append((sr, sc))
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if is_range(nr, nc) and v[nr][nc] == -1:
                # 조건 - 장애물에 겹치면 안 됨
                flag = 0
                for i in range(s_row):
                    for j in range(s_col): # 500 * 500 * 10 * 10 = 2500만
                        if not is_range(nr + i, nc + j):
                            flag = 1
                            break
                        else:
                            if arr[nr + i][nc + j] == 1:
                                flag = 1
                                break

                    if flag == 1:
                        break

                if not flag:
                    v[nr][nc] = v[r][c] + 1
                    q.append((nr, nc))

    return v[er][ec]


# 입력
b_row, b_col, s_row, s_col, o_num = map(int, input().split())
arr = [[0] * b_col for _ in range(b_row)]

# 장애물
for _ in range(o_num):
    obr, obc = map(int, input().split())
    obr, obc = obr-1, obc-1
    arr[obr][obc] = 1

# 시작, 끝
sr, sc = map(int, input().split())
sr, sc = sr-1, sc-1

er, ec = map(int, input().split())
er, ec = er-1, ec-1

# 풀이
print(move(sr, sc))
```
