def btk(d, s, lst):  # 깊이는 전체 집
    if d == daepi:
        mx = 0 # 한 경우의 수 안에서 최소 대피소 거리만 모여 있음
        for combi in range(house):
            if combi not in lst:
                # mn: 한 집의 위치에서 가장 가까운 대피소 거리
                mn = 200001
                for eli in lst:  # 이 루프가 끝나면 대피소까지의 최솟값 확정
                    pos = map[combi][eli]
                    if pos < mn:
                        mn = pos

                if mx < mn:
                    mx = mn

        ans.append(mx) # 이것 때매 제대로 안 나온 듯

        return  # return은 아무튼 조건 되면 하긴 하고

    for choice in range(s, house):
        lst.append(choice)
        btk(d + 1, choice + 1, lst)
        lst.pop()


# ===================================
house, daepi = map(int, input().split())

homes = []
for _ in range(house):
    r, c = map(int, input().split())
    homes.append((r, c))

map = [[] for _ in range(house)]
for i in range(house):
    for j in range(house):
        srt = homes[i]
        end = homes[j]
        distance = abs(srt[0] - end[0]) + abs(srt[1] - end[1])
        map[i].append(distance)

# 백트래킹으로 나온 조합 여기로 - map에서 찾는다
ans = []
btk(0, 0, [])
print(min(ans)) # 여러 경우의 수를 모두 보고 최대 거리 중 최솟값
