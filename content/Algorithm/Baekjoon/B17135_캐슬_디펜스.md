technique: [패딩]

### 걸린 시간: 1시간 정도 (10시 - 11시) (실패 횟수: -)

### 실패 / 성공 이유
- [함수 설계] 함수 안에서 바꾼 값은 꼭 return

### 코드 개선 방법
- [BFS] 궁수마다 bfs로 공격 대상 찾기. 거리 짧고 가장 왼쪽
- [동시 갱신] 궁수들이 고른 적은 set으로 모아 한 번에 제거
- [패딩] 성 줄을 패딩. 적 이동은 한 줄씩 내리기

### 코드
```python
from collections import deque
from itertools import *

# 함수:
def is_range(r, c):
    return 0 <= r < row and 0 <= c < col

# 1. bfs로 공격 대상 찾는 함수
def attack(jwa, arr):
    sr, sc = jwa
    v = [[-1] * col for _ in range(row+1)]
    v[sr][sc] = 0
    q = deque()
    q.append((sr, sc))
    pos = []
    pick = (-100, -100)
    while q:
        r, c = q.popleft()
        if v[r][c] > dist:
            break

        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if is_range(nr, nc) and v[nr][nc] == -1:
                if arr[nr][nc] != 2:
                    v[nr][nc] = v[r][c] + 1
                    q.append((nr, nc))
                    if arr[nr][nc] == 1:
                        pos.append((v[nr][nc], nr, nc))

    if pos:
        pos.sort(key=lambda x: (x[0], x[2]))  # 거리 가장 짧고, 가장 왼쪽
        if pos[0][0] <= dist:
            pick = (pos[0][1], pos[0][2])

    return pick

# 2. 적 이동하고, 원래 위치에서 지우기
def enemy(hit, copy_arr):
    for hr, hc in hit:
        copy_arr[hr][hc] = 0

    # 한 바꾸 돌림
    copy_arr = [[0] * col] + copy_arr[0:row-1] + [[2] * col]

    return copy_arr # 함수 안에서 쓰면 꼭 리턴 해줘야!!

# 2. 궁수 골라서 한번에 쌈싸먹는 함수
def locate():
    lst = []
    for c in range(col):
        lst.append((row, c))

    ans = 0
    for combi in combinations(lst, 3):
        if ans == mx_ans:
            return ans

        copy_arr = [line[:] for line in arr]

        pos_ans = 0
        for rep in range(row-mn_row):
            hit = set()
            for i in range(3):
                got = attack(combi[i], copy_arr)

                if got != (-100, -100):
                    hit.add(got)
            pos_ans += len(hit)
            copy_arr = enemy(hit, copy_arr)
        ans = max(ans, pos_ans)

    return ans

# 입력
row, col, dist = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)] + [[2] * col] # 패딩

# 종료조건
mn_row = row + 1
mx_ans = 0
for r in range(row):
    for c in range(col):
        if arr[r][c] == 1:
            mx_ans += 1
            if mn_row > r:
                mn_row = r

if mn_row == row+1:
    print(0) # 적이 없다
else:
    print(locate())
```
