### 코드 개선 방법
- [초기화] 1, -1을 미리 채우고 시작

### 코드
```python
# B10709 기상 캐스터
row, col = map(int, input().split())
arr = [list(input()) for _ in range(row)]
v = [[0] * col for _ in range(row)]
min = 0
q = []

for i in range(row):
    for j in range(col):
        if arr[i][j] == "c":
            v[i][j] = 0
            q.append((i, j))
        else:
            v[i][j] = -1 # 초깃값: 미리 1, -1 채우고 간다

while q:
    min += 1
    for k in range(len(q)):
        cur = q.pop(0)
        r, c = cur[0], cur[1]
        nr, nc = r, c+1
        if 0 <= nr < row and 0 <= nc < col:
            if v[nr][nc] == -1:
                v[nr][nc] = min
            q.append((nr, nc))

for rr in v:
    print(*rr)
```
