algorithm: [BFS, 시뮬레이션, 정렬]
data_structure: [2차원 배열, 덱]
technique: [중력, 배열 회전]
---

### 걸린 시간: 2시간 (구상 20분, 구현 1시간 10분, 디버깅 30분) (실패 횟수: 1회 (1% 틀렸습니다))

### 실패 / 성공 이유
- [방문 처리] 무지개 블록(0)까지 v 배열을 공유해서 다른 그룹에서 못 셈. 무지개는 bfs 안에서 따로 세야

### 코드 개선 방법
- [방문 처리] 색 블록은 v 공유(기준 블록 중복 방지), 무지개는 함수 안에서 매번 새로
- [중력] 위아래 순회하며 검은 블록 위에 쌓기. 회전은 반시계 90 = 시계 270
- [정렬] (크기, 무지개 수, 행, 열) 큰 것부터
- [속도] 슬라이싱 복사는 요소 수에 비례
- [습관] 문제는 메모하며 3번 읽기. 막히면 잠깐 나갔다 오기

### 코드
```python
# B21609 상어 중학교

# 가능 애들 모아서 리스트 sort도 있긴 하다

from collections import deque

# 입력
size, color_num = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(size)]

def bfs_choose(sr, sc, color):
    v0 = [[0] * size for _ in range(size)]
    # 0 세는 배열 또 필요 흑
    v[sr][sc] = 1
    q = deque()
    q.append((sr, sc))
    rainbow = 0
    b_size = 0
    while q:
        b_size += 1
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < size and 0 <= nc < size:
                # 무지개 블록 센다
                if v[nr][nc] == 0 and v0[nr][nc] == 0:
                    if arr[nr][nc] == 0:
                        # v[nr][nc] = 1 # 다른 애들 셀 때 또 연결될 수 있음
                        q.append((nr, nc))
                        v0[nr][nc] = 1
                        rainbow += 1
                    elif arr[nr][nc] == color:
                        v[nr][nc] = 1
                        q.append((nr, nc))
                    # -1이거나 다른 색깔이면 큐에 안 넣음

    return b_size, rainbow, sr, sc  # 기준 블록은 맨 처음


# 1-1 정답 넣고 터뜨리는 함수
def bfs_pop(sr, sc):
    inner_v = [[0] * size for _ in range(size)]
    color = arr[sr][sc]
    q = deque()
    q.append((sr, sc))
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < size and 0 <= nc < size:
                # 무지개 블록 센다
                if inner_v[nr][nc] == 0:
                    if arr[nr][nc] == 0 or arr[nr][nc] == color:
                        arr[nr][nc] = -2
                        q.append((nr, nc))
                    # -1이거나 다른 색깔이면 큐에 안 넣음


def gravity(board):
    b_copy = [line[:] for line in board]
    for col in range(size):
        temp = []
        for row in range(size):
            if b_copy[row][col] >= 0:
                # 색 블록이면 temp로
                temp.append(b_copy[row][col])
                b_copy[row][col] = -2  # 자리 비우기
            elif b_copy[row][col] == -1:
                # 뒤에 보지 말고 바로 돌아서 역으로 채우기
                for new_row in range(row - 1, -1, -1):
                    # 현 위치는 -1이니까 그 전칸부터 채우기
                    if temp:
                        b_copy[new_row][col] = temp.pop()
                    else:
                        break

        for n_row in range(size - 1, -1, -1):
            if temp:
                b_copy[n_row][col] = temp.pop()
            else:
                break
    return b_copy


ans = 0
while True:
    able_pop = []
    v = [[0] * size for _ in range(size)]
    for i in range(size):
        for j in range(size):
            if 0 < arr[i][j] and v[i][j] == 0:
                picked_color = arr[i][j]
                b_size, rainbow, sr, sc = bfs_choose(i, j, picked_color)
                if b_size > 1:  # 블록은 2부터!
                    able_pop.append((b_size, rainbow, sr, sc))

    # 터뜨릴 수 있는 거 없으면 탈출
    if not able_pop:
        break
    else:
        able_pop.sort(reverse=True)  # 큰 것 부터
        big_block = able_pop[0]

        ans += (big_block[0] ** 2)  # 사이즈 제곱만큼 점수에 더해짐
        bfs_pop(big_block[2], big_block[3])  # 여기서 arr 바꾼다

        # === 중력 구간 ===
        new_arr = gravity(arr)

        # 반시계 90 = 시계 270
        new_arr = [list(line[:]) for line in zip(*new_arr)][::-1]

        final_arr = gravity(new_arr)
        arr = final_arr

print(ans)

```
