### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [범위] range(1, 5)로 해서 강제 4지선다가 됨. range(1, 6)

### 코드 개선 방법
- [조건] 선택 가능: 연속 cnt가 2보다 작거나, 2여도 이전과 다른 경우. 조건 연산자로 정리

### 코드
```python
# B19949 영재의 시험
def btk(d, cnt, last, score):
    global ans
    if d == 10:
        if score >= 5:
            ans += 1
        return

    check = ans_lst[d]
    for i in range(1, 6): # 바보 아님???? 1,5로 했음... 강제 4지선다행
        if cnt == 2:
            if i != last: # 이전과 다름
                if i == check: # 답과 일치
                    btk(d+1, 1, i, score + 1)
                else:
                    btk(d + 1, 1, i, score)

        if cnt < 2:
            if i != last: # 이전과 다름
                if i == check:
                    btk(d + 1, 1, i, score+1)
                else:
                    btk(d + 1, 1, i, score)
            else: # 이전과 같음
                if i == check: # 답이면 +1
                    btk(d + 1, cnt + 1, i, score+1)
                else:
                    btk(d + 1, cnt + 1, i, score)

ans = 0
ans_lst = list(map(int, input().split()))
btk(0, 0, 0, 0)
print(ans)

# 리팩토링
def btk(d, cnt, last, score):
    global ans
    if d == 10:
        if score >= 5:
            ans += 1
        return

    check = ans_lst[d]
    for i in range(1, 6): # 바보 아님???? (1, 5)로 했음... 강제 4지선다행
        if cnt == 2 and i == last: # 이전과 다름
            continue

        if i == check: # 답과 일치
            btk(d+1, (cnt + 1 if i == last else 1), i, score + 1) # 조건 연산자
        else:
            btk(d+1, (cnt + 1 if i == last else 1), i, score)


ans = 0
ans_lst = list(map(int, input().split()))
btk(0, 0, 0, 0)
print(ans)
```
