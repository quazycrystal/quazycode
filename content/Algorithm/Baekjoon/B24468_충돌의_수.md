`W4/B24468_충돌의 수.py`

```python
# R = +1, L = -1
length, balls, time = map(int, input().split())
stat = [[] for _ in range(2)]
for _ in range(balls):
    loc, direct = input().split()
    loc = int(loc)
    if direct == "L":
        direct = -1
    else:
        direct = 1
    stat[0].append(loc)
    stat[1].append(direct)

ans = 0
# 시뮬레이션
for i in range(time):
    # 이동
    for ball in range(balls): # stat의 loc, direct 갯수
        stat[0][ball] += stat[1][ball] # 위치에 방향 더하기

        if stat[0][ball] == 0 or stat[0][ball] == length:
            stat[1][ball] = -stat[1][ball]

        # 공끼리 부딪힘
        for check in range(balls):
            if stat[0][ball] == stat[0][check]: # 위치 겹친다
                if ball != check: # 자기 자신 제외
                    # 충돌했으므로 방향 바꾼다
                    stat[1][ball] = -stat[1][ball]
                    stat[1][check] = -stat[1][ball]
                    ans += 1

print(ans)

```
