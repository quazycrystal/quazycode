`W4/B1874_스택 수열.py`

```python
N = int(input())
lst = [int(input()) for _ in range(N)]
opr = []
stk = []
i = 1

for num in lst:
    while i <= num:
        stk.append(i)
        opr.append("+")
        i += 1

    if stk and stk[-1] == num:
        stk.pop()
        opr.append("-")

    else:
        print("NO")
        break

if not stk:
    for i in opr:
        print(i)

```
