### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [BFS] 익은 토마토를 동시에 큐에 넣어야 함. 큐를 밖으로 빼기
- [큐] q가 비면 popleft 에러, 방지문 필수
- [조건] 끝나고 0이 남아 있으면 -1

### 코드 개선 방법
- [로직] 못 익는 토마토 미리 확인은 필요 없음. BFS가 알아서 끝남
- [로직] 1은 시작점에서만 보므로 다시 볼 필요 없음

from collections import deque

def tomato(): # 외부에서 q에다가 넣어버릴거임...
    global q
    global ans
    if q:
        init = q.popleft()
        ir, ic = init[0], init[1]
        v[ir][ic] = 0
        q.append(init) # 초기값 이미 큐에 들어있지만 pop 했기 때문에 다시 추가
        while q:
            cur = q.popleft()
            r, c = cur[0], cur[1]
            for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                nr = r + dr
                nc = c + dc
                if 0 <= nr < row and 0 <= nc < col:
                    if v[nr][nc] == "x":

                        if arr[nr][nc] == 0:
                            # arr[nr][nc] == 1
                            v[nr][nc] = v[r][c] + 1
                            q.append((nr, nc))

col, row = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)]
v = [["x"] * col for _ in range(row)]
q = deque()
ans = 0

# 사전 탐색
for i in range(row):
    for j in range(col):
        if arr[i][j] == 1:
            v[i][j] = 0 # 시작점이라 방문 횟수에 영향 안 주므로 0로
            q.append((i, j))
        if arr[i][j] == -1:
            v[i][j] = -1 # 방문 리스트 -1

tomato()

for rr in v:
    if "x" in rr:
        ans = -1
        break

if ans != -1: # 이 때 "x" 걸러질텐데?
    ans = max(map(max, v))

print(ans)

from collections import deque

def BFS(queue):
    while queue:
        r, c = queue.popleft()
        for idx in range(4):
            nr = r + dy[idx]
            nc = c + dx[idx]
            if 0<=nr<N and 0<=nc<M:
                if box[nr][nc]==0:
                    box[nr][nc]=box[r][c] + 1
                    queue.append((nr, nc))

# col, row

box = [list(map(int,input().split())) for _ in range(N)]

dx = [-1,1,0,0]
dy = [0,0,-1,1]

queue = deque()

# 익은 사과 큐에 추가
for i in range(N):
    for j in range(M):
        if box[i][j] == 1:
            queue.append((i, j))
BFS(queue)
