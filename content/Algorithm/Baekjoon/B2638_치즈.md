### 코드 개선 방법
- [BFS] 외곽에서 bfs로 바깥 공기만 방문 체크. 바깥 공기와 2면 이상 닿은 치즈만 녹이기
- [동시 갱신] 녹을 치즈를 다 모은 뒤 한 번에 녹이기

### 코드
```python
# B2638 치즈

from collections import deque

# 함수
def is_range(r, c):
    return 0 <= r < row and 0 <= c < col


# 1. 경계 찾는 함수
def outside(sr, sc):
    out = [[0] * col for _ in range(row)]
    out[sr][sc] = 1
    q = deque()
    q.append((sr, sc))
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if is_range(nr, nc) and out[nr][nc] == 0:
                if cheese[nr][nc] == 0:
                    out[nr][nc] = 1
                    q.append((nr, nc))
    return out


# 2. 녹을 치즈 좌표 찾는 함수
def get_melt(sr, sc, v):
    v[sr][sc] = 1
    q = deque()
    q.append((sr, sc))
    to_melt = []
    while q:
        r, c = q.popleft()
        cnt = 0
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if is_range(nr, nc):
                # 조건!
                if out[nr][nc] == 1:
                    cnt += 1

                if v[nr][nc] == 0 and cheese[nr][nc] == 1:
                    v[nr][nc] = 1
                    q.append((nr, nc))

        if cnt > 1:
            to_melt.append((r, c))

    return to_melt


# 3. 전체 녹을 치즈 알아내기
def melt():
    full_melt = []
    v = [[0] * col for _ in range(row)]
    for i in range(row):
        for j in range(col):
            if cheese[i][j] == 1 and v[i][j] == 0:
                to_melt = get_melt(i, j, v)
                full_melt.extend(to_melt)

    return full_melt


# 4. 다 녹이기
def disappear(lst):
    for r, c in lst:
        cheese[r][c] = 0


# 5. 종료 체크
def check():
    go = 0
    for rr in cheese:
        go += sum(rr)
    return go


# 입력
row, col = map(int, input().split())
cheese = [list(map(int, input().split())) for _ in range(row)]

ans = 0
while True:
    ans += 1
    out = outside(0, 0)
    full_melt = melt()
    disappear(full_melt)
    go = check()
    if not go:
        break

print(ans)
```
