# bfs 구현
def bfs(s):
    global v
    global ans
    v = {s:0} #lv 초기값
    q = []
    q.append(s)
    while q:
        cur = q.pop(0)
        if cur == goal:
            ans = v[cur]

        # 액션 2개
        n1 = cur + up
        n2 = cur - down

        if n1 not in v: # 중복 방지
            if 0 < n1 <= end:
                v[n1] = v[cur] + 1
                q.append(n1)
                if n1 == goal:
                    ans = v[n1]
        if n2 not in v:
            if 0 < n2 <= end:
                v[n2] = v[cur] + 1
                q.append(n2)
                if n2 == goal:
                    ans = v[n2]

end, srt, goal, up, down = map(int, input().split())

ans = -1
bfs(srt)
if ans == -1:
    ans = "use the stairs"

print(ans)
