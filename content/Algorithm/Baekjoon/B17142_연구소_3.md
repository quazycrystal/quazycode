### 걸린 시간: 54분 (12:49 - 13:43, 구상 15분, 구현 25분, 검증 14분) (실패 횟수: -)

### 실패 / 성공 이유
- [문제 읽기] 비활성 바이러스 = 빈 칸처럼 퍼짐. 단 다 퍼진 뒤 바이러스만 남으면 시간에 안 셈

### 코드 개선 방법
- [조합] 바이러스 좌표 리스트에서 itertools로 m개 뽑아 큐에 넣고 동시에 bfs
- [가지치기] 빈 칸 수를 세두고 꽉 찬 순간 종료
- [정답] 방문 배열에 -1(못 닿은 빈 칸)이 남으면 -1
- [검증] -1 나오는 케이스, 중간·마지막에 바이러스만 남는 케이스 확인
- [시간 복잡도] bfs(50*50) * 10C5(252)

### 코드
```python
# 구현: 25분

# -1이 남는 경우는 오로지 빈 칸, 닿지 않았을 때

from itertools import *
from collections import deque

# bfs 함수
def bfs():
    global q
    v = [[-1] * size for _ in range(size)]
    for sr, sc in q:
        v[sr][sc] = 0

    bfs0 = 0
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < size and 0 <= nc < size:
                if v[nr][nc] == -1:
                    # 여기서 종료
                    if bfs0 == cnt0: # 빈 칸 꽉 찼으면 더 볼 필요 없음
                        break

                    if arr[nr][nc] == 0:
                        v[nr][nc] = v[r][c] + 1
                        q.append((nr, nc))
                        bfs0 += 1

                    elif arr[nr][nc] == 2:
                        v[nr][nc] = v[r][c] + 1
                        q.append((nr, nc))

    if bfs0 == cnt0: # 모든 빈 칸이 다 찼다면
        pos_ans = 0
        for rr in v:
            pos_ans = max(pos_ans, max(rr))

        return pos_ans

    else:
        return -1


# 입력
size, virus_num = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(size)]
virus_lst = []

cnt0 = 0
for i in range(size):
    for j in range(size):
        if arr[i][j] == 2:
            virus_lst.append((i, j))
        if arr[i][j] == 0:
            cnt0 += 1

ans = 10**18
for combi in combinations(virus_lst, virus_num):
    if ans == 0:
        break # 0이 최소!

    q = deque(combi)
    pos_ans = bfs()
    if pos_ans != -1:
        ans = min(pos_ans, ans)

if ans == 10**18:
    ans = -1

print(ans)

```
