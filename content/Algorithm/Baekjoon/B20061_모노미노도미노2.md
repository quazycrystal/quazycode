### 걸린 시간: 구상 30분, 구현 2시간, 시간 종료 후 20분 (실패 횟수: 1회 (WA))

### 실패 / 성공 이유
- [중력] 밑에서부터 단순히 블록을 놓아서, 위에 뜬 블록 사이 빈 공간에 새 블록이 놓임
- [중력] 내려온 블록의 원래 위치를 안 지움
- [표시] 행/열을 1, 2로 표시했는데 정답이 개수 세기라 다 1로 바꿈

### 코드 개선 방법
- [중력] 위에서부터 내려오다 막히기 직전 칸에 놓기. 뒤를 클리어하며 내리기
- [자료구조] 파랑은 전치해서 초록과 같은 함수로 행만 관리
- [범위] 범위 처리와 같이 가서 예외처리 줄이기
- [순서] 꽉 찬 줄 터뜨리기 → 연한 칸(4, 5번 줄) 처리

### 코드
```python
# B20061 모노미노도미노2

def green_block(t, r, c):

    for row in range(4, 10):
        if t == 1:
            nxt = arr[row][c]
            if nxt == 1:
                arr[row-1][c] = 1
                break
            elif row == 9:
                arr[row][c] = 1

        elif t == 2:
            left = arr[row][c]
            right = arr[row][c+1]
            if left == 1 or right == 1:
                arr[row-1][c] = 1
                arr[row-1][c+1] = 1
                break
            elif row == 9:
                arr[row][c] = 1
                arr[row][c + 1] = 1


        elif t == 3:
            bottom = arr[row][c]
            up = arr[row-1][c]
            if up == 0 and bottom == 1:
                arr[row - 1][c] = 1
                arr[row - 2][c] = 1
                break

            elif up == 1:
                arr[row - 2][c] = 1
                arr[row - 3][c] = 1
                break

            elif row == 9:
                arr[row-1][c] = 1
                arr[row][c] = 1


def blue_block(t, r, c):

    for col in range(4, 10): # 초록 범위 거꾸로
        if t == 1:
            nxt = arr[r][col]
            if nxt == 1:
                arr[r][col-1] = 1 # 하나면 볼 것도 없이 즉시 표시
                break
            elif nxt == 0 and col == 9:
                arr[r][col] = 1

        elif t == 2:
            left = arr[r][col-1]
            right = arr[r][col]

            if left == 1:
                arr[r][col-2] = 1
                arr[r][col-3] = 1
                break

            elif left == 0 and right == 1:
                arr[r][col - 1] = 1
                arr[r][col - 2] = 1
                break

            elif left == 0 and right == 0 and col == 9:
                arr[r][col - 1] = 1
                arr[r][col] = 1


        elif t == 3:
            up = arr[r][col]
            down = arr[r+1][col]
            if up == 1 or down == 1:
                arr[r][col-1] = 1
                arr[r+1][col-1] = 1
                break

            elif up == 0 and down == 0 and col == 9:
                arr[r][col] = 1
                arr[r+1][col] = 1

def check():
    row_cnt = []
    col_cnt = []
    for row in range(9, 3, -1): # 초록 범위 거꾸로
        if all(arr[row][j] for j in range(4)) == 1:
            arr[row][0] = 0
            arr[row][1] = 0
            arr[row][2] = 0
            arr[row][3] = 0
            row_cnt.append(row)

    for col in range(9, 3, -1): # 초록 범위 거꾸로
        if all(arr[i][col] for i in range(4)) == 1:
            arr[0][col] = 0
            arr[1][col] = 0
            arr[2][col] = 0
            arr[3][col] = 0
            col_cnt.append(col)

    return row_cnt, col_cnt

# 입력
b_num = int(input())
arr = [[0] * 10 for _ in range(10)]

def gravity_green():
    new_arr = []
    for row in range(9, 3, -1):
        if any(arr[row][i] for i in range(10)) != 0:
            new_arr.append(arr[row])
        arr[row] = [0] * 10

    length = len(new_arr)
    for r in range(9, 9-length, -1):
        arr[r] = new_arr[9-r]

    # 옮기고도 만약 4, 5번줄에 있으면 한 번 더
    cnt = 0
    for row in range(4, 6):
        if any(arr[row][i] for i in range(10)) != 0:
            cnt += 1

    if cnt != 0:
        for move in range(9-cnt, 3, -1):
            arr[move+cnt] = arr[move]
            arr[move] = [0] * 10 # 뒤에를 클리어해가면서 가야한다

def gravity_blue():
    new_arr = []
    for col in range(9, 3, -1):
        if any(arr[i][col] for i in range(10)) != 0:
            new_arr.append([line[col] for line in arr])

        for r in range(10):
            arr[r][col] = 0

    length = len(new_arr)
    for c in range(9, 9-length, -1):
        for rr in range(10):
            arr[rr][c] = new_arr[9-c][rr]

    # 중력 다 하고도 위에 남았는지
    cnt = 0
    for col in range(4, 6):
        if any(arr[i][col] for i in range(10)) != 0:
            cnt += 1

    if cnt != 0:
        for c in range(9-cnt, 3, -1):
            for rr in range(10):
                arr[rr][c+cnt] = arr[rr][c]
                arr[rr][c] = 0

ans = 0
for _ in range(b_num):
    t, r, c = map(int, input().split())
    green_block(t, r, c)
    blue_block(t, r, c)
    row_cnt, col_cnt = check()

    if row_cnt:
        ans += len(row_cnt)
    if col_cnt:
        ans += len(col_cnt)

    gravity_green()
    gravity_blue()

block_sm = 0
for rr in arr:
    block_sm += sum(rr)

print(ans)
print(block_sm)
```
