### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [슬라이싱] arr[::-1][i]는 뒤집은 새 객체에서 행 전체를 가져옴. arr[i][::-1]도 뒤집힘
- [크기] 90도 회전마다 행열 크기가 바뀌어 고정 col로 만들면 안 됨. 4등분 회전은 크기 그대로

### 코드 개선 방법
- [배열 회전] 열을 가져오려면 리스트 컴프리헨션. 전치 후 상하/좌우 반전으로도 회전
- [슬라이싱] arr 앞에 슬라이스 쓰지 말고 뒤에 윈도우

### 코드
```python
def opr1():
    global arr
    # 열은 변함 없고 행 순서만 바뀜
    arr = arr[::-1]


def opr2():
    global arr
    # 행은 변함 없고 열 순서만 바뀜
    arr = [line[::-1] for line in arr]


def opr3():
    global arr

    new_arr = [[] for _ in range(len(arr[0]))]
    # 90도 회전 시키므로 열이 행이 됨

    arr = arr[::-1]  # 일단 아래 위 뒤집어
    # 여기서 열만 가져 와야 함 - 리스트 컴프리헨션

    for i in range(len(arr[0])):
        new_arr[i] = [r[i] for r in arr]
        # 전체 행의 i번째 데이터만 가져온다 - 열만 가져오는 셈

    arr = new_arr


def opr4():
    global arr
    new_arr = [[] for _ in range(len(arr[0]))]
    arr = [line[::-1] for line in arr]
    # 열 접근하려면 무조건 리스트 컴프리헨션

    for i in range(len(arr[0])):
        new_arr[i] = [r[i] for r in arr]
        # 전체 행의 i번째 데이터만 가져온다 - 열만 가져오는 셈

    arr = new_arr


def opr5():
    global arr
    r_mid = len(arr) // 2
    c_mid = len(arr[0]) // 2  # 여기도 자꾸 뒤집힘
    # 중심 기준 4덩이로 나눈다
    arr1 = [r[:c_mid] for r in arr[:r_mid]]  # 왼쪽 위
    arr2 = [r[c_mid:] for r in arr[:r_mid]]  # 오른쪽 위
    arr3 = [r[c_mid:] for r in arr[r_mid:]]  # 오른쪽 아래
    arr4 = [r[:c_mid] for r in arr[r_mid:]]  # 왼쪽 아래

    new_arr = [[] for _ in range(len(arr))]  # 여기도 계속 뒤집힘

    # 윗 반
    for i in range(r_mid):
        new_arr[i] = arr4[i] + arr1[i]

    # 아랫 반
    for j in range(r_mid):
        new_arr[j + r_mid] = arr3[j] + arr2[j]

    arr = new_arr


def opr6():
    global arr
    r_mid = len(arr) // 2
    c_mid = len(arr[0]) // 2
    # 중심 기준 4덩이로 나눈다
    arr1 = [r[:c_mid] for r in arr[:r_mid]]  # 왼쪽 위
    arr2 = [r[c_mid:] for r in arr[:r_mid]]  # 오른쪽 위
    arr3 = [r[c_mid:] for r in arr[r_mid:]]  # 오른쪽 아래
    arr4 = [r[:c_mid] for r in arr[r_mid:]]  # 왼쪽 아래

    new_arr = [[] for _ in range(len(arr))]

    # 윗 반
    for i in range(r_mid):
        new_arr[i] = arr2[i] + arr3[i]

    # 아랫 반
    for j in range(r_mid):
        new_arr[j + r_mid] = arr1[j] + arr4[j]

    arr = new_arr


# 입력
row, col, num = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)]
calls = list(map(int, input().split()))

# 연산
for cmd in calls:
    if cmd == 1:
        opr1()
    elif cmd == 2:
        opr2()
    elif cmd == 3:
        opr3()
    elif cmd == 4:
        opr4()
    elif cmd == 5:
        opr5()
    else:
        # cmd == 6
        opr6()

for rr in arr:
    print(*rr)
```
