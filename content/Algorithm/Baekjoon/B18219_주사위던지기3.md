### 코드

```python
def btk(n, l):
    if n == N:
        if sum(l) == sm:
            print(*l)
        return
    else:
        for i in range(1, 7):
            l.append(i)
            btk(n+1, l) # 부르기만 하고 아직 함수 안 끝
            l.pop()

T = int(input())
for tc in range (1, 1+T):
    lst = []
    N, sm = map(int, input().split())
    print(f"#{tc}")
    btk(0, lst)
```
