### 코드
```python
import heapq
lst = []
hq = []
while True:
    try:
        i = int(input())
        lst.append(i)
    except EOFError:
        break

N = lst.pop(0)

for x in range(N):
    if lst[x] == 0:
        if hq:
            ans = heapq.heappop(hq) # 괄호 안 리스트 이름
            print(ans)
        else:
            print(0)

    else:
        heapq.heappush(hq, lst[x])  # 모듈 이름.기능
```
