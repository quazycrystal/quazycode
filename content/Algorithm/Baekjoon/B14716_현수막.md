# bfs 구현
def bfs(sx, sy):
    global visit
    visit += 1
    v[sx][sy] = visit
    q = []
    q.append((sx, sy))
    # 구현
    while q:
        cur = q.pop(0)
        x, y = cur[0], cur[1]
        # 범위, 방문, 조건
        for d in range(8):
            nx = x+d_x[d]
            ny = y+d_y[d]
            if 0 <= nx < ix and 0 <= ny < iy:
                if v[nx][ny] == 0 and arr[nx][ny] == 1:
                    q.append((nx, ny))
                    v[nx][ny] = visit


ix, iy = map(int, input().split()) #행 열 헷갈리지 않게 주의, x가 열임
arr = [list(map(int, input().split())) for _ in range(ix)]
v = [[0] * iy for _ in range(ix)]
visit = 0
d_y = [-1, 1, 0, 0, -1, 1, -1, 1] # 상하 좌우, 우상부터 시계방향 대각선
d_x = [0, 0, -1, 1, -1, -1, 1, 1]

for i in range(ix):
    for j in range(iy):
        if v[i][j] == 0 and arr[i][j] == 1:
            bfs(i, j)

print(visit)
