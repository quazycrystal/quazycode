### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [인덱스] r, c는 1부터
- [문제 읽기] M은 폐업시키지 않을 치킨집 수. 치킨 거리는 집마다 가장 가까운 치킨집의 맨해튼 거리 합

### 코드 개선 방법
- [룩업 테이블] 집-치킨집 거리를 매핑 테이블로
- [구현] 어디서 어떤 정보를 얻을지 먼저 정하기

### 코드
```python
# 어디서 어떤 정보를 얻을건지? 왜 쓸데없이 길어지는지
def chicken(d, lst): #깊이 (치킨집 좌표 리스트 길이), 좌표
    # 치킨 깊이도 되돌려야?

    if len(lst) == remain: #리스트 길이가 남겨둔 길이랑 같다면
        # 치킨 거리 구하기
        c_dis = 0

        for home in homes:
            mn_distance = 2 * size + 1  # 아무리 멀어도 이거보단 작음
            # 집집별로 더해지니까 여기서 초기화
            for cls in lst:
                distance = abs(home[0] - cls[0]) + abs(home[1] - cls[1])
                mn_distance = min(mn_distance, distance)
            c_dis += mn_distance

        ans.append(c_dis) # 아까는 아예 함수 안 불렀었음
        return


    if d == len(chicks):
        return

    # 치킨집 좌표만큼 도니까, chicks[d]
    r, c = chicks[d][0], chicks[d][1]
    # 치킨집 추가 아님 안 추가
    chicken(d + 1, lst)

    lst.append((r, c))
    chicken(d + 1, lst)
    lst.pop()

size, remain = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(size)]
chicks = []
homes = []
for row in range(size):
    for col in range(size):
        if arr[row][col] == 2:
            chicks.append((row, col))
        if arr[row][col] == 1:
            homes.append((row, col))

print(min(ans))
```
