### 코드 개선 방법
- [유니온] find는 지나가며 부모를 다 바꾸기. union은 둘 다 find
- [종료 조건] union 횟수 = 섬 수 - 1이면 모두 연결
- [간선] 섬 bfs 그루핑 후 한 방향 직선으로 길이 2 이상만

### 코드
```python
# B17472 다리만들기 2

from collections import deque
from heapq import *

def is_range(r, c):
    return 0 <= r < row and 0 <= c < col

# 1. 섬끼리 모아 그루핑
def find_group(sr, sc, color):
    v[sr][sc] = color
    q = deque()
    q.append((sr, sc))
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if is_range(nr, nc) and v[nr][nc] == 0:
                if arr[nr][nc] == 1:
                    v[nr][nc] = color
                    q.append((nr, nc))

# 2. 간선을 다 찾기
def find_edge(sr, sc, now):
    pos = []
    # 근처 바다인지 판별
    for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
        cr, cc = sr, sc

        dist = 0
        while True:
            nxtr, nxtc = cr + dr, cc + dc
            if not is_range(nxtr, nxtc): break

            if v[nxtr][nxtc] == 0:
                dist += 1
                cr, cc = nxtr, nxtc
            else:
                break

        # while문 돌
        if is_range(nxtr, nxtc) and v[nxtr][nxtc] != now:
            if dist > 1:
                pos.append((dist, now, v[nxtr][nxtc]))

    return pos

# 3. 크루스칼 union find - 최소 신장 트리
def find(x):
    if p[x] != x:
        p[x] = find(p[x]) # 지나가면서 애들 다 바꾸기
    return p[x] # if문 밖 주의

def union(a, b):
    a_root = find(a)
    b_root = find(b)
    if a_root == b_root: return False

    p[a_root] = b_root # 둘 다 find!!!!
        # 하나는 p에 하나는 그냥
    return True

# 입력
row, col = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)]

# 풀이
v = [[0] * col for _ in range(row)]

# 전처리 - grouping
g = 1
for i in range(row):
    for j in range(col):
        if v[i][j] == 0 and arr[i][j] == 1:
            find_group(i, j, g)
            g += 1

# 간선 찾기
pos = []

for ii in range(row):
    for jj in range(col):
        if v[ii][jj] > 0:
            loc_pos = find_edge(ii, jj, v[ii][jj])
            if loc_pos:
                pos.extend(loc_pos)

# union / finds
ans = 0
p = [0] + [x for x in range(1, g)]
heapify(pos)
cnt = 0
while pos:
    d, g1, g2 = heappop(pos)
    if union(g1, g2):
        cnt += 1
        ans += d

if cnt == g-2:
    print(ans)
else:
    print(-1)
```
