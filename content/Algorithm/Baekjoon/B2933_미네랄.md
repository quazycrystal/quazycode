### 걸린 시간: 2시간 20분 (설계 18분, 구현 1시간 35분, 디버깅 27분) (실패 횟수: 2회 (WA))

### 실패 / 성공 이유
- [중력] 모양이 랜덤인 덩어리를 아래에서부터 쌓음. 위에서부터 내려 놓아야 함
- [순서] 내 전 위치를 비우고 놓을 곳을 찾아야 하는데 두 과정이 섞임

### 코드 개선 방법
- [BFS] 덩어리를 bfs로 좌표 리스트로 따고 바닥에 붙었는지 판별
- [중력] 내 칸부터 지우고, 모든 칸이 내려갈 수 있는 만큼 한 칸씩 내리기
- [방향] 막대는 왼→오, 오→왼 번갈아
- [습관] 체크리스트를 다시 적고 하나씩 보기

### 코드
```python
# B2933 미네랄

from collections import deque

# 함수
def is_range(r, c):
    return 0 <= r < row and 0 <= c < col

def shoot(nowtime, cmd):
    h = row - cmd # 0 인덱스인데 row여서 그냥 빼도 되는듯
    if nowtime % 2 == 0:
        for c in range(col):
            if arr[h][c] == "x":
                arr[h][c] = "."
                break
    else:
        for rc in range(col-1, -1, -1):
            if arr[h][rc] == "x":
                arr[h][rc] = "."
                break

# 2. 덩어리 판별 bfs
def bfs(sr, sc, v):
    v[sr][sc] = 1
    q = deque()
    q.append((sr, sc))
    shape = [(sr, sc)]
    # 그 도형이 바닥에 붙어있는지 아닌지?
    bottom = sr
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if is_range(nr, nc) and v[nr][nc] == 0:
                if arr[nr][nc] == "x":
                    q.append((nr, nc))
                    v[nr][nc] = 1
                    shape.append((nr, nc))

                    if nr > bottom:
                        bottom = nr

    return bottom, shape

# 3. 중력 함수:
def gravity(shape):
    # 일단 나부터 지우고 ㅠㅠ
    for sr, sc in shape:
        arr[sr][sc] = "."

    pr = -100
    # 위에서 놓아야 함, 미생물 안녕이 너무 머리에 깊이 남았나봐.
    for r in range(row):
        # 가능 범위에 그냥 다 넣어보기
        flag = 0
        for shaper, shapec in shape:
            nr = shaper + r
            if not is_range(nr, shapec):
                flag = 1
                break
            if arr[nr][shapec] == "x":
                flag = 1
                break

        if flag:
            pr = r - 1
            break

    for shr, shc in shape:
        arr[shr+pr][shc] = "x"


# 4. 덩어리 감싸서 한 방에 처리히는 함수
def drop():
    v = [[0] * col for _ in range(row)]
    picked = []
    for r in range(row):
        for c in range(col):
            if arr[r][c] == "x" and v[r][c] == 0:
                bottom, shape = bfs(r, c, v)
                if bottom != (row - 1):
                    picked.append(shape)

    if picked:
        for p in picked:
            gravity(p)


# 입력
row, col = map(int, input().split())
arr = [list(input()) for _ in range(row)]
hit_num = int(input())
commands = list(map(int, input().split()))

for i, cmd in enumerate(commands):
    # 1. 화살 쏘기
    shoot(i, cmd)
    # 2. 중력으로 떨어지기
    drop()

    #print("nowtime=========")

for rr in arr:
    print("".join(rr))
```
