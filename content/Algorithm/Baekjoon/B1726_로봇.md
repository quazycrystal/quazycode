### 실패 / 성공 이유
- [범위] n, m이 100
- [방향] 남/북을 아래/위로 헷갈림. 입력 방향 번호 3 ↔ 4
- [테케] 커스텀 테케가 가로로만 가서 좌/우 회전 수가 같아 안 잡힘

### 코드 개선 방법
- [상태 확장] (r, c, 방향)으로 방문 배열. 명령은 1~3칸 전진 / 좌·우 회전
- [이동] 1~3칸 전진은 한 칸씩. 막히면 그 너머로 못 감
- [회전] 회전도 한 명령. 온 방향까지 체크
- [인덱스] 입력은 1 인덱스
- [검증] 위아래로 움직이는 테케도 넣기

def is_range(r, c):
    return 0 <= r < row and 0 <= c < col


# 함수
direction = {1: (0, 1), 2: (0, -1), 3: (-1, 0), 4: (1, 0)}


def bfs(sr, sc):
    global temp
    v = [[-1] * col for _ in range(row)]
    v[sr][sc] = 0
    q = deque()
    q.append((sr, sc, []))
    while q:
        r, c, lst = q.popleft()
        if (r, c) == (er, ec):
            return lst
        for d in (1, 2, 3, 4):
            dr, dc = direction[d]
            nr, nc = r + dr, c + dc
            if is_range(nr, nc) and v[nr][nc] == -1:
                if arr[nr][nc] == 0:
                    v[nr][nc] = v[r][c] + 1
                    new_lst = lst[:]
                    new_lst.append(d)
                    q.append((nr, nc, new_lst))


# 입력
row, col = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)]

sr, sc, sd = map(int, input().split())
sr, sc = sr - 1, sc - 1  # 1 인덱스
er, ec, ed = map(int, input().split())
er, ec = er - 1, ec - 1

route = bfs(sr, sc)
route = [sd] + route + [ed]

side = {1: {3, 4}, 2: {3, 4}, 3: {1, 2}, 4: {1, 2}}
ans = 0
for idx in range(len(route)-1):
    cnt = 0 # 연속 등장 수

    # 이동 명령
    if route[idx] == route[idx + 1]:
        cnt += 1
        if cnt >= 3:
            ans += 1

    else: # 방향 전환
        cnt = 0
        ans += 1
        if route[idx+1] in side[route[idx]]:
            # 1 차이 난다면~
            ans += 1
        else:
            ans += 2

# B1726 로봇

from collections import deque

direction = {1: (0, 1), 2: (0, -1), 3: (1, 0), 4: (-1, 0)} # 3 <-> 4 ㅠㅠ 유유
right = {1: 3, 3: 2, 2: 4, 4: 1}  # 시계방향
left = {1: 4, 4: 2, 2: 3, 3: 1}  # 반시계방향


# 동서남북 - 우좌하상

# 함수
def is_range(r, c):
    return 0 <= r < row and 0 <= c < col

# 1. bfs로 다 가보자
def bfs(sr, sc, sd):
    v = [[[-1, 10**18, 10**18, 10**18, 10**18] for _ in range(col)] for _ in range(row)]
    v[sr][sc][sd] = 0
    q = deque()
    q.append((sr, sc, sd))
    while q:
        r, c, d = q.popleft()
        dr, dc = direction[d]
        for i in range(5):
            go = 1
            nr, nc = r, c
            nd = d
            if i == 0:
                nr, nc = r + dr, c + dc

            elif i == 1:
                for dist in range(1, 3):
                    nr, nc = r + (dist * dr), c + (dist * dc)
                    if is_range(nr, nc) and arr[nr][nc] == 1:
                        go = 0
                        break

            elif i == 2:
                for dist in range(1, 4):
                    nr, nc = r + (dist * dr), c + (dist * dc)
                    if is_range(nr, nc) and arr[nr][nc] == 1:
                        go = 0
                        break

            # 2. 회전
            elif i == 3:
                nd = right[d]
            elif i == 4:
                nd = left[d]

            # 온 방향까정!!!
            if is_range(nr, nc) and v[nr][nc][nd] > v[r][c][d] + 1:
                if arr[nr][nc] == 0 and go:
                    q.append((nr, nc, nd))
                    v[nr][nc][nd] = v[r][c][d] + 1

    return v[er][ec][ed]

# 입력
row, col = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)]

sr, sc, sd = map(int, input().split())
sr, sc = sr - 1, sc - 1
er, ec, ed = map(int, input().split())
er, ec = er - 1, ec - 1

# 풀이
ans = bfs(sr, sc, sd)

print(ans)
