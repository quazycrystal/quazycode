### 걸린 시간: 구상 27분, 로직 풀이 1시간 50분 (실패 횟수: 시간초과)

### 실패 / 성공 이유
- [시간] 순열 8! 안에서 이닝 수만큼 while이 또 돌아서 덱 rotate 오버헤드가 누적
- [초기화] 이닝마다 아웃과 베이스 초기화

### 코드 개선 방법
- [속도] 베이스는 덱 대신 int(또는 1/0 변수), rotate는 인덱스로 직접 구현
- [속도] 이닝 정보는 2차원 리스트로 미리 파싱. 4번 타자 자리에 1번 선수 고정 후 나머지 8명 순열

### 코드
```python
# from collections import deque
from itertools import *
import sys

# 입력
N = int(input()) # 이닝 횟수
innings = [list(map(int, input().split())) for _ in range(N)]

# 함수 -  이닝 돌리는 거
def run(num):
    global b0, b1, b2, b3
    local_score = 0
    # print(f"base", base)
    if num != 0:  # 아웃 아니면 다 진루
        b0 += 1
        for _ in range(num):
            # base.rotate(-1) # Overhead
            b0, b1, b2, b3 = b3, b0, b1, b2 # rotate 직접 구현
            local_score += b0 # 들어올 때마다 점수 세어줘야
            b0 = 0

    return local_score

score = 0 # 경우의 수
mx_score = 0
pos_batter = [x for x in range(1, 9)]

for combi in permutations(pos_batter, 8):
    batter = list(combi)
    batter.insert(3, 0)

    score = 0 # 경우의 수 바뀔 때 score도 초기화

    idx = 0  # deque rotate 구현용, 이닝 지나고도 유지

    for times in range(N): # 이닝 N번 돌 때 마다 이닝 정보에 접근해서 점수내요
        inning = innings[times] # 한 이닝의 정보 불러온다

        # 이닝마다 아웃 초기화, 베이스도 초기화!!!
        out = 0 # 아웃 세야
        b0, b1, b2, b3 = 0, 0, 0, 0 # 심지어 리스트도 안 됨

        while out < 3: # 한 주자
            if idx == 9:
                idx = 0

            person = batter[idx] # 지금 어떤 주자?
            number = inning[person] # 그 주자의 결과?

            if number == 0: # 결과 out이면 센다
                out += 1

            # 아웃 3번 되면 이닝 종료
            if out == 3:
                break

            else: # 아웃이 아닌 경우에만 진행
                l_score = run(number) # 그 주자로 인해 나온 점수?
                score += l_score # 한 경우의 수에다가 점수 다 더해
                # batter.rotate(-1) # 이닝 내부에서 계속 돌 때
                idx += 1

        idx += 1

    # 이닝 다 돌면 하나의 경우의 수가 완성됨
    mx_score = max(score, mx_score)

print(mx_score)

# 정답
import sys
from itertools import permutations

input_data = sys.stdin.read().split()

N = int(input_data[0])
# 이닝 정보를 2차원 리스트로 미리 파싱 (접근 속도 향상)
innings = []
ptr = 1
for _ in range(N):
    innings.append(list(map(int, input_data[ptr:ptr + 9])))
    ptr += 9

max_score = 0
# 0번(1번 타자)을 제외한 나머지 8명의 순열
for p in permutations(range(1, 9)):
    # 타순 구성: 4번 타자 자리에 0번 선수 고정
    order = list(p[:3]) + [0] + list(p[3:])

    score = 0
    idx = 0  # 타순 인덱스

    for inning in innings:
        out = 0
        b1 = b2 = b3 = 0  # 1루, 2루, 3루 상태 (있으면 1, 없으면 0)

        while out < 3:
            res = inning[order[idx]]

            if res == 0:
                out += 1
            elif res == 1:  # 안타
                score += b3
                b1, b2, b3 = 1, b1, b2
            elif res == 2:  # 2루타
                score += (b2 + b3)
                b1, b2, b3 = 0, 1, b1
            elif res == 3:  # 3루타
                score += (b1 + b2 + b3)
                b1, b2, b3 = 0, 0, 1
            elif res == 4:  # 홈런
                score += (b1 + b2 + b3 + 1)
                b1 = b2 = b3 = 0

            idx = (idx + 1) % 9

    if score > max_score:
        max_score = score

print(max_score)
```
