### 걸린 시간: 2시간 15분 (7:20 - 9:35), 디버깅 55분 (실패 횟수: -)

### 실패 / 성공 이유
- [방향] 벽에 튕길 때 덱은 안 돌리고 좌표만 -로 처리. 방향 자체가 바뀐 걸 반영 안 해서 다음 이동부터 어긋남
- [상태 저장] dice를 함수 안에서 만들면 돌 때마다 원상복구됨. 전역으로 유지해야
- [디버깅] 테케 1~3은 맞고 4만 틀리는 상황이 50분 내내 계속

### 코드 개선 방법
- [상태] 주사위를 하나의 상태로: 인덱스는 고정, 면 값만 바뀐다고 생각
- [방향] 튕기면 방향도 바꾸고 다음 좌표도 바꾸기
- [문제 읽기] '연속 이동'은 같은 수 bfs
- [디버깅] 변수 이름과 값을 같이 출력해 한눈에 보이게
- [구상] 한 번에 로직을 잘 짜야 디버깅이 편함

### 코드

from collections import deque
direction = deque([(0, 0, 1), (1, 1, 0),
           (2, 0, -1), (3, -1, 0)]) # 시계 방향, 우하좌상 - 0, 1, 2, 3

dice = [x for x in range(1, 7)] # 각 면에 숫자 있다요

def dice_roll(sr, sc):
    # 주사위 돌리기만 한다
    global dice # dice 여기서 자꾸 초기화 하면 안됨
    a, b, c, d, e, f = dice[0], dice[1], dice[2], dice[3], dice[4], dice[5]
    # 중요한 점, 돌릴 때 인덱스는 그대로고 면만 바뀐다고 생각
    dirr, dr, dc = direction[0]
    nr, nc = sr + dr, sc + dc

    if 0 <= nr < row and 0 <= nc < col:
        pass
    else:
        if dirr >= 2:
            direction.rotate(2)
        else:
            direction.rotate(-2)

        # 튕기고 나서는 방향도 바뀌고, 다음 좌표도 바뀌고
        dirr, dr, dc = direction[0]
        nr, nc = sr + dr, sc + dc

    if dirr == 0:
        dice = [d, b, a, f, e, c]
        # 4 2 1 6 5 3
    elif dirr == 1:
        dice = [b, f, c, d, a, e]
        # 2 6 3 4 1 5
    elif dirr == 2:
        dice = [c, b, f, a, e, d]
        # 3 2 6 1 5 4
    else: # up
        dice = [e, a, c, d, f, b]
        # 5 1 3 4 6 2
    return nr, nc, dice[5] # 밑바탕만 리턴 고고

def change_dir(sr, sc, num):
    # 옮겨간 자리와 밑바탕 숫자를 받아와요
    A = num # 주사위 밑바탕 값
    B = arr[sr][sc] # 주사위가 있는 칸의 값

    if A > B:
        direction.rotate(-1) # 시계방향 회전
    elif A < B:
        direction.rotate(1) # 반시계 방향 회전
    else:
        pass

    return B

def continue_move(sr, sc, value):
    v = [[0] * col for _ in range(row)]
    cnt = 1 # 현재 칸 포함
    v[sr][sc] = 1
    q = deque()
    q.append((sr, sc))
    while q:
        r, c = q.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < row and 0 <= nc < col:
                if v[nr][nc] == 0 and arr[nr][nc] == value:
                    q.append((nr, nc))
                    v[nr][nc] = 1
                    cnt += 1 # ok되는 칸 마다 cnt 추가

    score = value * cnt
    return score

# 입력
row, col, times = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)]

r, c = 0, 0
# 문제 풀이:
ans = 0
for time in range(times): # 이번에는 while 안 써도 됨
    nr, nc, bottom = dice_roll(r, c)
    arr_val = change_dir(nr, nc, bottom)
    loc_score = continue_move(nr, nc, arr_val)
    ans += loc_score
    r, c = nr, nc # 다음에 넣을 좌표 갱신

print(ans)
