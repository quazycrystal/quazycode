### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [시간 복잡도] 200*200*31이지만 한 번만 도니 충분

### 코드 개선 방법
- [상태 확장] (r, c, 말 이동 사용 횟수)로 3차원 방문
- [가지치기] 같은 칸이라도 말 횟수가 적을 때만 추가
- [입력] W, H 순서 주의

### 코드
```python
# 같은 건 볼 필요 없고, 말 갯수 작을 때만 추가!
from collections import deque

horse = ((-2, -1), (-1, -2), (1, -2), (2, -1),
         (2, 1), (1, 2), (-1, 2), (-2, 1))
# 함수

def is_range(r, c):
    return 0 <= r < row and 0 <= c < col

def bfs3d(sr, sc, horse_num):
    v = [[10**18] * col for _ in range(row)]
    v[sr][sc] = 0
    q = deque()
    q.append((sr, sc, 0, 0))
    ans = -1
    while q:
        r, c, h_move, m_move = q.popleft()
        if (r, c) == (er, ec):
            ans = h_move + m_move
            return ans

        # 1) 원숭이로 간다
        for mr, mc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + mr, c + mc
            if is_range(nr, nc) and v[nr][nc] > h_move:
                # 뭔가 다익스트라 같기도?
                if arr[nr][nc] == 0: # 장애물 없어야만 갈 수 있다
                    v[nr][nc] = h_move
                    q.append((nr, nc, h_move, m_move + 1))

        # 2) 말로 간다
        if h_move < horse_num: # 각 경우를 봐야하니까
            for hr, hc in horse:
                nhr, nhc = r + hr, c + hc
                if is_range(nhr, nhc) and v[nhr][nhc] > h_move+1:
                    if arr[nhr][nhc] == 0:
                        v[nhr][nhc] = h_move + 1
                        q.append((nhr, nhc, h_move + 1, m_move))
    return ans

# 입력
horse_num = int(input())
col, row = map(int, input().split()) # W, H 헷갈림 주의

sr, sc = 0, 0
er, ec = row-1, col-1

arr = [list(map(int, input().split())) for _ in range(row)]

ans = bfs3d(sr, sc, horse_num)
print(ans)
```
