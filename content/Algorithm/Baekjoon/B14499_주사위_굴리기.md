### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [문제 읽기] 처음 주사위는 모든 면이 0. arr는 북쪽부터, 북이 위
- [로직] 주사위 맨 위가 항상 dice[1]인 건 아님. 맨 아래와 마주보는 면
- [로직] 위/아래로 굴리면 옆은 그대로, 양옆으로 굴리면 바뀜. 쌍으로 움직임
- [범위] 밖이면 굴리지도 위치 갱신도 안 함

### 코드 개선 방법
- [자료구조] 주사위를 하나의 상태로, 방향은 dict로 바로 접근
- [변수] c가 col인지 주사위인지 변수 이름 주의

`W3/B14499_주사위 굴리기.py`

```python
dice = ["x"] + [0]*6
# 자료형은 큐 / stack
right = [3, 1, 4, 6] # 동
left = [4, 1, 3, 6] # 서
up = [2, 1, 5, 6] # 북
down = [5, 1, 2, 6] # 남
flag1 = [3, 5, 4, 2]
flag2 = [4, 5, 3, 2]

ops = {1: 6, 5: 2, 4: 3, 3: 4, 2: 5, 6: 1}

com = {1: (0, 1), 2: (0, -1), 3: (-1, 0), 4: (1, 0)} # 동: 오른쪽, 서: 왼쪽, 북: 위쪽, 남: 아래쪽
direction = {1: right, 2: left, 3: up, 4: down, 5: flag1, 6: flag2}

# 세로, 가로, 좌표, 명령 갯수
row, col, srt_r, srt_c, com_num = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)]
commands = list(map(int, input().split()))
srt = (srt_r, srt_c)
flag = 0

def c_changer(comm):
    # 위, 아래는 옆 숫자가 안 바뀌지만, 양 옆으로 돌 때는 바뀜...
    global flag # 누적 되어야 한다
    if comm == 1: # 오른쪽
        flag += 1
    elif comm == 2: # 왼쪽
        flag -= 1

    else:
        how = flag % 4 # 상 하 커맨드 어떻게 바꿀 지
        if how == 1:
            if comm == 3:
                comm = 1
            else:
                comm = 2
        elif how == 2:
            if comm == 3:
                comm = 4
            else:
                comm = 3

        elif how == 3:
            if comm == 3:
                comm = 2
            else:
                comm = 1
    return(comm)


def move(comm, m, s):
    global dice
    r, c = s[0], s[1]
    if comm in com: # 딕셔너리 안에 명령 있으면
        cur = com[comm]
        dr, dc = cur[0], cur[1]
        nr, nc = r+dr, c+dc
        if 0 <= nr < row and 0 <= nc < col:
            r, c = nr, nc # 조건 만족 시 좌표 갱신
            move_lst = direction[m]
            if m % 2 == 0: # 왼, 아래 (key = 2, 4)
                pair_lst = direction[m - 1]
                # 움직이는 방향 리스트는 맨 앞이 맨 뒤로
                bottom = move_lst.pop(0)
                move_lst.append(bottom)

                # 쌍으로 움직이는 리스트는 맨 뒤가 맨 앞으로
                p_bottom = pair_lst.pop()
                pair_lst.insert(0, p_bottom)

            else: # 오, 위 (key = 1, 3)
                pair_lst = direction[m + 1]
                bottom = move_lst.pop(0)
                move_lst.append(bottom)

                p_bottom = pair_lst.pop()
                pair_lst.insert(0, p_bottom)


            up = ops[bottom]

            if arr[r][c] == 0:
                arr[r][c] = dice[bottom] # 다른 변수에 정의하면 그 변수만 바뀜

            else:
                dice[bottom] = arr[r][c]
                arr[r][c] = 0

            print(dice[up])
    return ((r, c))

for command in commands:
    mm = c_changer(command)
    srt = move(command, mm, srt) # 좌표 주의, 위에서 시작 좌표 선언 하고 갱신해야
```

`W6/B14499_주사위 굴리기.py`

```python
dice = [0] * 6

# 변수명도 좀 고정을 할까
direction = {1: (0, 1), 2: (0, -1), 3: (-1, 0), 4: (1, 0)}


def roll_dice(rr, cc, cmd):
    global dice
    a, b, c, d, e, f = dice[0], dice[1], dice[2], dice[3], dice[4], dice[5]

    # 위치 이동 가능 한 지
    dr, dc = direction[cmd]
    nr, nc = rr + dr, cc + dc  # 변수 이름 주의! c가 col 인지 밑에 주사위인지
    if 0 <= nr < row and 0 <= nc < col:
        # 가능하면 주사위 굴리기
        if cmd == 1:
            dice = [d, b, a, f, e, c]  # 4 2 1 6 5 3
        elif cmd == 2:
            dice = [c, b, f, a, e, d]  # 3 2 6 1 5 4
        elif cmd == 3:
            dice = [e, a, c, d, f, b]  # 5 1 3 4 6 2
        else:  # cmd == 4
            dice = [b, f, c, d, a, e]  # 2 6 3 4 1 5

        rr, cc = nr, nc

    # 범위 밖이면 주사위도 안 구르고 위치 갱신도 안 됨
    return rr, cc  # 이동 좌표 반환


# 입력
row, col, s_r, s_c, cmd_num = map(int, input().split())
arr = [list(map(int, input().split())) for _ in range(row)]
cmd_lst = list(map(int, input().split()))

# ans = []

for command in cmd_lst:
    nr, nc = roll_dice(s_r, s_c, command)
    # 정답은 위에 어떤 수가 오느냐
    if (nr, nc) != (s_r, s_c):  # 이동을 한 경우에만 출력
        print(dice[0])

        # 주사위 바닥의 수와 도착한 장소의 수 비교
        if arr[nr][nc] == 0:
            arr[nr][nc] = dice[5]
        else:
            dice[5] = arr[nr][nc]
            arr[nr][nc] = 0

        s_r, s_c = nr, nc  # 변수 갱신

```
