### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [로직] 후보가 하나로 좁혀진 칸만 채움. 후보가 여럿인 칸은 끝까지 못 채움

### 코드 개선 방법
- [백트래킹] 후보를 일단 넣어보고 안 되면 되돌리기
- [중복 방지] 가로·세로·박스 체크를 한 번에
- [인덱스] 박스 비교는 시작점(0, 3, 6)부터, 가로 체크는 열 좌표가 겹치는지 보기

### 코드
```python
# 백트래킹!!!

# 가로 체킹
def garo(r, c):
    check = {1, 2, 3, 4, 5, 6, 7, 8, 9} # 빈 칸에 들어갈 수 있는 수
    for i in range(9):
        if i != c: # 행이 아니라 열 좌표 겹치는 거 봐야,,,
            now = arr[r][i] # 현재 체킹하고 있는 칸
            if now != 0:
                check.discard(now)
    #if len(check) == 1:
    arr[r][c] = check.pop()

# 세로 체킹
def saero(r, c):
    check = {1, 2, 3, 4, 5, 6, 7, 8, 9} # 빈 칸에 들어갈 수 있는 수
    for i in range(9):
        rr = arr[i]
        if i != r:
            now = rr[c] # 현재 체킹하고 있는 칸
            if now != 0:
                check.discard(now)
    # if len(check) == 1:
    arr[r][c] = check.pop()

# 네모 칸 체킹
def box(r, c):
    # srt_r, srt_c = 0, 0 # 순회 시작 할 시작 점
    check = {1, 2, 3, 4, 5, 6, 7, 8, 9}  # 빈 칸에 들어갈 수 있는 수

    # 순회 시작 할 시작 점
    if 0 <= r < 3:
        srt_r = 0
    elif 3 <= r < 6:
        srt_r = 3
    else:
        srt_r = 6 # 행

    if 0 <= c < 3:
        srt_c = 0
    elif 3 <= c < 6:
        srt_c = 3
    else:
        srt_c = 6 # 열

    for box_r in range(srt_r, srt_r+3):
        for box_c in range(srt_c, srt_c + 3):
            if (box_r, box_c) != (r, c):
                now = arr[box_r][box_c] # 현재 체킹하고 있는 칸
                if now != 0:
                    check.discard(now)

    #if len(check) == 1:
    arr[r][c] = check.pop()

# 입력
arr = [list(map(int, input().split())) for _ in range(9)]

# 문제 풀이
for row in range(9):
    for col in range(9):
        if arr[row][col] == 0:
            garo(row, col)
            saero(row, col)
            box(row, col)

# 답 출력
for rr in arr:
```
