# 힙: 특정 값 뽑고 버리거나, 힙 새로 만들기
from heapq import *
from collections import defaultdict

ppl, chobab = map(int, input().split())
# 각 손님의 주문 목록 안에서 초밥 종류는 다 다름
round_chobab = defaultdict(list) # 들어오는 것에 대해서만 만든다
eat = [0] * (ppl+1) # 각 손님이 몇 개씩 먹었는지

for i in range(1, ppl+1):
    a, *temp = list(map(int, input().split()))
    for cho in temp:
        heappush(round_chobab[cho], i) # 초밥 번호, 사람 번호 넣기

#print(round_chobab)
made_chobab = list(map(int, input().split()))
made_chobab.sort()

for made_cho in made_chobab:
    if round_chobab[made_cho]:
        eat_person = heappop(round_chobab[made_cho])
        eat[eat_person] += 1

print(*eat[1:])
