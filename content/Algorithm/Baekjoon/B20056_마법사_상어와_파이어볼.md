### 걸린 시간: 2트 3시간 14분 (구상 16분, 풀이 55분, 디버깅 못 찾음) / 재풀이 1시간 13분 (구상 11분, 풀이 36분, 디버깅 26분) (실패 횟수: 2트 디버깅 중 도움 받아 마무리)

### 실패 / 성공 이유
- [자료구조] 자료형이 겹겹이라 추적 놓침: fire_balls[fb]인데 fire_balls 길이를 구함
- [원본 수정] for문 도는 중 원본 값을 바꿈. 질량 0일 때 원본을 안 바꾸면 이전 데이터가 남음
- [쓰레기 값] mass 0을 바로 안 치워서 다음 턴 이동에 영향
- [변수] sd를 ss로 적음
- [범위] 무한 격자(%), 질량·속도 1000

### 코드 개선 방법
- [자료구조] dict보다 3차원 배열이 단순. 같은 key를 리스트로 다시 연산할 땐 arr가 나음
- [원본 수정] 새 값은 다른 자료구조에 모았다가 전체를 새로 갈아주기
- [체크리스트] 지금 변수가 old인가 new인가 / 빈 칸에서 루프가 안전한가 / 바꾼 값이 다음 계산에 영향을 주는가
- [분기] 플래그 대신 다 넣고 길이로 판단
- [문법] ⌊ ⌋는 버림 기호

```python
direction = {0: (-1, 0),
             1: (-1, 1),
             2: (0, 1),
             3: (1, 1),
             4: (1, 0),
             5: (1, -1),
             6: (0, -1),
             7: (-1, -1)}

def move():
    new_arr = [[[] for _ in range(size)] for _ in range(size)]
    # mass, speed 다 1000!!!
    for r in range(size):
        for c in range(size):
            if arr[r][c]:
                for info in arr[r][c]:
                    m, s, d = info
                    dr, dc = direction[d]
                    nr, nc = (r + (dr * s)) % size, (c + (dc * s)) % size
                    new_arr[nr][nc].append(info)

    return new_arr

def check(num):
    if num % 2 == 0:
        return 0
    else:
        return 1

def mix():
    for r in range(size):
        for c in range(size):
            if len(arr[r][c]) > 1:
                num = len(arr[r][c])
                sm, ss, sd = 0, 0, 0
                for info in arr[r][c]:
                    m, s, d = info
                    sm += m
                    ss += s
                    sd += check(d)

                # 홀 짝 방향 판별
                d_lst = []
                if sd == num or sd == 0: # 변수 주의ㅜㅠ 디버깅 포인트 ss 아니고 sd...
                    d_lst = [0, 2, 4, 6] # 같으면 상하좌우
                else:
                    d_lst = [1, 3, 5, 7] # 다르면 대각선

                # arr[r][c] 갱신
                arr[r][c] = []
                nm, ns = sm // 5, ss // num

                if nm > 0: # 질량 0이면 소멸
                    for nd in d_lst:
                        arr[r][c].append([nm, ns, nd])

# 3. 정답 구하기
def get_ans():
    ans = 0
    for r in range(size):
        for c in range(size):
            if arr[r][c]:
                for mass, _, _ in arr[r][c]:
                    ans += mass

    return ans

# 입력
size, a_num, times = map(int, input().split())
arr = [[[] for _ in range(size)] for _ in range(size)]

# 원자 정보 (1 인덱스!)
for _ in range(a_num):
    r, c, m, s, d = map(int, input().split())
    r, c = r - 1, c - 1
    arr[r][c].append([m, s, d])

# 풀이
for nowtime in range(times):
    new_arr = move()
    arr = new_arr
    mix()

ans = get_ans()
print(ans)

# ⌊0.11⌋ 버림 기호임... 몰랐다 ㄴㅇㄱ - 진짜 알아두자,,,
direction = {0: (-1, 0), 1: (-1, 1), 2: (0, 1), 3: (1, 1),
             4: (1, 0), 5: (1, -1), 6: (0, -1), 7: (-1, -1)}

# 입력
size, ball_num, move_num = map(int, input().split())
# 모든 파이어 볼 여기에서 관리
fire_balls = {}
for _ in range(ball_num):
    r, c, m, s, d = map(int, input().split())
    fire_balls[(r - 1, c - 1)] = [(m, s, d)]  # 1, 1부터 준다요!
    # 질량, 거리, 방향


# 함수 : 이동 함수, 합치고 나누는 함수
def move():
    global fire_balls

    temp = dict()
    for fb in fire_balls:
        # fb는 key이므로 r, c, 나머지는 질량, 거리, 방향
        r, c = fb

        for info in fire_balls[fb]:  # 한 좌표에 여럿 있을 수도 있잖아
            m, s, d = info

            # 원래 위치 지우고 이동한 위치로 키 다시 만들어준다
            dr, dc = direction[d]
            nr, nc = (r + (dr * s)) % size, (c + (dc * s)) % size
            # 좌표 도넛 행성처럼 돌아감
            if (nr, nc) not in temp:  # 이동 완료 좌표 기준 없으면
                temp[(nr, nc)] = [(m, s, d)]  # 만들고
            else:
                temp[(nr, nc)].append((m, s, d))  # 같은 좌표면 추가

    fire_balls = temp


def combine_split():
    global fire_balls, ans
    ans = 0  # 초기화
    new_fire_balls = dict()  # for문 안에서 원본 바꾸면서 가서 오류가 났던 거였음...
    for fb in fire_balls:
        if len(fire_balls[fb]) > 1:
            # 한 좌표 기준 들어있는 정보가 2개 이상
            total_m = 0  # 모두 합쳐진 질량
            total_s = 0  # 모두 합쳐진 속도
            total_num = len(fire_balls[fb])  # 몇 개가 합쳐지는지

            dirr = []

            temp = []  # fire_balls[fb] 갈아 끼울 것

            for info in fire_balls[fb]:
                m, s, d = info
                if m > 0:  # 질량 0이면 소멸
                    total_m += m
                    total_s += s
                    dirr.append(d % 2)  # 1이면 홀, 0이면 짝

            # 새로 나누기 - 새로 나누어 넣을 때도 0인 애들은 넣지 말자
            nm = total_m // 5
            if nm > 0:  # 0인 애들 새로 나눌 때도 빼야

                ns = total_s // total_num

                # 방향만 루프에서 더해서 temp로
                if sum(dirr) == 0 or sum(dirr) == total_num:  # 모든 방향 홀, 짝이 같음
                    for direct in (0, 2, 4, 6):
                        nd = direct  # 새 방향 지정
                        temp.append((nm, ns, nd))
                        ans += nm  # 질량 미리 더해준다요

                else:
                    for direct in (1, 3, 5, 7):
                        nd = direct  # 새 방향 지정
                        temp.append((nm, ns, nd))
                        ans += nm  # 질량 미리 더해준다요

                new_fire_balls[fb] = temp

        else:  # 길이 1이다
            new_fire_balls[fb] = fire_balls[fb]
            ans += fire_balls[fb][0][0]  # 맨 앞이 mass

    fire_balls = new_fire_balls


# 풀이
ans = 0
for time in range(move_num):
    move()
    combine_split()

print(ans)
```
