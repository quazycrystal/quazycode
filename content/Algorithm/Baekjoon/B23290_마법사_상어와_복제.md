### 걸린 시간: 4시간 (구상 18분, 구현 1시간 27분, 디버깅 1시간 36분, 2차 디버깅 49분) (실패 횟수: 테케 5, 6 틀림)

### 실패 / 성공 이유
- [방문 처리] 상어 3칸 이동에서 직전 방향의 역방향만 막음. 2칸 갔다 돌아오는 건 허용해야 하는데 막아버림
- [순서] 냄새 사라지는 코드 위치. 상어 이동 바로 뒤면 냄새를 3으로
- [테케] 오픈 테케가 다 맞아 직접 만들기 어려운 경우라 1시간 넘게 못 찾음

### 코드 개선 방법
- [방문 처리] 예외처리 하나하나 말고 경로 내 방문 set을 만들어 이미 먹은 칸만 안 세기
- [자료구조] 칸마다 방향별 물고기 수 [0]*8로 관리하면 물고기 수가 많아도 8*N*N
- [완전탐색] 4*4라 상어 이동 4^3 product로 완탐, 먹는 수가 같으면 사전순 첫 번째

### 코드
```python
v_shark = [[False] * 4 for _ in range(4)]  # 한 이동 경로 내 중복 방문 체크 이터 툴 안에서
visited = set(), visited.add((x, y))


from itertools import *

# 함수
direction = {1: (0, -1),
             2: (-1, -1),
             3: (-1, 0),
             4: (-1, 1),
             5: (0, 1),
             6: (1, 1),
             7: (1, 0),
             0: (1, -1)}

# 상하 좌우 순 탐색
shark_dirr = {0: (-1, 0), 1: (0, -1), 2: (1, 0), 3: (0, 1)}
reversed = {0: 2, 1: 3, 2: 0, 3: 1}


# 범위 함수
def is_range(r, c):
    return 0 <= r < 4 and 0 <= c < 4


# 물고기 움직 함수
def fish_move(r, c, sharkr, sharkc):
    for idx, d in arr[r][c]:
        move = 0
        for nxt in range(8):
            now_d = (d - nxt) % 8
            dr, dc = direction[now_d]
            nr, nc = r + dr, c + dc
            if is_range(nr, nc) and smell[nr][nc] == 0:

                # 칸 존재, 냄새 없음
                if (nr, nc) != (sharkr, sharkc):
                    # 상어칸도 아니면
                    new_arr[nr][nc].append((idx, now_d))
                    move = 1
                    break

        if move == 0:
            # 움직일 수 없었음
            new_arr[r][c].append((idx, d))


# 상어 움직 함수
def pick_shark(sharkr, sharkc):
    pos_move = []
    # 물고기 제일 많이 먹는 쪽, 그것도 여럿이면 제일 빨리 나온 쪽
    mx_fish = 0
    for combi in product(range(4), repeat = 3):
        eaten = 0
        dr, dc = shark_dirr[combi[0]]  # 첫 칸
        nr, nc = sharkr + dr, sharkc + dc
        if is_range(nr, nc):
            eaten += len(new_arr[nr][nc])

            dr2, dc2 = shark_dirr[combi[1]]  # 두 번째 칸
            nr2, nc2 = nr + dr2, nc + dc2

            if is_range(nr2, nc2):
                eaten += len(new_arr[nr2][nc2])

                dr3, dc3 = shark_dirr[combi[2]]  # 세 번째 칸
                nr3, nc3 = nr2 + dr3, nc2 + dc3

                if is_range(nr3, nc3):
                    if combi[2] != reversed[combi[1]]:
                        eaten += len(new_arr[nr3][nc3])

                    # 3번째 칸 까지 도달!
                    if eaten > mx_fish:
                        pos_move = []
                        mx_fish = eaten

                    pos_move.append(combi)

    return pos_move[0]


# 실제로 받아서 new_arr 갱신, 좌표 반환
def move_shark(sharkr, sharkc, combi):
    print(combi)
    dr, dc = shark_dirr[combi[0]]  # 첫 칸
    nr, nc = sharkr + dr, sharkc + dc

    if len(new_arr[nr][nc]) > 0:
        smell[nr][nc] = 3
        new_arr[nr][nc] = []

    dr2, dc2 = shark_dirr[combi[1]]  # 두 번째 칸
    nr2, nc2 = nr + dr2, nc + dc2

    if len(new_arr[nr2][nc2]) > 0:
        smell[nr2][nc2] = 3
        new_arr[nr2][nc2] = []

    dr3, dc3 = shark_dirr[combi[2]]  # 세 번째 칸
    nr3, nc3 = nr2 + dr3, nc2 + dc3

    if len(new_arr[nr3][nc3]) > 0:
        smell[nr3][nc3] = 3
        new_arr[nr3][nc3] = []

    return nr3, nc3


# 입력
fish_num, time = map(int, input().split())
arr = [[[] for _ in range(4)] for _ in range(4)]
smell = [[0] * 4 for _ in range(4)]

# 1 인덱스!
for idx in range(fish_num):
    r, c, d = map(int, input().split())
    r, c = r - 1, c - 1
    arr[r][c].append((idx, d))

sharkr, sharkc = map(int, input().split())
sharkr, sharkc = sharkr - 1, sharkc - 1


for idx in range(time):
    new_arr = [[[] for _ in range(4)] for _ in range(4)]
    for row in range(4):
        for col in range(4):
            if arr[row][col]:
                fish_move(row, col, sharkr, sharkc)


    # 물고기 방향 고름
    nxt_shark_move = pick_shark(sharkr, sharkc)

    # 이동 "후"에 냄새가 사라져야 함


    # 그 방향대로 이행 - 물고기 치우고 좌표 옮기고
    new_sharkr, new_sharkc = move_shark(sharkr, sharkc, nxt_shark_move)
    sharkr, sharkc = new_sharkr, new_sharkc # 갱신
    # 냄새 격자 renew
    for i in range(4):
        for j in range(4):
            if smell[i][j] > 0:
                smell[i][j] -= 1


    # 복제 마법 완료, arr 갱신
    for newr in range(4):
        for newc in range(4):
            arr[newr][newc].extend(new_arr[newr][newc])

# 정답
ans = 0

for finalr in range(4):
    for finalc in range(4):
        ans += len(arr[finalr][finalc])

print(ans)
```
