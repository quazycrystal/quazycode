### 실패 / 성공 이유
- [조건] a, b 중 어느 쪽이 큰지 안 나옴
- [범위] 징검다리 범위 밖은 갈 수 없음. idx 0이면 곱해도 0

### 코드
```python
def bfs(s, e): # 몇 번째 이므로 idx
    global lst
    global v
    v = {s:0}
    q = []
    q.append(s)
    while q:
        cur = q.pop(0)
        dol = lst[cur]
        # 답은 여기서
        if cur == e:
            #ans = v[cur]
            return v[cur]

        if cur != e:
            # if cur not in v: # 현재 말고 새로운 거 볼 때
            if 0 < cur <= num: # 징검다리 범위 벗어날 수 없으니
                for i in range(1, (num//dol)+1): # idx 0이면 곱해도 0
                    nxt1 = cur + (i * dol)
                    nxt2 = cur - (i * dol)
                    if nxt1 not in v:
                        if 0 < nxt1 <= num:
                            v[nxt1] = v[cur] + 1
                            q.append(nxt1)
                    if nxt2 not in v:
                        if 0 < nxt2 <= num:
                            v[nxt2] = v[cur] + 1
                            q.append(nxt2)
    return -1


# 입력
num = int(input()) # 징검다리 갯수
lst = [0] + list(map(int, input().split())) # 징검다리 idx, 쓰인 숫자
srt, goal = map(int, input().split())
# ans = 0
v = {}


print(bfs(srt, goal))
```
