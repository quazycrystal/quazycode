### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [분기] 선이 그어진 경우에만 집중하고 아닌 경우를 못 잡음
- [엣지 케이스] 갔다가 돌아온 경우 등 예외 처리가 어려움
- [범위] 좌표가 튀어나가면 현 명령 무시

### 코드 개선 방법
- [분기] 선이 있는 경우 / 없는 경우부터 나누기
- [이동] 다음 위치가 밖이면 원래 위치 반환, 움직임 무시
- [표시] 현 칸과 다음 칸에 이미 무엇이 그려져 있는지 보고 표시. 다른 방향 선이 있으면 교차로

`W4/B1730_판화.py`

```python
d = {"U":(-1, 0), "D":(1, 0), "L":(0, -1), "R":(0, 1)}
g = [("U", "D"), ("R", "L")]

def draw(jwa, now, pre):
    # now는 현재 커맨드
    r, c = jwa[0], jwa[1]
    dr, dc = d[now]
    nr, nc = r + dr, c + dc
    if 0 <= nr < size and 0 <= nc < size:

        if now in g[0]: # 현재 세로
            arr[r][c] = "|"  # 일단 현재 좌표 세로
            if pre in g[1]:
                arr[r][c] = "+"

            if arr[nr][nc] != "|":  # 다음 좌표에 가로가 가 있으면?
                arr[nr][nc] = "+"
            else:
                arr[nr][nc] = "|"


        if now in g[1]: # 현재 가로
            arr[r][c] = "-"  # 일단 현재 좌표 가로
            if pre in g[0]:
                arr[r][c] = "+"

            if arr[nr][nc] != "-": # 다음 좌표에 세로가 가 있으면?
                arr[nr][nc] = "+"
            else:
                arr[nr][nc] = "-"

        return (nr, nc), now
    else: # 만약 좌표 튀어나갔으면 현 명령 무시
        return (r, c), pre

size = int(input())
arr = [["."] * size for _ in range(size)]
commands = list(input())

srt = (0, 0)
pre = ''


for com in commands:
    srt, pre = draw(srt, com, pre)

for rr in arr:
    print("".join(rr))
```

# B1730 판화
direction = {"U": (-1, 0), "D": (1, 0), "L": (0, -1), "R": (0, 1)}

vertical = {"U", "D"}
horizontal = {"L", "R"}

def liner(r, c, d):
    dr, dc = direction[d]
    nr, nc = r + dr, c + dc
    if 0 <= nr < size and 0 <= nc < size:
        pass
    else:
        nr, nc = r, c # 다시 원복

    if (r, c) != (nr, nc):
        for cur_r, cur_c in ((r, c), (nr, nc)):

            if arr[cur_r][cur_c] == ".":
                # 도착한 칸이 빈 칸
                if d in vertical:
                    arr[cur_r][cur_c] = "|" # 재정의 한거 쓰지 말고 직접 변경해야
                elif d in horizontal:
                    arr[cur_r][cur_c] = "-"

            elif arr[cur_r][cur_c] == "|":
                # 도착한 칸에 세로 줄
                if d in horizontal:
                    arr[cur_r][cur_c] = "+"
            elif arr[cur_r][cur_c] == "-":
                if d in vertical:
                    arr[cur_r][cur_c] = "+"

    return nr, nc # 다음 좌표 반환



# 입력
size = int(input())
cmd_lst = list(input())
arr = [["."] * size for _ in range(size)]
sr, sc = 0, 0 # 맨 왼쪽 위부터 시작

# 정답
for cmd in cmd_lst:
    sr, sc = liner(sr, sc, cmd)

for rr in arr:
    print("".join(rr))
