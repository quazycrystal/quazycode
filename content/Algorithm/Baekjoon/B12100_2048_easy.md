### 걸린 시간: 2시간 26분 (구상 8분, 구현 57분, 디버깅 1시간 21분) (실패 횟수: 1회 (1시간 47분 경 제출 틀림), 1분 차이로 시간 내 제출 실패)

### 실패 / 성공 이유
- [중복 합치기] 이미 합쳐진 칸이 연속으로 또 합쳐짐. 막으려 세운 플래그가 일부 분기에서 0으로 초기화 안 됨
- [디버깅] 커스텀 테케까지 다 맞았는데 제출 틀림. 디버깅만 1시간 21분
- [함수 설계] 스택 로직은 같은데 방향별 탐색 범위가 달라 리팩토링 중 멘붕

### 코드 개선 방법
- [큐] 스택에 넣을 때가 아니라 큐에서 뺄 때 합치면 분기가 없음 (빠진 건 아직 연산 안 된 숫자)
- [플래그] 들어오는 숫자 자체를 플래그로 써서 분기 제거
- [분기] nesting 대신 조건을 and로 묶어 위계 줄이기
- [슬라이싱] 스택을 새 배열에 채울 땐 슬라이싱으로 붙이거나 인덱스 지정
- [문법] 중복 순열은 product(range(4), repeat=5)

### 코드
```python
# 분기 줄이고 continue 사용 하려고 더더 노력하자

from itertools import *

# 입력
size = int(input())
arr = [list(map(int, input().split())) for _ in range(size)]


# 최댓값 서치는 글로벌 변수

# 함수 설계
def combine(cmd, v):  # 리팩토링 필요;;
    new_v = [[0] * size for _ in range(size)]  # 새 배열 만든다

    if cmd == 0:  # 상, 위에 부터 서치
        for col in range(size):
            stk = []  # 스택을 열 별로 만들어
            cnt = 0
            for row in range(size):
                now = v[row][col]  # 받은 배열에서 누적해서 상변화
                if now != 0:
                    # 같은 수 들어왔을 때 연쇄적으로 어떻게 안 합쳐지게 막지?

                    if stk:

                        if stk[-1] == now:
                            if cnt == 0:  # 배타적 ㅠ
                                stk.pop()
                                stk.append(now * 2)
                                cnt = 1

                            elif cnt == 1:
                                # 1인거 먼저 체크해야ㅠ 밑에 보면 +1 해 주니까
                                stk.append(now)
                                cnt = 0


                        else:
                            stk.append(now)
                            cnt = 0 # 이 분기들에서 0 초기화 안 함 ㅠㅠ

                    else:
                        stk.append(now)
                        cnt = 0

            # 만든 스택을 새 배열에 넣자요
            i = 0

            while stk:
                new_v[i][col] = stk.pop(0)
                i += 1

    if cmd == 1:  # 하, 아래에 부터 서치
        for col in range(size):
            stk = []  # 스택을 열 별로 만들어
            cnt = 0
            for row in range(size - 1, -1, -1):  # 아래서 부터 읽으니까
                now = v[row][col]  # 받은 배열에서 누적해서 상변화
                if now != 0:
                    # 같은 수 들어왔을 때 연쇄적으로 어떻게 안 합쳐지게 막지?

                    if stk:

                        if stk[-1] == now:
                            if cnt == 0:  # 배타적 ㅠ
                                stk.pop()
                                stk.append(now * 2)
                                cnt = 1

                            elif cnt == 1:
                                # 1인거 먼저 체크해야ㅠ 밑에 보면 +1 해 주니까
                                stk.append(now)
                                cnt = 0


                        else:
                            stk.append(now)
                            cnt = 0

                    else:
                        stk.append(now)
                        cnt = 0
            # 만든 스택을 새 배열에 넣자요
            i = 1
            while stk:
                new_v[size - i][col] = stk.pop(0)

                i += 1

    if cmd == 2:  # 좌, 왼쪽 부터 서치
        for row in range(size):
            stk = []  # 스택을 열 별로 만들어
            cnt = 0
            for col in range(size):  # 원래 읽던 대로
                now = v[row][col]  # 받은 배열에서 누적해서 상변화

                if now != 0:
                    # 같은 수 들어왔을 때 연쇄적으로 어떻게 안 합쳐지게 막지?

                    if stk:

                        if stk[-1] == now:
                            if cnt == 0:  # 배타적 ㅠ
                                stk.pop()
                                stk.append(now * 2)
                                cnt = 1

                            elif cnt == 1:
                                # 1인거 먼저 체크해야ㅠ 밑에 보면 +1 해 주니까
                                stk.append(now)
                                cnt = 0


                        else:
                            stk.append(now)
                            cnt = 0

                    else:
                        stk.append(now)
                        cnt = 0

            # 만든 스택을 새 배열에 넣자요
            i = 0
            while stk:
                new_v[row][i] = stk.pop(0)

                i += 1

    if cmd == 3:  # 우, 오른쪽 부터 서치
        for row in range(size):
            stk = []  # 스택을 열 별로 만들어
            cnt = 0
            for col in range(size - 1, -1, -1):  # 뒤에서 부터 읽으니까
                now = v[row][col]  # 받은 배열에서 누적해서 상변화
                if now != 0:
                    # 같은 수 들어왔을 때 연쇄적으로 어떻게 안 합쳐지게 막지?

                    if stk:

                        if stk[-1] == now:
                            if cnt == 0:  # 배타적 ㅠ
                                stk.pop()
                                stk.append(now * 2)
                                cnt = 1

                            elif cnt == 1:
                                # 1인거 먼저 체크해야ㅠ 밑에 보면 +1 해 주니까
                                stk.append(now)
                                cnt = 0


                        else:
                            stk.append(now)
                            cnt = 0

                    else:
                        stk.append(now)
                        cnt = 0
            # 만든 스택을 새 배열에 넣자요
            i = 1
            while stk:
                new_v[row][size - i] = stk.pop(0)

                i += 1

    return new_v


ans = 0

for i in range(4):
    for j in range(4):
        for k in range(4):
            for l in range(4):
                for m in range(4):
                    v1 = combine(i, arr)
                    v2 = combine(j, v1)
                    v3 = combine(k, v2)
                    v4 = combine(l, v3)
                    v5 = combine(m, v4)

                    mx = 0

                    for rr in v5:
                        mx = max(mx, max(rr))  # 열별로 맥스 뽑아 갱신 - 하나의 경우의 수가 됨

                    ans = max(ans, mx)

print(ans)
```
