### 코드 개선 방법
- [방향] 모두 아래를 보고 시작. 좀비는 벽이면 제자리에서 뒤돌고, 아리는 제자리
- [순서] 아리 이동 → 불 켜기 → 충돌 판정 → 좀비 이동 → 충돌 판정
- [자료구조] 좀비는 dict로 관리, arr는 있냐 없냐만 표시

### 코드
```python
# B23031 으어어... 에이쁠 주세요

direction = {0: (1, 0), 1: (0, -1), 2: (-1, 0), 3: (0, 1)}
reverse = {0: 2, 1: 3, 2: 0, 3: 1}
def is_range(r, c):
    return 0 <= r < size and 0 <= c < size

# 1. 아리 움직임 + 불 켜기
def ari_move(cmd, ari):
    # 이동
    sr, sc, d = ari
    nd = d
    stun = 0
    if cmd == "F":
        dr, dc = direction[d]
        nr, nc = sr + dr, sc + dc
        if is_range(nr, nc):
            sr, sc = nr, nc

    elif cmd == "L":
        nd = (d - 1) % 4
    elif cmd == "R":
        nd = (d + 1) % 4

    # 불 켜기
    if arr[sr][sc] == -1:
        for r in range(sr - 1, sr + 2):
            for c in range(sc - 1, sc + 2):
                if is_range(r, c):
                    light[r][c] = 1


    # 충돌 판정
    if z_arr[sr][sc] > 0 and light[sr][sc] == 0:
        stun = 1

    return (sr, sc, nd), stun

# 2. 좀비 움직임
def zombi_move():
    new_arr = [[0] * size for _ in range(size)]
    for zi in range(1, 1+idx):
        zr, zc, zd = z_info[zi]
        dr, dc = direction[zd]
        nzr, nzc = zr + dr, zc + dc
        if not is_range(nzr, nzc):
            zd = reverse[zd]
            dr, dc = direction[zd]
            nzr, nzc = zr + dr, zc + dc

        # dict 갱신
        z_info[zi] = [nzr, nzc, zd]
        # arr 갱신
        new_arr[zr][zc] = zi # 어차피 있냐 없냐만 따지므로 여러 마리 겹쳐도 상관 X

    return new_arr

# 입력
size = int(input())
commands = list(input())
arr = [list(input()) for _ in range(size)]
light = [[0] * size for _ in range(size)]
z_arr = [[0] * size for _ in range(size)]

# 좀비 정보
z_info = {}
idx = 0
for i in range(size):
    for j in range(size):
        if arr[i][j] == "Z":
            idx += 1
            z_info[idx] = [i, j, 0]
            z_arr[i][j] = idx
        elif arr[i][j] == "S":
            arr[i][j] = -1
        elif arr[i][j] == "O":
            arr[i][j] = 0

# 아리는 항상 1, 1 (0,0)
ari = (0, 0, 0)

# 풀이
flag = 0
for cmd in commands:
    ari, stun = ari_move(cmd, ari)
    if stun:
        flag = 1
        break
    else:
        z_arr = zombi_move()

if flag:
    print("Aaaaaah!")
else:
    print("Phew...")
```
