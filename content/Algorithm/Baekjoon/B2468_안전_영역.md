### 실패 / 성공 이유
- [초기화] 높이 k가 바뀔 때 v 초기화를 빼먹어서 탐색을 안 하고 끝남
- [범위] 0부터 mx까지 봐야 다 잠긴 경우도 포함
- [엣지 케이스] 다 잠기면 영역 리스트가 비어서 max 못 씀

### 코드 개선 방법
- [범위] 안전하게 다 포함하는 범위로 돌기

### 코드
```python
def bfs(sx, sy, maxi):
    global visit
    visit += 1
    v[sx][sy] = visit
    q = []
    q.append((sx, sy))
    # 구현
    while q:
        cur = q.pop(0)
        x, y = cur[0], cur[1]
        # 방문, 범위, 조건(k보다 크면) q 갱신

        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):  # 상하 좌우
            nx = x + dx
            ny = y + dy
            if 0 <= nx < size and 0 <= ny < size:
                if v[nx][ny] == 0 and arr[nx][ny] > maxi:  #
                    v[nx][ny] = visit
                    q.append((nx, ny))  # 체계 루프 더 확실히!


# 입력
size = int(input())
arr = [list(map(int, input().split())) for _ in range(size)]
mx = 0

v = [[0] * size for _ in range(size)]
ans = []

# 영역의 최댓값 찾아두기
for i in range(size):
    for j in range(size):
        if arr[i][j] > mx:
            mx = arr[i][j]

for k in range(0, mx+1):
    v = [[0] * size for _ in range(size)]  # k가 바뀔 땐 바뀌어야
    # 단지 문제 처럼
    visit = 0

    for y in range(size):
        for x in range(size):
            if v[x][y] == 0 and arr[x][y] > k:
                bfs(x, y, k)
    ans.append(visit)
    ans.reverse() # list 자체가 뒤집힘

if ans == []:
    answer = 0 # 안전 지대 없이 다 잠김
else:
    answer = max(ans) # max 빈 리스트엔 못 씀 !!!!11 !!!!!!!!!!!!!!!!
print(answer)
```
