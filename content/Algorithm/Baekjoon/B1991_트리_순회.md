### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [조건] 노드 이름은 A부터 대문자

### 코드 개선 방법
- [자료구조] 노드를 dict에 담기
- [분기] neg 조건(안 되는 것 먼저 return)으로 분기 줄이기

### 코드
```python
def pre_ord(n, s):
    if 0 <= n < node:
        if s == "":
            return
        else:
            lst1.append(s)

        if d[s][0] != ".": # neg 조건 써보자... 분기 줄이자
            pre_ord(n+1, d[s][0])

        if d[s][1] != ".": # neg 조건 써보자... 분기 줄이자
            pre_ord(n+1, d[s][1])

        if d[s][0] == ".":
            if d[s][1] == ".":
                return

            in_ord(n + 1, d[s][0])

        if s == "":
            return
        else:
            lst2.append(s)


        if d[s][1] != ".":  # neg 조건 써보자... 분기 줄이자

        if d[s][0] == ".":
            if d[s][1] == ".":
                return
# 후위
def post_ord(n, s):
    if 0 <= n < node:

        if d[s][0] != ".":  # neg 조건 써보자... 분기 줄이자
            post_ord(n + 1, d[s][0])

        if d[s][1] != ".":  # neg 조건 써보자... 분기 줄이자
            post_ord(n + 1, d[s][1])

        if s == "":
            return
        else:
            lst3.append(s)

        if d[s][0] == ".":
            if d[s][1] == ".":
                return


node = int(input())
d = {} # 노드 담아 둘 딕셔너리
for i in range(node):
    p, c1, c2 = map(str, input().split()) # 각 요소를 인자에 적용 하므로...
    d[p] = (c1, c2)
```
