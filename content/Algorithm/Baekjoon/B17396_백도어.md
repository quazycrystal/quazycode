### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [조건] 보이는 분기(1)는 갈 수 없음. 단 마지막 노드는 항상 보임

### 코드 개선 방법
- [전처리] 마지막 노드를 갈 수 있는 곳(0)으로 바꾸면 조건 분기가 간단해짐
- [그래프] 인접 리스트엔 안 보이는 곳만, 양방향으로 채우기
- [가지치기] 꺼낸 거리가 테이블 값보다 크면 컷 (등호 주의)

### 코드
```python
from heapq import *
# 연결 양방향, 보이는지 안 보이는지 중요

nodes, edges = map(int, input().split())
visible = list(map(int, input().split()))

# 인접 리스트
adj = [[] for _ in range(nodes)] # 다음 좌표 여기서

# 인접 채우자 - 안 보이는 곳만!
for _ in range(edges):
    start, end, weight = map(int, input().split())
    if visible[start] == 1 and start != nodes-1:
        continue
    if visible[end] == 1 and end != nodes-1:
        continue

     # 양방향
    adj[start].append((weight, end))
    adj[end].append((weight, start))

# 다익스트라 테이블!!!!!!!! - 2중일땐 arr 같이 생긴거
INF = float('inf')

dj = [0] + [INF] * (nodes-1)

# 함수 - 다익스트라
hq = []
heappush(hq, (0, 0))
while hq:
    dist, now = heappop(hq)

    if dist <= dj[now]: # 일단 지금까지 최단으로 왔는지? ### 등호!!!!
        for nxt_weight, nxt in adj[now]:
            if dj[nxt] > dj[now] + nxt_weight:
                dj[nxt] = dj[now] + nxt_weight
                heappush(hq, (dj[nxt], nxt))

ans = dj[nodes-1]
if ans == INF:
    ans = -1
print(ans)

import heapq

# 0 현재 / 1 ~ N - 2 중간분기 / N-1 도착지
N, M = map(int, input().split())
# 1은 보인다. 0은 보이지 않는다. 1이라면 갈 수 없다.
branches = list(map(int, input().split()))
branches[N-1] = 0 # 갈 수 있는 곳으로 바꾸기


adj_lst = [[] for _ in range(N)]
for _ in range(M):
    a, b, t = map(int, input().split())

    # 노드, 간선 가지치기
    if (branches[a] == 1 or branches[b] == 1):
        continue

    adj_lst[a].append((t, b))
    adj_lst[b].append((t, a))

dist = [10 ** 18] * N  # 거리 배열
hq = [(0, 0)]
heapq.heapify(hq)
while hq:
    cost, node = heapq.heappop(hq)

    if dist[node] < cost: # 여기는 크면 안농
        continue

    # 가지치기
    if node == N - 1: # 여기서 가지쳐도 되나보구나! 근데 위험할 듯, 뒤에서도 오니까
        break

    for w, nxt in adj_lst[node]:
        if dist[nxt] > w + cost:  # 비용이 더 싸다면
            if branches[nxt] == 0:  # 0인 곳만
                # 일종의 방문 표시 / 갱신
                dist[nxt] = w + cost
                heapq.heappush(hq, (w + cost, nxt))


if dist[N - 1] != 10 ** 18:
    print(dist[N - 1])
else:
    print(-1)
```
