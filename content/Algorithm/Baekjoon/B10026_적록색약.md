### 실패 / 성공 이유
- [방문 처리] 방문 표시는 v에 따로 하고 두 번째 탐색 전에 리셋

### 코드 개선 방법
- [전처리] 적록 연결은 for문으로 미리 처리하거나 함수가 rgb를 다 보게

def bfs(sr, sc):
    global visit
    visit += 1
    v[sr][sc] = visit
    q = []
    q.append((sr, sc))
    while q:
        cur = q.pop(0)
        r, c = cur[0], cur[1]
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr = r + dr
            nc = c + dc
            if 0 <= nr < size and 0 <= nc < size:
                if v[nr][nc] == 0: # 방문 안 했으면
                    if arr[r][c] == arr[nr][nc]: # 색깔 같으면
                        v[nr][nc] = visit
                        q.append((nr, nc))

size = int(input())
arr = [list(input()) for _ in range(size)]
v = [[0] * size for _ in range(size)]
visit = 0
for a in range(size):
    for b in range(size):
        if v[a][b] == 0: # 방문 안 했으면
            bfs(a, b)

RGB = v
mx_RGB = max(map(max, RGB))
