### 코드
```python
num, goal = map(int, input().split())
lst = list(map(int, input().split()))
sm_lst = []
mn_sub = goal # n은 3 이상이므로 sub은 무조건 goal보다 작음
for i in range (num-2):
    for j in range (i+1, num-1):
        for k in range (j+1, num):
            sm = lst[i] + lst[j] + lst[k]
            sm_lst.append(sm)

for l in sm_lst:
    sub = goal - l
    if sub < mn_sub and sub >= 0:
        mn_sub = sub
ans = goal - mn_sub
print(ans)
```
