### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [방문 처리] 벽을 부순 경로와 안 부순 경로를 같은 방문 배열로 비교하면 안 됨

### 코드 개선 방법
- [상태 확장] (r, c, 부쉈는지)로 방문 배열 분리
- [거리] 시작 칸과 끝 칸 포함

### 코드
```python
# B2206 벽 부수고 이동하기

from collections import deque

def bfs(sr, sc):
    crash = [[2] * col for _ in range(row)]
    # v = [[-1] * col for _ in range(row)]
    crash[sr][sc] = 0
    q = deque()
    q.append((sr, sc, 0, 1))
    # 시작 칸과 끝 칸도 포함해서 센다
    while q:
        r, c, use, can = q.popleft()
        if (r, c) == (row-1, col-1):
            return can
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < row and 0 <= nc < col:
                if crash[nr][nc] <= use: continue
                if arr[nr][nc] == 0:
                    crash[nr][nc] = use
                    q.append((nr, nc, use, can + 1))

                elif arr[nr][nc] == 1:
                    if crash[nr][nc] > use:
                        crash[nr][nc] = use + 1
                        q.append((nr, nc, use + 1, can + 1))
    return -1

# 입력
row, col = map(int, input().split())
arr = [list(map(int, input())) for _ in range(row)]

# 풀이
ans = bfs(0, 0)
print(ans)
```
