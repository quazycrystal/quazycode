### 코드
```python
from collections import deque
hole = [(-1, 0),
       (-1, 1),
       (0, 1),
       (1, 1),
       (1, 0),
       (0, -1)] # 열 홀수: 상 우상 우 우하 하 좌

jack = [(-1, 0),
       (-1, -1),
       (0, -1),
       (1, -1),
       (1, 0),
       (0, 1)] # 열 짝수: 상, 좌상, 좌, 좌하, 하, 우

def bfs(sr, sc):
    arr[sr][sc] = 0 # 방문표시
    q = deque()
    q.append((sr, sc))
    while q:
        cur = q.popleft()
        r, c = cur[0], cur[1]
        if r % 2 == 0: # 열이 짝수
            direction = jack
        else: # 열이 홀수
            direction = hole

        for dr, dc in direction:
            # row 가 짝수면 1, 1(우 하) 못 감
            nr = r + dr
            nc = c + dc
            if 0 <= nr < row and 0 <= nc < col:
                if arr[nr][nc] == -1: # 안 갔던, 갈 수 있는 곳이면
                    arr[nr][nc] = arr[r][c] + 1
                    q.append((nr, nc))

row, col, num = map(int, input().split())
arr = [[-1] * col for _ in range(row)]

for _ in range(num):
    r, c = map(int, input().split())
    arr[r][c] = -2 # 장애물 표시
bfs(0, 0)
print(arr[row-1][col-1])


```
