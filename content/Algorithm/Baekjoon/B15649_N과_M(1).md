### 코드

```python
def btk(d, lst):
    if d == M:
        print(*lst)
        return
    for i in range(1, N+1):
        if v[i] == 0:
            v[i] = 1
            lst.append(i)
            btk(d+1, lst)
            lst.pop()
            v[i] = 0

N, M = map(int, input().split())
v = [0] * (N+1)
btk(0, [])

```
