### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [문제 읽기] 점프 범위는 상하좌우가 아니라 맨해튼 거리(절댓값)
- [문제 읽기] 점프 횟수에 따라 색이 바뀜. 잉크가 안 나와도 색은 바뀜

### 코드 개선 방법
- [분기] 잉크가 없으면 돌지 않기

### 코드
```python
d = {"U":(-1, 0), "D":(1, 0), "L":(0, -1), "R":(0, 1)} #상하좌우
def INK(com, jwa): # 커맨드, 좌표
    # 이동
    global ink
    global jump

    r, c = jwa[0], jwa[1] # 밑에서도 쓴다
    nxtr, nxtc = r, c

    # 이동
    if com == "U" or com == "D" or com == "L" or com == "R":
        dr, dc = d[com]
        nr, nc = r+dr, c+dc

        if 0 <= nr < size and 0 <= nc < size:
            # 장애물 = 알파벳 또는 #, 평지라면,,,
            if arr[nr][nc] == ".":
                arr[nr][nc] = "@"
                arr[r][c] = "."
                nxtr, nxtc = nr, nc

    # 잉크 충전
    if com == "j":
        ink += 1

    if com == "J":
        c_idx = jump % (len(colors))
        jump += 1
        color = colors[c_idx]

        if ink != 0: # 잉크 없으면 안 돌아도 됨
            for j_row in range(size): #잉크가 든 범위만큼
                for j_col in range(size):
                    if 0 < abs(j_row-r) + abs(j_col-c) <= ink:
                        if arr[j_row][j_col] != ".":
                            arr[j_row][j_col] = color
            ink = 0


    return ((nxtr, nxtc))

# 입력
colors_l, size, com_l = map(int, input().split())
colors = list(input())
arr = [list(input()) for _ in range(size)]
commands = list(input())

ink = 0 # 현재 잉크 양
jump = 0 # 이 전 까지의 점프 횟수

srt = tuple()
# 현재 위치 찾기
for row in range(size):
    for col in range(size):
        if arr[row][col] == "@":
            srt = (row, col)

for colors_idx in range(com_l):
    command = commands[colors_idx]
    srt = INK(command, srt)

for rr in arr:
    print("".join(rr))

```
