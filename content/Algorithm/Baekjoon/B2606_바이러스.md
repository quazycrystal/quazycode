### 코드
```python
# https://www.acmicpc.net/problem/2606
def dfs1(s):
    global ans
    v[s] = 1
    for nxt in adj[s]:
        if v[nxt] == 0:
            ans += 1 # 첫 node인 1은 제외 (1번 컴퓨터를 통한 이므로)
            dfs1(nxt)

num = int(input())
edge = int(input())
adj = [[] for _ in range (num+1)] # v+1, 0 안 씀
ans = 0
v = [0] * (num+1)
for _ in range (edge): # 줄 갯수
    temp = list(map(int, input().split()))
    adj[temp[0]].append(temp[1])
    adj[temp[1]].append(temp[0]) # 그림에서는 아무튼 양방향
dfs1(1)
print(ans)

```
