### 걸린 시간: 약 2시간 3분 (13:50 - 15:53, 구상 10분, 구현 1시간 10분, 디버깅 약 40분) (실패 횟수: -)

### 실패 / 성공 이유
- [동시 갱신] 냄새 뿌리는 시점이 어긋남. 루프 끝에 찍은 arr가 똑같아야 하는데 나중에 갱신된다고 멋대로 생각
- [문제 읽기] 내 냄새 칸도 빈 칸도 없을 때 처리가 명시 안 됨 → 이동 불가로 추측해 해결

### 코드 개선 방법
- [자료구조] 상어는 dict, 냄새는 arr. 겹치면 번호 작은 상어만 남김
- [우선순위] 냄새 없는 칸 → 내 냄새 칸 순. 현 방향별 우선순위 테이블
- [디버깅] 어디에 print하느냐가 성과를 결정. 루프 맨 끝에 찍기
- [구상] 자료구조뿐 아니라 언제 갱신되는지, 언제 동시 처리할지도 구상

def is_range(r, c):
    return 0 <= r < size and 0 <= c < size

# 함수 - 상어 이동

def shark_move(idx, dirr, r, c):
    # 가장 번호 작은 상어로 계속 물갈이 하면 됨
    now_prefer = prefer_dir[idx][dirr-1]
    no_smell = ()
    yes_smell = ()

    for dirr_num in now_prefer:
        dr, dc = direction[dirr_num]
        nr, nc = r + dr, c + dc
        if is_range(nr, nc):
            if not no_smell and smell[nr][nc] == [0, 0]:
               no_smell = (nr, nc, dirr_num)
            if not yes_smell and smell[nr][nc][0] == idx:
                yes_smell = (nr, nc, dirr_num)

    # 냄새 뿌리기, 상어 좌표 갱신 - 잡아먹기
    nxtr, nxtc, nxtd = 0, 0, 0
    if no_smell:
        nxtr, nxtc, nxtd = no_smell
    elif yes_smell:
        nxtr, nxtc, nxtd = yes_smell

    if new_shark[nxtr][nxtc] == 0 or new_shark[nxtr][nxtc] > idx:
        new_shark[nxtr][nxtc] = idx
        new_shark[r][c] = 0
        now_dir[idx-1] = nxtd
        # 방향 바로 바꿔주고 냄새도 뿌리기
        smell[nxtr][nxtc] = [idx, smell_last]



# 입력
size, shark_num, smell_last = map(int, input().split())
shark = [list(map(int, input().split())) for _ in range(size)]
now_dir = list(map(int, input().split()))

# 대응 방향들,,,
direction = {1: (-1, 0), 2: (1, 0), 3: (0, -1), 4: (0, 1)}
prefer_dir = {}

for idx in range(1, shark_num+1):
    d1 = list(map(int, input().split()))
    d2 = list(map(int, input().split()))
    d3 = list(map(int, input().split()))
    d4 = list(map(int, input().split()))
    prefer_dir[idx] = [d1, d2, d3, d4]

smell = [[[0, 0] for _ in range(size)] for _ in range(size)]


# 초기 냄새:
for i in range(size):
    for j in range(size):
        if shark[i][j] > 0:
            smell[i][j] = [shark[i][j], smell_last]

#
time = 0
while True:
    time += 1
    new_shark = [line[:] for line in shark]


    for r in range(size):
        for c in range(size):
            if shark[r][c] > 0:
                idx = shark[r][c]
                dirr = now_dir[idx-1]
                shark_move(idx, dirr, r, c)

    # break point?
    shark = new_shark

    for rr in range(size):
        for cc in range(size):
            if smell[rr][cc][1] == 0:
                smell[rr][cc] = [0, 0]

            if smell[rr][cc][0] > 0 and smell[rr][cc][1] > 0:
                smell[rr][cc][1] -= 1

    for ccc in smell:
        print(ccc)

    print("new_shark")
    for rrr in new_shark:
        print(rrr)

    print("dir", now_dir)

print(time)

def shark_move(sr, sc, idx):

    d = shark_d[idx] # 현 방향
    search_d = prior[idx][d] # 현 방향에서의 우선순위
    no_smell = ()
    yes_smell = ()
    for now_d in search_d:
        dr, dc = direction[now_d]
        nr, nc = sr + dr, sc + dc
        if is_range(nr, nc):
            if smell[nr][nc] == [0, 0]:
                no_smell = (nr, nc, now_d)
                break
            if smell[nr][nc][0] == idx and not yes_smell:
                # 맨 처음 나온거!!
                yes_smell = (nr, nc, now_d)

    if no_smell: # 냄새 없는게 우선
        # 방향, 좌표 갱신
        shark_d[idx] = no_smell[2]

        else:

time = 0
