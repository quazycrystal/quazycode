### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [좌표] m(세로), n(가로) 헷갈림. 원본은 (0,0)이 왼쪽 아래, 행렬은 왼쪽 위인데 넓이만 구하면 돼서 상관 없음
- [로직] 영역별 넓이라 단순하게 합해서는 안 됨

### 코드 개선 방법
- [DFS] 이중 for문으로 돌다가 방문 안 한 0에서 탐색 시작. 시작 횟수 = 영역 개수

```python
def dfs(x, y):
    global ans
    global sm
    v[x][y] = 1
    sm += 1 # 연속된 빈 공간의 합
    for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)): # 상하 좌우
        nx = x+dx
        ny = y+dy
        if 0 <= nx < n and 0 <= ny < m:
            if arr[nx][ny] == 0 and v[nx][ny] == 0: # 방문도 안 하고 0인 칸
                dfs(nx, ny) # 여기서 어차피 다 1이면 안 불러짐

m, n, k = map(int, input().split())
arr = [[0] * m for _ in range(n)] # 세로만큼 곱하고 행 만듦
paper = [list(map(int, input().split())) for _ in range(k)]
v = [[0] * m for _ in range(n)] # 방문 좌표 수
ans = []
sm = 0

for row in paper:
    for i in range(row[0], row[2]):
        for j in range(row[1], row[3]):
            arr[i][j] = 1

for a_x in range(n):
    for a_y in range(m):
        if arr[a_x][a_y] == 0 and v[a_x][a_y] == 0: # 미방문 0!!!
            dfs(a_x, a_y)
            ans.append(sm)


print(ans)
```

```python
# 영역 구하기
def bfs(sr, sc):
    v[sr][sc] = visit
    q = []
    q.append((sr, sc))
    while q:
        cur = q.pop(0)
        r = cur[0]
        c = cur[1]
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr = r + dr
            nc = c + dc
            if 0 <= nr < row and 0 <= nc < col:
                if arr[nr][nc] == 0 and v[nr][nc] == 0:  # 안 덮인 부분의 넓이 재야 함
                    v[nr][nc] = visit
                    q.append((nr, nc))


# 입력 받기
row, col, area = map(int, input().split())
visit = 0
ans = []
areas = 0

v = [[0] * col for _ in range(row)]
arr = [[0] * col for _ in range(row)]

for _ in range(area):
    x1, y1, x2, y2 = map(int, input().split())
    for rr in range(y1, y2):
        for cc in range(x1, x2):  # 원본 그림에서 뒤집힌 형태,
            arr[rr][cc] = 1

for i in range(row):
    for j in range(col):
        if arr[i][j] == 0 and v[i][j] == 0:
            visit += 1
            bfs(i, j)

for k in range(1, visit + 1):
    areas = 0
    for row in v:
        areas += row.count(k)  # 2차원 함수는 한번에 못 셈
    ans.append(areas)

ans.sort()
print(visit)
print(*ans)
```
