### 코드
```python
def free(n, money):
    global ans
    if n == N:
        ans.append(money)
        return

    cur = lst[n]

    day, pay = cur[0], cur[1]
    # 포함
    if n + day <= N:
        free(n + day, money + pay)
    # 미포함
    free(n + 1, money)


N = int(input())
lst = []
for _ in range(N):
    day, pay = map(int, input().split())
    lst.append((day, pay))
ans = []

free(0, 0)
print(max(ans))
```
