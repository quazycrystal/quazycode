### 걸린 시간: 3시간 38분 (16:18 - 22:46 중, 구상 24분, 구현 1시간 28분 + 1시간 44분) (실패 횟수: 금요일 시험 중 버린 문제)

### 실패 / 성공 이유
- [구현] 말기(roll)에 1시간 28분. 가로/세로 갱신 타이밍을 못 잡음 → 홀짝으로 나누니 됨
- [인덱스] 상대/절대 좌표, 뒤집어 읽기 하나하나에 시간 소모
- [시간 복잡도] 5000턴으로 크게 잡았는데 실제는 20턴 내외

### 코드 개선 방법
- [슬라이싱] 말기는 리스트를 쪼개 2차원으로 합치기만. 가로 길이만큼 잘라 뒤집어 붙이면 세로는 알아서 생김
- [자료구조] 전체를 2차원으로 보지 말고 돌아다니는 부분(리스트 묶음) + 아래 1차원으로 보기
- [함수 설계] 90도, 180도 회전 함수 분리
- [동시 갱신] 인접 어항은 큰 쪽에서 작은 쪽으로만
- [검증] 정사각형 조각은 잘못 돌려도 답이 나올 수 있으니 직사각형 케이스로 확인

### 코드
```python
# 간결한 코드에 비해 생각보다 시간은 얼마 차이 안 나넹

def roll(lst, num):
    row = num // 2
    col = num

    arr = [[0] * col for _ in range(row)]
    arr[0] = lst
    garo = 1
    sero = 1 # 세로가 먼저 는다


    # 가로만큼 읽어서
    start_c = 0
    times = 0
    flag = 0
    while True:
        read = [line[start_c:start_c+garo] for line in arr[:sero]]

        if start_c + sero + garo > col:
            flag = 1
            break

        if flag == 0:
            # 열 clear 함수
            for r in range(sero):
                for c in range(start_c, start_c + garo):
                    arr[r][c] = 0

            # 새로 시작할 부분 넘기기
            start_c += garo

            # read 90도 회전
            read = [list(line[::-1]) for line in zip(*read)]

            for newr in range(garo):
                for newc in range(sero): # 상대 좌표
                    set_r = newr + 1
                    set_c = start_c + newc
                    arr[set_r][set_c] = read[garo - newr - 1][sero - newc - 1]

            if times % 2 == 0:
                sero += 1
            else:
                garo += 1

            times += 1


    return arr, row, col

# 3, 6의 인접 어항 처리 개체 조절 함수
def adj(arr, row, col):
    diff_num = [[0] * col for _ in range(row)]
    for r in range(row):
        for c in range(col):
            if arr[r][c] > 0:
                for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)): # 이거 방향 2개만 해도 됨
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < row and 0 <= nc < col:
                        if arr[r][c] > arr[nr][nc]:
                            if arr[r][c] > 0 and arr[nr][nc] > 0:
                                share = (arr[r][c] - arr[nr][nc]) // 5
                                diff_num[r][c] -= share
                                diff_num[nr][nc] += share
    for r in range(row):
        for c in range(col):
            diff_num[r][c] += arr[r][c]

    return diff_num, row, col

# 4, 6의 쭉쭉 펴주는 함수
def spread(arr, row, col):
    new_tanks = [0] * tank_num

    i = 0
    for c in range(col):
        for r in range(row):
            if arr[r][c] > 0:
                new_tanks[i] = arr[r][c]
                i += 1

    return new_tanks

# 5의 돌리고 쌓아서 가로 2, 세로 N//2로 만들어주는 함수
def turn_stack(lst):
    half = tank_num // 2
    half_half = half // 2

    arr1 = [[0] * half for _ in range(2)]

    # 첫번째 돌려 쌓기
    for up in range(half-1, -1, -1):
        arr1[0][half - up - 1] = lst[up]
    for down in range(half):
        arr1[1][down] = lst[down + half]

    # 두번째 돌려 쌓기
    arr2 = [[0] * half_half for _ in range(4)]

    half_arr1 = [line[half_half-1::-1] for line in arr1[::-1]]

    half_arr2 = [line[half_half:half] for line in arr1]

    for r in range(2):
        for up_c in range(half_half):
            arr2[r][up_c] = half_arr1[r][up_c]

    for r in range(2):
        for down_c in range(half_half):
            # 또 상대 / 절대 좌표!!!
            arr2[r + 2][down_c] = half_arr2[r][down_c] # 인덱스 주의

    return arr2, 4, half_half


# 다 배열 받고, 배열 반환

# 입력
tank_num, diff = map(int, input().split())
tanks = list(map(int, input().split()))

# 풀이
ans = 0
while True:
    ans += 1
    # 일단 완탐하면서 추가
    mn_tank = min(tanks)
    for i in range(tank_num):
        if tanks[i] == mn_tank:
            tanks[i] += 1

    # 말아요
    roll_arr, roll_r, roll_c = roll(tanks, tank_num)

    # 근처로 옮겨요
    adj_arr, adj_r, adj_c = adj(roll_arr, roll_r, roll_c)

    # 펴요
    new_tanks = spread(adj_arr, adj_r, adj_c)


    # 돌리고 쌓고
    turn_arr, turn_r, turn_c = turn_stack(new_tanks)

    # 근처로 옮겨요
    turn_adj, adjt_r, adjt_c = adj(turn_arr, turn_r, turn_c)

    turn_adj = turn_adj[::-1]
    final_tanks = spread(turn_adj, adjt_r, adjt_c)

    fish_diff = max(final_tanks) - min(final_tanks)

    if fish_diff <= diff:
        break

    # 리스트 갱신
    tanks = final_tanks

print(ans)
```
