### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [방문 처리] 시작점 방문 표시를 빼먹어서 엣지 케이스
- [방문 처리] v에 +1 누적하면 한 번 돌 때 여러 레벨을 셈
- [좌표] y = 행, x = 열

### 코드 개선 방법
- [로직] arr == 1일 때만 호출하므로 0은 신경 안 써도 됨
- [재귀] setrecursionlimit은 BFS에는 필요 없음

from collections import deque
def bfs(sx, sy): #sz는 사이즈... 변수명 고정 요망, ex, ey는 없어도 됨, 필요할 때마다 부를거라
    # 초기화: 방문 표시, 큐, 큐에 추가
    global danji
    danji += 1 # 여기에 넣으니까 엣지케이스 생기는데 왜?
    # v[sx][sy] = 1 # 이거!!!!! 방문표시!!!!
    v[sx][sy] = danji
    q = deque()
    q.append((sx, sy))
    # 구현
    while q:
        cur = q.popleft() # 정말 중요하다........ 현 위치 설정
        x, y = cur[0], cur[1]
        for dx, dy in ((-1, 0), (1, 0), (0, -1), (0, 1)): # 상하좌우
            nx = x+dx
            ny = y+dy
            if 0 <= nx < size and 0 <= ny < size:

                if v[nx][ny] == 0 and arr[nx][ny] == 1:
                    v[nx][ny] = danji
                    q.append((nx, ny))

# 입력 받기
size = int(input())
# arr, 방문록
arr = [list(map(int, input())) for _ in range(size)]
v = [[0] * size for _ in range(size)]

danji = 0

for y in range(size):
    for x in range(size):
        if arr[y][x] == 1 and v[y][x] == 0: # 방문 안 한 거
            bfs(y, x) # 근데 돈 횟수 만큼 v[] 갱신할 때 곱해져야,..?

sm = 0
num = []
for k in range(1, danji+1):
    sm = 0
    for row in v:
        sm += row.count(k)
    num.append(sm)
num.sort()

print(danji)
for l in range(len(num)):
    print(num[l])
