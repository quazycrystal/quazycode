### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [인덱스] 톱니 번호도 1부터. s는 1, N은 0
- [로직] 맞닿은 톱니가 같으면 다음으로 안 감. 다를 때만 큐에 추가
- [방향] c_dir = -c_dir로 덮어쓰지 말고 반대 방향 값을 따로 넘기기

### 코드 개선 방법
- [속도] bfs에서 바뀌는 톱니만 갱신. 2번, 6번 값만 저장한 리스트 사용
- [초기화] v는 부를 때마다 갱신, 시작점은 한 번만 돌리기

### 코드
```python
from pprint import pprint
from collections import deque
def check(): # 이건 전체 체크
    for topni in range(top_num):
        temp = []
        for idx in (6, 2): # 이건 range 아니고..
            temp.append(arr[topni][idx])
            if len(temp) == 2:
                lst.append(temp)

def bingle(n, d):
    if d == 1:
        last = arr[n].pop()
        arr[n].insert(0, last) # insert, object
    elif d == -1:
        first = arr[n].pop(0)
        arr[n].append(first)

# 리소스 절약 위해 bfs에서 바뀌는 애들만 갱신?
def bfs(num, direct):
    v = [0] * top_num # v 여러 번 부르니까 부를 때 마다 갱신
    v[num] = 1
    q = deque()
    q.append((num, direct))
    bingle(num, direct)  # 일단 나 돌리고 (아직 리스트 업뎃 안 되어서 괜찮음)
    # 시작점은 한 번만 돌려야 한다

    while q:
        # 인덱스 정의
        pop = q.popleft()
        cur = pop[0]
        c_dir = pop[1]

        l_nxt = cur - 1
        r_nxt = cur + 1

        # 현 톱니의 오른쪽 왼쪽
        left = lst[cur][0] # 6번
        right = lst[cur][1] # 2번

        # 맞닿은 톱니 인덱스 범위 체크
        if 0 <= l_nxt and v[l_nxt] == 0:
            v[l_nxt] = 1 # 방문 체크
            nxt_left = lst[l_nxt][1] # 2번 인덱스

            # 맞닿은 톱니가 같은 지 다른지 판별
            if left != nxt_left: # 왼쪽 끼리 닿는다
                # c_dir = -c_dir # 반대로 회전,,,
                bingle(l_nxt, -c_dir)
                q.append((l_nxt, -c_dir))  # 한 발짝 멀리

        if r_nxt < top_num and v[r_nxt] == 0:
            v[r_nxt] = 1
            nxt_right = lst[r_nxt][0] # 6번 인덱스

            # 맞닿은 톱니가 같은 지 다른지 판별
            if right != nxt_right: # 오른쪽 끼리 닿는다
                # c_dir = -c_dir 정의 하지 말고!!!
                bingle(r_nxt, -c_dir) # 반대로 회전,,,
                q.append((r_nxt, -c_dir))

# 입력
top_num = int(input())
arr = [list(map(int, input())) for _ in range(top_num)]
calls_num = int(input())
calls = []
ans = 0
for _ in range(calls_num):
    num, direction = map(int, input().split())
    calls.append((num-1, direction)) # 인덱스 1부터 들어오므로

lst = [] # 필요한 톱니만 저장한 리스트, 6, 2 순으로 저장되어 있음

# 핵심 풀이
for com in calls:
    number, command = com[0], com[1]
    lst = [] # 갱신 전 리스트 초기화
    check() # 돌리기 전 바뀐 톱니 다시 읽어들여 리스트 갱신
    bfs(number, command)

# 정답
for rr in arr:
    if rr[0] == 1:
        ans += 1
print(ans)
```
