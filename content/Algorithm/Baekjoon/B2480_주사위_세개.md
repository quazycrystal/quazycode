### 코드
```python
lines = []
while True:
    try:
        lines.append(input()) # 입력 값 그대로 들어감 (자동 이중 리스트 x)
    except EOFError:
        break

for test_case in lines:
    answer = 0
    x, y, z = map(int, test_case.split())
    if x == y == z:
        answer = 10000 + (x * 1000)
    elif x == y != z or x != y == z:
        answer = 1000 + (y * 100)
    elif z == x != y:
        answer = 1000 + (z * 100)
    else:
        answer = 100 * max (x, y, z)

    print(answer)
```
