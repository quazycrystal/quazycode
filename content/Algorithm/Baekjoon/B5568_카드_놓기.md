### 코드
```python
def btk(d, s):
    if d == choose:
        if len(s) > 0:
            ans.add(s)
        return
    for idx in range(cards):
        if v[idx] == 0:
            v[idx] = 1
            btk(d+1, s + card_lst[idx])
            v[idx] = 0

# 입력
cards = int(input())
choose = int(input())
card_lst = [input() for _ in range(cards)]
ans = set()

v = [0] * cards
btk(0, '')
print(len(ans))

```
