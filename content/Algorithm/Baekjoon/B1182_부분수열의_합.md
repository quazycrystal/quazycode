### 코드
```python
def btk(n, sm):
    global ans
    if n == N:
        if sm == S:
            ans += 1
        return

    btk(n+1, sm+lst[n])
    btk(n+1, sm)

ans = 0
N, S = map(int, input().split())
lst = list(map(int, input().split()))
btk(0, 0)
if S == 0:
    ans -= 1
print(ans)

```
