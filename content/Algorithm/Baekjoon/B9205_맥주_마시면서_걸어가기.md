### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [문제 읽기] 50m씩 세는 게 아니라 맨해튼 거리 1000 이하면 이동 가능
- [좌표] x=c, y=r로 받았으면 c, r 순서로 bfs. 일관성이 중요

### 코드 개선 방법
- [그래프] 집, 편의점, 페스티벌을 노드로 보고 거리 1000 이하면 연결

### 코드
```python
def beer(sr, sc, er, ec):
    global ans
    v[(sr, sc)] = 1
    q = []
    q.append((sr, sc))
    while q:
        cur = q.pop(0)
        r, c = cur[0], cur[1]
        if abs(r-er) + abs(c-ec) <= 1000: # 편의점에서 페스티벌까지 갈 수 있다면
            ans = "happy"
            return

        else: # 다른 편의점에 갈 수 있는가?
            if con:
                for combi in con: # 변수 이름 잘 지을 방법 없나, 변수 섞임 이슈
                    if combi != (r, c) and combi not in v: #방문 안 하고 자기 자신이 아니면
                        nr = combi[0]
                        nc = combi[1]
                        if abs(r-nr) + abs(c-nc) <= 1000:
                            v[(nr, nc)] = 1
                            q.append((nr, nc))

    ans = "sad"
    return

t = int(input())

for tc in range(t):
    c_num = int(input())

    hr, hc = map(int, input().split()) # 집의 좌표
    con = [] # 주어진 편의점 다 넣어
    ans = ""
    for _ in range(c_num):  # 다 노드로 치고 갈 수 있는지 없는지?
        # x1- x2 + y1 - y2가 1000 이하면 갈 수 있음
        cc, rr = map(int, input().split())
        con.append((cc, rr))
    fr, fc = map(int, input().split())  # 페스티벌 좌표
    v = {}
    sm = 0 # 갈 수 있는 편의점 없는 flag

    beer(hr, hc, fr, fc)

    print(ans)

    # print(v)
```
