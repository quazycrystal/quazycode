### 걸린 시간: - (실패 횟수: -)

### 실패 / 성공 이유
- [문제 읽기] 시험 때 힌트를 이해 못 함. 0이면 계속 퍼지고 1이면 멈춤 = 가중치
- [입력] 초기 좌표, str 주의

### 코드 개선 방법
- [0-1 BFS] 가중치 0이면 덱 맨 앞, 1이면 맨 뒤. 한 번의 bfs로 끝남
- [다익스트라] 테이블 값보다 나을 때만 갱신, 같을 때는 갱신 불필요

# 0-1 bfs
def bfs(sr, sc):
    global ans
    v[sr][sc] = 1
    q = []
    q.append((sr, sc))
    while q:
        r, c = q.pop(0) # 0 항상 주의ㅜㅠ
        #  상태 변화, 정답 여기서
        if arr[r][c] == '1':
            arr[r][c] = '0'

        if (r, c) == (g_r, g_c):
            ans = v[r][c]
            return

        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc

            if 0 <= nr < row and 0 <= nc < col:
                if v[nr][nc] == 0:
                    if arr[nr][nc] == '0':
                        v[nr][nc] = v[r][c]
                        q.insert(0, (nr, nc)) # 가중치 0이면 q의 맨 앞에
                    elif arr[nr][nc] != '0': # 목적지도 포함...
                        v[nr][nc] = v[r][c] + 1
                        q.append((nr, nc)) # 가중치 있으면 q의 맨 뒤에 -> 한 레벨에 한 번만 동작

from heapq import *
def bfs(sr, sc, gr, gc):
    hq = []
    heappush(hq, (0, sr, sc))
    while hq:
        w, r, c = heappop(hq)
        # 정답 처리
        if (r, c) == (gr, gc):
            return

        # 중복 계산 막자
        if w <= v[r][c]: # 지금 테이블에 든 값보다 나은 경우

            for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                nr, nc = r + dr, c + dc

                if 0 <= nr < row and 0 <= nc < col:
                    if v[nr][nc] > w + arr[nr][nc]: # 여기는 비용 같을 때 갱신할 필요가 없다
                        v[nr][nc] = w + arr[nr][nc]
                        heappush(hq, (v[nr][nc], nr, nc))
