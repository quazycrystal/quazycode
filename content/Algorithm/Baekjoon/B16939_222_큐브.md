### 걸린 시간: 1시간 21분 (9:30 - 10:51), 리팩토링 18분 (실패 횟수: -)

### 실패 / 성공 이유
- [로직] 축이 3개, 축마다 쓰지 않는 면이 있음
- [초기화] 돌릴 때마다 초기화 주의

### 코드 개선 방법
- [구현] 덱으로 돌리는 대신 현 상태만 보고 판단
- [문법] all(x == lst[0] for x in lst) 버릇 들이기
- [입력] input도 dict로

### 코드
```python
same_side = {1: [1, 2, 3, 4],
             2: [5, 6, 7, 8],
             3: [9, 10, 11, 12],
             4: [13, 14, 15, 16],
             5: [17, 18, 19, 20],
             6: [21, 22, 23, 24]}

rows = {'x1': [[1, 3], [5, 7], [9, 11], [22, 24]],
        'x2': [[2, 4], [6, 8], [10, 12], [21, 23]],
        'y1': [[13, 14], [5, 6], [17, 18], [21, 22]],
        'y2': [[15, 16], [7, 8], [19, 20], [23, 24]],
        'z1': [[1, 2], [18, 20], [11, 12], [13, 15]],
        'z2': [[3, 4], [17, 19], [9, 10], [14, 16]]}

axis = ["X", "Y", "Z"]

stat = {x + 1: y for x, y in enumerate(map(int, input().split()))}


def side_check(s):
    flag = 0
    tpl = ()

    if s == "X":
        tpl = (4, 5)
    if s == "Y":
        tpl = (1, 3)
    if s == "Z":
        tpl = (2, 6)

    for i in tpl:
        x = stat[same_side[i][0]]
        if all(x == stat[name] for name in same_side[i]):
            flag += 1

    if flag == 2:
        return True
    else:
        return False


def row_check(s):
    TF = 0  # 전체 조건 만족 여부 체크
    temp = []

    if s == "X":
        temp.append(rows['x1'])
        temp.append(rows['x2'])
    if s == "Y":
        temp.append(rows['y1'])
        temp.append(rows['y2'])
    if s == "Z":
        temp.append(rows['z1'])
        temp.append(rows['z2'])

    r1, r2 = temp[0], temp[1]
    # 각 요소 별 색 같은지 체크
    flag = 0
    for i in r1:
        if stat[i[0]] != stat[i[1]]:
            flag = 1
            break
    if flag == 0:
        for j in r2:
            if stat[j[0]] != stat[j[1]]:
                flag = 1
                break

    if flag == 0:
        if all(stat[r1[k % 4][0]] == stat[r2[(k + 1) % 4][0]] for k in range(4)):
            TF = 1

        if all(stat[r1[k % 4][0]] == stat[r2[(k - 1) % 4][0]] for k in range(4)):
            TF = 1

    if TF == 1:
        return True
    else:
        return False

ans = 0

for com in axis:
    check1 = side_check(com)
    if check1:
        check2 = row_check(com)
        if check2:
            ans = 1
            break

print(ans)
```
