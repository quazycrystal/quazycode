import heapq
def djk(s, e): # 꼭 시작이 1이 아니어도 그냥 쓰면 됨
    hq = []
    heapq.heappush(hq, (0, s))
    while hq:
        dis, cur = heapq.heappop(hq)
        if d[cur] >= dis:
            for n_dis, nxt in adj[cur]:
                if d[nxt] > d[cur] + n_dis:
                    heapq.heappush(hq, (d[cur] + n_dis, nxt))
                    d[nxt] = d[cur] + n_dis

        if cur == e:
            break

nodes = int(input())
edges = int(input())
d = [100001]*(nodes+1)
adj = [[] for _ in range(nodes+1)] # 1부터 시작

for _ in range(edges):
    s, e, w = map(int, input().split())
    adj[s].append((w, e))
    adj[s].append((w, e)) # 버스면 양방향? 언급 없음....

st, end = map(int, input().split())
djk(st, end)
print(d[end])
