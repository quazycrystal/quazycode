### 코드

```python
# 유기농 배추
def bfs(sr, sc):
    v[sr][sc] = visit
    q = []
    q.append((sr, sc))
    while q:
        cur = q.pop(0)
        r = cur[0]
        c = cur[1]
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr = r + dr
            nc = c + dc
            if 0 <= nr < row and 0 <= nc < col:
                if arr[r][c] == 1 and v[nr][nc] == 0:
                    v[nr][nc] = visit
                    q.append((nr, nc))


# 입력:
T = int(input())
for _ in range(T):
    col, row, num = map(int, input().split())
    arr = [[0] * col for _ in range(row)]
    v = [[0] * col for _ in range(row)]
    visit = 0

    for _ in range(num):  # arr에 배추 채우기
        y, x = map(int, input().split())  # 행렬 주의
        arr[x][y] = 1

    # bfs 시작 (최소의 이므로)
    for rr in range(row):
        for cc in range(col):
            if arr[rr][cc] == 1 and v[rr][cc] == 0:
                visit += 1
                bfs(rr, cc)

    print(visit)
```
